# US-04 test-only Pop-demand multipliers used by archived injection probes.
# Evaluated from Pop scope only when the isolated probe package supplies the
# experimental vanilla pop_demand database object.
# Missing, disabled, uninitialized, or invalid ModeU5 state returns 1 (vanilla).

cbp_us04_live_pop_demand_multiplier_wheat = {
	value = 1
	if = {
		limit = {
			has_global_variable = cbp_pop_demand_live_integration_enabled
			has_global_variable = cbp_us04_multiplier_initialization_version
			global_var:cbp_us04_multiplier_initialization_version >= 1
			location = {
				has_variable_map = cbp_pop_demand_multiplier
				is_key_in_variable_map = {
					name = cbp_pop_demand_multiplier
					target = goods:wheat
				}
			}
		}
		value = {
			location = {
				value = "variable_map(cbp_pop_demand_multiplier|goods:wheat)"
			}
		}
	}
}
