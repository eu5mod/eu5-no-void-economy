cbp_debug_run_pr182_market_stockpile_capacity_probe = {
	remove_global_variable = test_cbp_pr182_market_stockpile_probe_passed
	remove_global_variable = test_cbp_pr182_market_stockpile_probe_failed
	remove_global_variable = test_cbp_pr182_market_stockpile_probe_blocked
	remove_global_variable = test_cbp_pr182_market_stockpile_probe_active

	set_variable = { name = cbp_pr182_stockpile_probe_country_count value = 0 }
	set_variable = { name = cbp_pr182_stockpile_probe_capacity value = 0 }
	set_variable = { name = cbp_pr182_stockpile_probe_fixed_modifier_seen value = 0 }

	debug_log = "ModeU5 TEST ENTERED scenario=pr182_market_stockpile_capacity_probe"

	if = {
		limit = { NOT = { has_global_variable = cbp_economy_rebalance_loaded } }
		set_global_variable = test_cbp_pr182_market_stockpile_probe_blocked
		debug_log = "ModeU5 TEST BLOCKED scenario=pr182_market_stockpile_capacity_probe reason=economy_package_not_initialized"
	}
	else = {
		save_temporary_scope_as = cbp_pr182_probe_country
		capital.market = { save_temporary_scope_as = cbp_pr182_probe_market }

		cbp_prepare_promoted_market_capacity_cache = {
			market = scope:cbp_pr182_probe_market
		}

		if = {
			limit = { has_global_variable_list = cbp_countries_present_in_market }
			every_in_global_list = {
				variable = cbp_countries_present_in_market
				save_temporary_scope_as = cbp_pr182_capacity_country
				if = {
					limit = {
						has_variable_map = cbp_stock_cap_by_market
						is_key_in_variable_map = {
							name = cbp_stock_cap_by_market
							target = scope:cbp_pr182_probe_market
						}
					}
					save_temporary_scope_value_as = {
						name = cbp_pr182_current_country_capacity
						value = "variable_map(cbp_stock_cap_by_market|scope:cbp_pr182_probe_market)"
					}
					scope:cbp_pr182_probe_country = {
						change_variable = {
							name = cbp_pr182_stockpile_probe_country_count
							add = 1
						}
						change_variable = {
							name = cbp_pr182_stockpile_probe_capacity
							add = scope:cbp_pr182_current_country_capacity
						}
					}
				}
			}
		}

		set_global_variable = test_cbp_pr182_market_stockpile_probe_active
		cbp_refresh_market_center_stockpile_capacity = {
			market = scope:cbp_pr182_probe_market
		}
		remove_global_variable = test_cbp_pr182_market_stockpile_probe_active

		scope:cbp_pr182_probe_market.location = {
			remove_location_modifier = cbp_market_stockpile_capacity
			add_location_modifier = {
				modifier = cbp_market_stockpile_capacity
				months = 2
				mode = replace
				size = 1234
				recalculate_immediately = yes
			}
			if = {
				limit = { has_location_modifier = cbp_market_stockpile_capacity }
				scope:cbp_pr182_probe_country = {
					set_variable = {
						name = cbp_pr182_stockpile_probe_fixed_modifier_seen
						value = 1
					}
				}
			}
		}

		debug_log = "ModeU5 PR-182 STOCKPILE_PROBE computed_capacity=[THIS.GetVariable('cbp_pr182_stockpile_probe_capacity').GetValue|2] countries=[THIS.GetVariable('cbp_pr182_stockpile_probe_country_count').GetValue|0] fixed_modifier_seen=[THIS.GetVariable('cbp_pr182_stockpile_probe_fixed_modifier_seen').GetValue|0] fixed_size=1234"

		if = {
			limit = {
				var:cbp_pr182_stockpile_probe_country_count > 0
				var:cbp_pr182_stockpile_probe_capacity > 0
				var:cbp_pr182_stockpile_probe_fixed_modifier_seen = 1
			}
			set_global_variable = test_cbp_pr182_market_stockpile_probe_passed
			debug_log = "ModeU5 TEST PASS scenario=pr182_market_stockpile_capacity_probe"
		}
		else = {
			set_global_variable = test_cbp_pr182_market_stockpile_probe_failed
			debug_log = "ModeU5 TEST FAIL scenario=pr182_market_stockpile_capacity_probe"
		}
	}

	remove_variable = cbp_pr182_stockpile_probe_country_count
	remove_variable = cbp_pr182_stockpile_probe_capacity
	remove_variable = cbp_pr182_stockpile_probe_fixed_modifier_seen
}
