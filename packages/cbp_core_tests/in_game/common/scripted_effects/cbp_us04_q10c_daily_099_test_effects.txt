# US-04 Q10c daily 0.99 replacement cache/lifecycle diagnostic.
#
# This test holds the destructive REPLACE:pop_demand source at 0.99 and logs
# books demand plus wool control for ten consecutive days. It is diagnostic only:
# it should reveal whether demand updates intra-month, on a specific day, or only
# at a later monthly refresh.

cbp_us04_q10c_begin = {
	debug_log = "ModeU5 TEST ENTERED scenario=us04_q10c_replace_pop_demand_daily_099"
	if = {
		limit = { has_global_variable = cbp_us04_q10c_daily_099_in_progress }
		set_global_variable = test_cbp_us04_q10c_daily_099_blocked
		debug_log = "ModeU5 TEST BLOCKED scenario=us04_q10c_replace_pop_demand_daily_099 reason=probe_already_in_progress"
	}
	else = {
		cbp_us04_q10c_clear_state = yes
		set_global_variable = cbp_us04_q10c_daily_099_in_progress
		capital.market = { save_scope_as = cbp_us04_q10c_market }
		set_global_variable = { name = cbp_us04_q9_replace_global_books value = 1.0 }
		debug_log = "ModeU5 US-04 VANILLA DEMAND DUMP q10c_daily_099 phase=begin source=1.0000 wait=one_day"
	}
}

cbp_us04_q10c_capture_baseline_set_099 = {
	scope:cbp_us04_q10c_market = {
		save_temporary_scope_value_as = { name = cbp_us04_q10c_books_current value = "goods_demand_in_market(goods:books)" }
		save_temporary_scope_value_as = { name = cbp_us04_q10c_control_current value = "goods_demand_in_market(goods:wool)" }
	}
	set_global_variable = { name = cbp_us04_q10c_books_baseline value = scope:cbp_us04_q10c_books_current }
	set_global_variable = { name = cbp_us04_q10c_control_baseline value = scope:cbp_us04_q10c_control_current }
	set_variable = { name = cbp_us04_q10c_ui_books value = global_var:cbp_us04_q10c_books_baseline }
	set_variable = { name = cbp_us04_q10c_ui_control value = global_var:cbp_us04_q10c_control_baseline }
	debug_log = "ModeU5 US-04 VANILLA DEMAND DUMP q10c_daily_099 day=00 source=1.0000 books=[THIS.GetVariable('cbp_us04_q10c_ui_books').GetValue|4] control_wool=[THIS.GetVariable('cbp_us04_q10c_ui_control').GetValue|4]"
	set_global_variable = { name = cbp_us04_q9_replace_global_books value = 0.99 }
	debug_log = "ModeU5 US-04 VANILLA DEMAND DUMP q10c_daily_099 next_source=0.9900 hold_days=10"
}
