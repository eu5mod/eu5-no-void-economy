	save_temporary_scope_value_as = { name = cbp_us04_original_cmm_present value = 0 }
	save_temporary_scope_value_as = { name = cbp_us04_original_cmm_value value = 0 }

		limit = {
			has_variable_map = cmm
			is_key_in_variable_map = {
				name = cmm
				target = flag:no_void_economy__cbp_economic_balance_design_pop_consumption_influenced_by_offer_demand_settings
			}
		}

		limit = { scope:cbp_us04_original_cmm_present > 0 }

			name = cmm
