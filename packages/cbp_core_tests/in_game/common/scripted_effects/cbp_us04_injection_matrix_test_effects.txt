cbp_us04_matrix_begin = {
	debug_log = "ModeU5 TEST ENTERED scenario=us04_pop_demand_injection_matrix"
	if = {
		limit = { has_global_variable = cbp_us04_matrix_in_progress }
		set_global_variable = test_cbp_us04_pop_demand_injection_matrix_blocked
		debug_log = "ModeU5 TEST BLOCKED scenario=us04_pop_demand_injection_matrix reason=matrix_already_in_progress"
	}
	else = {
		cbp_us04_matrix_clear_state = yes
		set_global_variable = { name = cbp_us04_matrix_pass_count value = 0 }
		set_global_variable = { name = cbp_us04_matrix_fail_count value = 0 }
		set_global_variable = { name = cbp_us04_matrix_blocked_count value = 0 }
		if = {
			limit = { NOT = { cbp_economy_rebalance_loaded_trigger = yes } }
			set_global_variable = test_cbp_us04_pop_demand_injection_matrix_blocked
			debug_log = "ModeU5 TEST BLOCKED scenario=us04_pop_demand_injection_matrix reason=economy_package_not_loaded"
		}
		else = {
			set_global_variable = cbp_us04_matrix_in_progress
			cbp_us04_matrix_capture_original_state = yes
			set_global_variable = cbp_pop_demand_live_integration_enabled
			set_global_variable = { name = cbp_us04_multiplier_initialization_version value = 1 }
			cbp_us04_matrix_write_neutral = yes
			debug_log = "ModeU5 US-04 INJECTION MATRIX PHASE begin neutral_sources_written=1 scheduled_low_capture=1"
		}
	}
}

cbp_us04_matrix_evaluate_01_wheat = {
	save_temporary_scope_value_as = { name = cbp_us04_matrix_wheat_delta value = { value = global_var:cbp_us04_matrix_wheat_high subtract = global_var:cbp_us04_matrix_wheat_low } }
	set_global_variable = { name = cbp_us04_matrix_wheat_delta value = scope:cbp_us04_matrix_wheat_delta }
	set_variable = { name = cbp_us04_matrix_ui_low value = global_var:cbp_us04_matrix_wheat_low }
	set_variable = { name = cbp_us04_matrix_ui_high value = global_var:cbp_us04_matrix_wheat_high }
	set_variable = { name = cbp_us04_matrix_ui_delta value = global_var:cbp_us04_matrix_wheat_delta }
	set_variable = { name = cbp_us04_matrix_ui_source_low value = global_var:cbp_us04_matrix_wheat_source_low }
	set_variable = { name = cbp_us04_matrix_ui_source_high value = global_var:cbp_us04_matrix_wheat_source_high }
	debug_log = "ModeU5 US-04 INJECTION CANDIDATE id=01 syntax=plain_child good=wheat source_low=[THIS.GetVariable('cbp_us04_matrix_ui_source_low').GetValue|4] source_high=[THIS.GetVariable('cbp_us04_matrix_ui_source_high').GetValue|4] low=[THIS.GetVariable('cbp_us04_matrix_ui_low').GetValue|4] high=[THIS.GetVariable('cbp_us04_matrix_ui_high').GetValue|4] delta=[THIS.GetVariable('cbp_us04_matrix_ui_delta').GetValue|4]"
	if = { limit = { global_var:cbp_us04_matrix_wheat_low <= 0.0001 } cbp_us04_matrix_increment_blocked = yes debug_log = "ModeU5 US-04 INJECTION RESULT id=01 syntax=plain_child good=wheat BLOCKED reason=baseline_demand_nonpositive" }
	else_if = { limit = { NOT = { has_global_variable = cbp_us04_matrix_control_stable } } cbp_us04_matrix_increment_blocked = yes debug_log = "ModeU5 US-04 INJECTION RESULT id=01 syntax=plain_child good=wheat BLOCKED reason=unstable_control" }
	else_if = { limit = { global_var:cbp_us04_matrix_wheat_high > global_var:cbp_us04_matrix_wheat_low } cbp_us04_matrix_increment_pass = yes debug_log = "ModeU5 US-04 INJECTION RESULT id=01 syntax=plain_child good=wheat PASS" }
	else = { cbp_us04_matrix_increment_fail = yes debug_log = "ModeU5 US-04 INJECTION RESULT id=01 syntax=plain_child good=wheat FAIL reason=no_target_response" }
}
