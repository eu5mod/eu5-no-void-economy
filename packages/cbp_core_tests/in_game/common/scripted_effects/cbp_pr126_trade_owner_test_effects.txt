# PR126 PR6 country trade-owner pass probe.
#
# This probe validates the country-scoped trade-owner pass surface and a controlled
# explicit inter-market transfer request. It does not switch the live monthly
# dispatcher. Native every_trade quantities are converted through generated
# literal-good transport helpers for diagnostics only; stock-affecting mutation
# remains on the explicit ModeU5 inter-market request path in PR6.1.
#
# Vanilla trade cannot have source_market == target_market. Same-market local
# stock consumption is therefore intentionally absent from this probe.

cbp_pr126_trade_owner_ensure_metric_globals = {
	if = { limit = { NOT = { has_global_variable = cbp_country_trade_owner_pass_runs } } set_global_variable = { name = cbp_country_trade_owner_pass_runs value = 0 } }
	if = { limit = { NOT = { has_global_variable = cbp_country_trade_owner_trades_seen } } set_global_variable = { name = cbp_country_trade_owner_trades_seen value = 0 } }
	if = { limit = { NOT = { has_global_variable = cbp_country_trade_owner_inter_market_trades } } set_global_variable = { name = cbp_country_trade_owner_inter_market_trades value = 0 } }
	if = { limit = { NOT = { has_global_variable = cbp_country_trade_owner_owner_mismatches } } set_global_variable = { name = cbp_country_trade_owner_owner_mismatches value = 0 } }
	if = { limit = { NOT = { has_global_variable = cbp_country_trade_owner_quantity_confirmed_trades } } set_global_variable = { name = cbp_country_trade_owner_quantity_confirmed_trades value = 0 } }
	if = { limit = { NOT = { has_global_variable = cbp_country_trade_owner_quantity_blocked_trades } } set_global_variable = { name = cbp_country_trade_owner_quantity_blocked_trades value = 0 } }
	if = { limit = { NOT = { has_global_variable = cbp_country_trade_owner_explicit_requests } } set_global_variable = { name = cbp_country_trade_owner_explicit_requests value = 0 } }
	if = { limit = { NOT = { has_global_variable = cbp_country_trade_owner_inter_market_transfers } } set_global_variable = { name = cbp_country_trade_owner_inter_market_transfers value = 0 } }
	if = { limit = { NOT = { has_global_variable = cbp_country_trade_owner_requested_quantity } } set_global_variable = { name = cbp_country_trade_owner_requested_quantity value = 0 } }
	if = { limit = { NOT = { has_global_variable = cbp_country_trade_owner_confirmed_trade_quantity } } set_global_variable = { name = cbp_country_trade_owner_confirmed_trade_quantity value = 0 } }
	if = { limit = { NOT = { has_global_variable = cbp_country_trade_owner_transferred_quantity } } set_global_variable = { name = cbp_country_trade_owner_transferred_quantity value = 0 } }
	if = { limit = { NOT = { has_global_variable = cbp_country_trade_owner_unsatisfied_quantity } } set_global_variable = { name = cbp_country_trade_owner_unsatisfied_quantity value = 0 } }
}

cbp_pr126_trade_owner_select_two_markets = {
	remove_global_variable = test_cbp_pr126_trade_owner_market_selection_failed
	set_local_variable = { name = test_cbp_pr126_trade_owner_market_index value = 0 }

	every_market_present_in_country = {
		if = {
			limit = {
				scope:cbp_pr126_trade_owner_test_country = {
					local_var:cbp_pr126_trade_owner_market_index = 0
				}
			}
			save_temporary_scope_as = cbp_pr126_trade_owner_source_market
			# Default target to source so later comparisons never dereference an unset
			# target. A second market replaces this below; otherwise the probe blocks.
			save_temporary_scope_as = cbp_pr126_trade_owner_target_market
		}
		else_if = {
			limit = {
				scope:cbp_pr126_trade_owner_test_country = {
					local_var:cbp_pr126_trade_owner_market_index = 1
				}
			}
			save_temporary_scope_as = cbp_pr126_trade_owner_target_market
		}
		scope:cbp_pr126_trade_owner_test_country = {
			change_local_variable = { name = test_cbp_pr126_trade_owner_market_index add = 1 }
		}
	}

	if = {
		limit = { NOT = { exists = scope:cbp_pr126_trade_owner_source_market } }
		set_global_variable = test_cbp_pr126_trade_owner_market_selection_failed
	}
	else_if = {
		limit = { scope:cbp_pr126_trade_owner_source_market = scope:cbp_pr126_trade_owner_target_market }
		set_global_variable = test_cbp_pr126_trade_owner_market_selection_failed
	}
}

cbp_debug_run_pr126_country_trade_owner_pass_test = {
	remove_global_variable = test_cbp_pr126_country_trade_owner_pass_passed
	remove_global_variable = test_cbp_pr126_country_trade_owner_pass_failed
	remove_global_variable = test_cbp_pr126_country_trade_owner_pass_blocked
	remove_global_variable = test_cbp_pr126_trade_owner_market_selection_failed

	remove_variable = gui_cbp_pr126_ui_trade_owner_pass_runs
	remove_variable = gui_cbp_pr126_ui_trade_owner_trades_seen
	remove_variable = gui_cbp_pr126_ui_trade_owner_inter_market
	remove_variable = gui_cbp_pr126_ui_trade_owner_owner_mismatches
	remove_variable = gui_cbp_pr126_ui_trade_owner_confirmed
	remove_variable = gui_cbp_pr126_ui_trade_owner_blocked
	remove_variable = gui_cbp_pr126_ui_trade_owner_explicit_requests
	remove_variable = gui_cbp_pr126_ui_trade_owner_inter_market_transfers
	remove_variable = gui_cbp_pr126_ui_trade_owner_requested
	remove_variable = gui_cbp_pr126_ui_trade_owner_confirmed_quantity
	remove_variable = gui_cbp_pr126_ui_trade_owner_transferred
	remove_variable = gui_cbp_pr126_ui_trade_owner_unsatisfied

	save_temporary_scope_value_as = { name = test_cbp_perf14_original_cmm_main_present value = 0 }
	save_temporary_scope_value_as = { name = test_cbp_perf14_original_cmm_main_value value = 0 }
	if = {
		limit = {
			has_variable_map = cmm
			is_key_in_variable_map = {
				name = cmm
				target = flag:no_void_economy__cbp_no_void_economy_main
			}
		}
		save_temporary_scope_value_as = { name = test_cbp_perf14_original_cmm_main_present value = 1 }
		save_temporary_scope_value_as = {
			name = test_cbp_perf14_original_cmm_main_value
			value = "variable_map(cmm|flag:no_void_economy__cbp_no_void_economy_main)"
		}
	}

	debug_log = "ModeU5 TEST ENTERED scenario=pr126_country_trade_owner_pass"

	save_temporary_scope_as = cbp_pr126_trade_owner_test_country
	cbp_perf14_set_cmm_main_mode_for_test = { value = 2 }
	cbp_reset_country_trade_owner_pass_metrics = yes
	cbp_pr126_trade_owner_select_two_markets = yes

	if = {
		limit = { has_global_variable = test_cbp_pr126_trade_owner_market_selection_failed }
		set_global_variable = test_cbp_pr126_country_trade_owner_pass_blocked
		debug_log = "ModeU5 TEST BLOCKED scenario=pr126_country_trade_owner_pass reason=not_enough_distinct_current_country_markets"
	}
	else = {
		cbp_run_monthly_country_trade_owner_cycle = yes

		cbp_perf14_clear_wheat_stock_in_market = {
			market = scope:cbp_pr126_trade_owner_source_market
		}
		cbp_perf14_clear_wheat_stock_in_market = {
			market = scope:cbp_pr126_trade_owner_target_market
		}
		cbp_debug_replace_country_wheat_capacity = {
			country = scope:cbp_pr126_trade_owner_test_country
			market = scope:cbp_pr126_trade_owner_source_market
			value = 100
		}
		cbp_debug_replace_country_wheat_capacity = {
			country = scope:cbp_pr126_trade_owner_test_country
			market = scope:cbp_pr126_trade_owner_target_market
			value = 100
		}
		cbp_add_stock = {
			country = scope:cbp_pr126_trade_owner_test_country
			market = scope:cbp_pr126_trade_owner_source_market
			good = wheat
			quantity = 10
			capacity_policy = allow_over_capacity
		}

		cbp_process_country_trade_owner_explicit_request = {
			buyer_country = scope:cbp_pr126_trade_owner_test_country
			source_market = scope:cbp_pr126_trade_owner_source_market
			target_market = scope:cbp_pr126_trade_owner_target_market
			good = wheat
			requested_quantity = 4
		}
	}

	cbp_pr126_trade_owner_capture_metrics = yes
	debug_log = "ModeU5 PR126 TRADE_OWNER pass_runs=[THIS.GetVariable('cbp_pr126_ui_trade_owner_pass_runs').GetValue|0] trades_seen=[THIS.GetVariable('cbp_pr126_ui_trade_owner_trades_seen').GetValue|0] inter_market=[THIS.GetVariable('cbp_pr126_ui_trade_owner_inter_market').GetValue|0] owner_mismatches=[THIS.GetVariable('cbp_pr126_ui_trade_owner_owner_mismatches').GetValue|0] confirmed_trades=[THIS.GetVariable('cbp_pr126_ui_trade_owner_confirmed').GetValue|0] blocked=[THIS.GetVariable('cbp_pr126_ui_trade_owner_blocked').GetValue|0] confirmed_quantity=[THIS.GetVariable('cbp_pr126_ui_trade_owner_confirmed_quantity').GetValue|2] explicit_requests=[THIS.GetVariable('cbp_pr126_ui_trade_owner_explicit_requests').GetValue|0] inter_market_transfers=[THIS.GetVariable('cbp_pr126_ui_trade_owner_inter_market_transfers').GetValue|0] requested=[THIS.GetVariable('cbp_pr126_ui_trade_owner_requested').GetValue|2] transferred=[THIS.GetVariable('cbp_pr126_ui_trade_owner_transferred').GetValue|2] unsatisfied=[THIS.GetVariable('cbp_pr126_ui_trade_owner_unsatisfied').GetValue|2]"

	if = {
		limit = {
			NOT = { has_global_variable = test_cbp_pr126_country_trade_owner_pass_blocked }
			global_var:cbp_country_trade_owner_pass_runs = 1
			global_var:cbp_country_trade_owner_explicit_requests = 1
			global_var:cbp_country_trade_owner_inter_market_transfers = 1
			global_var:cbp_country_trade_owner_requested_quantity = 4
			global_var:cbp_country_trade_owner_transferred_quantity >= 4
			global_var:cbp_country_trade_owner_unsatisfied_quantity = 0
		}
		set_global_variable = test_cbp_pr126_country_trade_owner_pass_passed
		debug_log = "ModeU5 TEST PASS scenario=pr126_country_trade_owner_pass"
	}
	else_if = {
		limit = {
			NOT = { has_global_variable = test_cbp_pr126_country_trade_owner_pass_blocked }
		}
		set_global_variable = test_cbp_pr126_country_trade_owner_pass_failed
		debug_log = "ModeU5 TEST FAIL scenario=pr126_country_trade_owner_pass"
	}

	cbp_perf14_restore_cmm_main_mode_after_test = yes
}
