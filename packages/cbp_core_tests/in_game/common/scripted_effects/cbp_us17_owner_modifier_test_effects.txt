# Focused US-17 native trade-profit modifier probe.
# Run through event cbp_us17_owner_modifiers.1.

cbp_us17_owner_modifiers_clear_test_results = {
	remove_global_variable = test_cbp_us17_owner_modifiers_started
	remove_global_variable = test_cbp_us17_owner_modifiers_passed
	remove_global_variable = test_cbp_us17_owner_modifiers_failed
	remove_global_variable = test_cbp_us17_owner_modifiers_blocked
}

cbp_us17_owner_modifiers_note_failure = {
	save_temporary_scope_value_as = {
		name = test_cbp_us17_owner_modifiers_failure_count
		value = { value = scope:test_cbp_us17_owner_modifiers_failure_count add = 1 }
	}
}

cbp_us17_assert_native_correction_near = {
	if = {
		limit = {
			OR = {
				scope:$actual$ < $minimum$
				scope:$actual$ > $maximum$
			}
		}
		cbp_us17_owner_modifiers_note_failure = yes
		error_log = "ModeU5 US17 NATIVE_MODIFIERS ASSERT FAIL assertion=$assertion$"
	}
}

cbp_debug_run_us17_owner_modifier_probe = {
	cbp_us17_owner_modifiers_clear_test_results = yes
	set_global_variable = test_cbp_us17_owner_modifiers_started
	debug_log = "ModeU5 TEST ENTERED scenario=us17_trade_owner_modifiers"
	debug_log = "ModeU5 US17 NATIVE_MODIFIERS AUDIT START expected=native_ai_ui_profit_without_division"
	save_temporary_scope_value_as = { name = test_cbp_us17_owner_modifiers_failure_count value = 0 }

	# Positive baseline below the define cap: I=0.10, S=0.05, M=0.20, C=0.15.
	# Corrections are -0.10, -0.05, and C - M = -0.05.
	save_temporary_scope_value_as = { name = cbp_us17_native_baseline_import_efficiency value = 0.10 }
	save_temporary_scope_value_as = { name = cbp_us17_native_baseline_selling_efficiency value = 0.05 }
	save_temporary_scope_value_as = { name = cbp_us17_native_baseline_maintenance_efficiency value = 0.20 }
	cbp_compute_us17_native_profit_corrections_from_baselines = yes
	cbp_us17_assert_native_correction_near = { actual = cbp_us17_native_combined_efficiency minimum = 0.149 maximum = 0.151 assertion = positive_combined_efficiency }
	cbp_us17_assert_native_correction_near = { actual = cbp_us17_native_import_correction_result minimum = -0.101 maximum = -0.099 assertion = import_cancelled }
	cbp_us17_assert_native_correction_near = { actual = cbp_us17_native_selling_correction_result minimum = -0.051 maximum = -0.049 assertion = selling_cancelled }
	cbp_us17_assert_native_correction_near = { actual = cbp_us17_native_maintenance_correction_result minimum = -0.051 maximum = -0.049 assertion = maintenance_replaced_by_combined_efficiency }
	debug_log = "ModeU5 US17 NATIVE_MODIFIERS CHECK case=positive import=0.10 selling=0.05 maintenance=0.20 combined=0.15 corrections=-0.10/-0.05/-0.05 effective_maintenance=0.15 final_maintenance_factor=0.85"

	# After the define has bounded C, its monetary-base role is common to the
	# target and effective native amounts. Verify those amounts remain equal
	# without assuming the loaded define's numerical value.
	save_temporary_scope_value_as = {
		name = test_cbp_us17_base_maintenance_cost
		value = define:NCountry|MERCHANT_MAINTENANCE_COST
	}
	save_temporary_scope_value_as = {
		name = test_cbp_us17_target_maintenance_factor
		value = {
			value = 1
			add = { value = scope:cbp_us17_native_combined_efficiency multiply = -1 }
		}
	}
	save_temporary_scope_value_as = {
		name = test_cbp_us17_effective_maintenance_factor
		value = {
			value = 1
			add = { value = scope:cbp_us17_native_baseline_maintenance_efficiency multiply = -1 }
			add = { value = scope:cbp_us17_native_maintenance_correction_result multiply = -1 }
		}
	}
	save_temporary_scope_value_as = {
		name = test_cbp_us17_target_maintenance_cost
		value = {
			value = scope:test_cbp_us17_base_maintenance_cost
			multiply = scope:test_cbp_us17_target_maintenance_factor
		}
	}
	save_temporary_scope_value_as = {
		name = test_cbp_us17_effective_maintenance_cost
		value = {
			value = scope:test_cbp_us17_base_maintenance_cost
			multiply = scope:test_cbp_us17_effective_maintenance_factor
		}
	}
	save_temporary_scope_value_as = {
		name = test_cbp_us17_maintenance_cost_delta
		value = {
			value = scope:test_cbp_us17_effective_maintenance_cost
			add = { value = scope:test_cbp_us17_target_maintenance_cost multiply = -1 }
		}
	}
	cbp_us17_assert_native_correction_near = { actual = test_cbp_us17_maintenance_cost_delta minimum = -0.001 maximum = 0.001 assertion = capped_native_maintenance_amount_equivalent }
	debug_log = "ModeU5 US17 NATIVE_MODIFIERS CHECK case=maintenance_define_policy result=equivalent define=NCountry.MERCHANT_MAINTENANCE_COST roles=combined_efficiency_cap_and_native_monetary_base"

	# Negative efficiencies remain negative. C=-0.60 replaces M=0.20 and raises
	# the maintenance factor to 1.60 through a -0.80 correction.
	save_temporary_scope_value_as = { name = cbp_us17_native_baseline_import_efficiency value = -0.40 }
	save_temporary_scope_value_as = { name = cbp_us17_native_baseline_selling_efficiency value = -0.20 }
	save_temporary_scope_value_as = { name = cbp_us17_native_baseline_maintenance_efficiency value = 0.20 }
	cbp_compute_us17_native_profit_corrections_from_baselines = yes
	cbp_us17_assert_native_correction_near = { actual = cbp_us17_native_combined_efficiency minimum = -0.601 maximum = -0.599 assertion = negative_sum_preserved }
	cbp_us17_assert_native_correction_near = { actual = cbp_us17_native_maintenance_correction_result minimum = -0.801 maximum = -0.799 assertion = negative_efficiency_increases_maintenance }
	debug_log = "ModeU5 US17 NATIVE_MODIFIERS CHECK case=negative combined=-0.60 maintenance_correction=-0.80 effective_maintenance=-0.60 final_maintenance_factor=1.60"

	# Positive sums are capped by the loaded maintenance-cost define, but never
	# lower-clamped.
	save_temporary_scope_value_as = { name = cbp_us17_native_baseline_import_efficiency value = 1.40 }
	save_temporary_scope_value_as = { name = cbp_us17_native_baseline_selling_efficiency value = 1.20 }
	save_temporary_scope_value_as = { name = cbp_us17_native_baseline_maintenance_efficiency value = 0.20 }
	cbp_compute_us17_native_profit_corrections_from_baselines = yes
	save_temporary_scope_value_as = {
		name = test_cbp_us17_define_cap_delta
		value = {
			value = scope:cbp_us17_native_combined_efficiency
			add = { value = define:NCountry|MERCHANT_MAINTENANCE_COST multiply = -1 }
		}
	}
	cbp_us17_assert_native_correction_near = { actual = test_cbp_us17_define_cap_delta minimum = -0.001 maximum = 0.001 assertion = positive_sum_capped_by_merchant_maintenance_define }

	# The non-CBP maintenance baseline is replaced even when it exceeds 100%.
	save_temporary_scope_value_as = { name = cbp_us17_native_baseline_import_efficiency value = -0.40 }
	save_temporary_scope_value_as = { name = cbp_us17_native_baseline_selling_efficiency value = -0.20 }
	save_temporary_scope_value_as = { name = cbp_us17_native_baseline_maintenance_efficiency value = 1.20 }
	cbp_compute_us17_native_profit_corrections_from_baselines = yes
	cbp_us17_assert_native_correction_near = { actual = cbp_us17_native_maintenance_correction_result minimum = -1.801 maximum = -1.799 assertion = maintenance_baseline_fully_replaced }

	# Idempotence: effective values already containing the previous CBP correction
	# must reconstruct the same 0.10/0.05/0.20 non-CBP baseline.
	save_temporary_scope_value_as = { name = cbp_us17_native_effective_import_efficiency value = 0.00 }
	save_temporary_scope_value_as = { name = cbp_us17_native_effective_selling_efficiency value = 0.00 }
	save_temporary_scope_value_as = { name = cbp_us17_native_effective_maintenance_efficiency value = 0.15 }
	save_temporary_scope_value_as = { name = cbp_us17_native_previous_import_correction value = -0.10 }
	save_temporary_scope_value_as = { name = cbp_us17_native_previous_selling_correction value = -0.05 }
	save_temporary_scope_value_as = { name = cbp_us17_native_previous_maintenance_correction value = -0.05 }
	cbp_reconstruct_us17_native_baselines_from_effective_values = yes
	cbp_compute_us17_native_profit_corrections_from_baselines = yes
	cbp_us17_assert_native_correction_near = { actual = cbp_us17_native_baseline_import_efficiency minimum = 0.099 maximum = 0.101 assertion = idempotent_import_baseline }
	cbp_us17_assert_native_correction_near = { actual = cbp_us17_native_baseline_selling_efficiency minimum = 0.049 maximum = 0.051 assertion = idempotent_selling_baseline }
	cbp_us17_assert_native_correction_near = { actual = cbp_us17_native_baseline_maintenance_efficiency minimum = 0.199 maximum = 0.201 assertion = idempotent_maintenance_baseline }
	cbp_us17_assert_native_correction_near = { actual = cbp_us17_native_maintenance_correction_result minimum = -0.051 maximum = -0.049 assertion = idempotent_recalculation }
	debug_log = "ModeU5 US17 NATIVE_MODIFIERS CHECK case=idempotence baseline=0.10/0.05/0.20 corrections=-0.10/-0.05/-0.05"

	if = {
		limit = { scope:test_cbp_us17_owner_modifiers_failure_count = 0 }
		set_global_variable = test_cbp_us17_owner_modifiers_passed
		debug_log = "ModeU5 TEST PASS scenario=us17_trade_owner_modifiers hard_failures=0 mode=native_auto_modifiers combination=sum_without_division clamp=merchant_maintenance_cost_define negative_efficiency=preserved idempotence=passed treasury_reconciliation=none"
	}
	else = {
		set_global_variable = test_cbp_us17_owner_modifiers_failed
		debug_log = "ModeU5 TEST FAIL scenario=us17_trade_owner_modifiers outcome=HARD_FAILURE see_ASSERT_FAIL_entries"
	}
}
