}

	}

	}
	else = {

		save_temporary_scope_value_as = { name = cbp_us17_native_previous_maintenance_correction value = 0 }
		if = {
			limit = { has_variable = cbp_us17_native_selling_correction }

			limit = { has_variable = cbp_us17_native_import_correction }

			limit = { has_variable = cbp_us17_native_export_correction }

		save_temporary_scope_value_as = { name = cbp_us17_native_effective_import_efficiency value = modifier:import_efficiency }

		save_temporary_scope_value_as = { name = cbp_us17_native_effective_maintenance_efficiency value = modifier:merchant_maintenance_efficiency }

		set_variable = { name = test_cbp_us17_probe_expected_route_loss_coefficient value = scope:cbp_us20_route_loss_coefficient_result }
		set_variable = { name = cbp_us20_route_loss_coefficient value = scope:cbp_us20_route_loss_coefficient_result }

		set_variable = {
			name = test_cbp_us17_probe_expected_selling
			value = { value = scope:cbp_us17_native_baseline_selling_efficiency add = scope:cbp_us17_native_selling_correction_result }
		}
		set_variable = {
			name = test_cbp_us17_probe_expected_import
			value = { value = scope:cbp_us17_native_baseline_import_efficiency add = scope:cbp_us17_native_import_correction_result }

			name = test_cbp_us17_probe_expected_export
			value = { value = scope:cbp_us17_native_baseline_export_efficiency add = scope:cbp_us17_native_export_correction_result }
		}

		cbp_refresh_us17_native_profit_modifiers_for_current_country = yes

			limit = {
				OR = {
					NOT = { has_variable = cbp_us20_route_loss_coefficient }

					NOT = { has_variable = cbp_us17_native_import_correction }

		}
		else = {
			set_global_variable = test_cbp_us17_owner_modifiers_awaiting_live_check

		}
	}
}
