			limit = { has_variable = cbp_us17_native_import_correction }

		save_temporary_scope_value_as = { name = cbp_us17_native_effective_import_efficiency value = modifier:import_efficiency }

		set_variable = { name = test_cbp_us17_probe_baseline_selling value = scope:cbp_us17_native_baseline_selling_efficiency }
		set_variable = { name = test_cbp_us17_probe_baseline_import value = scope:cbp_us17_native_baseline_import_efficiency }
		set_variable = { name = test_cbp_us17_probe_baseline_export value = scope:cbp_us17_native_baseline_export_efficiency }
		set_variable = { name = test_cbp_us17_probe_baseline_maintenance value = scope:cbp_us17_native_baseline_maintenance_efficiency }

		# Calculate and persist the shared coefficient before deriving US-17 values.
		cbp_compute_us20_route_loss_coefficient_from_selling_baseline = yes
		set_variable = { name = test_cbp_us17_probe_expected_route_loss_coefficient value = scope:cbp_us20_route_loss_coefficient_result }
		set_variable = { name = cbp_us20_route_loss_coefficient value = scope:cbp_us20_route_loss_coefficient_result }
		cbp_compute_us17_native_corrections_from_baselines = yes
		set_variable = {
			name = test_cbp_us17_probe_expected_selling
			value = { value = scope:cbp_us17_native_baseline_selling_efficiency add = scope:cbp_us17_native_selling_correction_result }
		}
		set_variable = {
			name = test_cbp_us17_probe_expected_import
			value = { value = scope:cbp_us17_native_baseline_import_efficiency add = scope:cbp_us17_native_import_correction_result }
		}
		set_variable = {
			name = test_cbp_us17_probe_expected_export
			value = { value = scope:cbp_us17_native_baseline_export_efficiency add = scope:cbp_us17_native_export_correction_result }
		}
		set_variable = { name = test_cbp_us17_probe_expected_maintenance value = scope:cbp_us17_native_baseline_maintenance_efficiency }

		cbp_refresh_us17_native_profit_modifiers_for_current_country = yes
		if = {
			limit = {
				OR = {
					NOT = { has_variable = cbp_us20_route_loss_coefficient }
					NOT = { has_variable = cbp_us17_native_selling_correction }
					NOT = { has_variable = cbp_us17_native_import_correction }
					NOT = { has_variable = cbp_us17_native_export_correction }
					NOT = { has_variable = cbp_us17_native_selling_baseline }
					NOT = { has_variable = cbp_us17_native_import_baseline }
					NOT = { has_variable = cbp_us17_native_export_baseline }
					NOT = { has_variable = cbp_us17_native_modifier_state_version }
					has_variable = cbp_us17_native_maintenance_correction
				}
			}
			set_global_variable = test_cbp_us17_owner_modifiers_failed
			error_log = "ModeU5 US17 CURVE ASSERT FAIL assertion=live_refresh_persisted_shared_coefficient_state"
			debug_log = "ModeU5 TEST FAIL scenario=us17_trade_owner_modifiers outcome=HARD_FAILURE phase=live_refresh reason=invalid_persisted_shared_coefficient_state"
		}
		else = {
			set_global_variable = test_cbp_us17_owner_modifiers_awaiting_live_check

			debug_log = "ModeU5 TEST PENDING scenario=us17_trade_owner_modifiers phase=live_auto_modifier_application wait_days=1"
		}
	}
}
