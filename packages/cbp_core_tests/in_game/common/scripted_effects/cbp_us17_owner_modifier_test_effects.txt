# Focused US-17 trade-owner modifier, maintenance-define, and maximum-only cap probe.
# Run through event cbp_us17_owner_modifiers.1.

cbp_us17_owner_modifiers_clear_test_results = {
	remove_global_variable = test_cbp_us17_owner_modifiers_started
	remove_global_variable = test_cbp_us17_owner_modifiers_passed
	remove_global_variable = test_cbp_us17_owner_modifiers_failed
	remove_global_variable = test_cbp_us17_owner_modifiers_blocked
}

cbp_us17_owner_modifiers_note_failure = {
	save_temporary_scope_value_as = {
		name = test_cbp_us17_owner_modifiers_failure_count
		value = {
			value = scope:test_cbp_us17_owner_modifiers_failure_count
			add = 1
		}
	}
}

cbp_us17_owner_modifiers_seed_common_formula_inputs = {
	save_temporary_scope_value_as = { name = cbp_trade_efficiency_formula_inputs_available value = 1 }
	save_temporary_scope_value_as = { name = cbp_trade_efficiency_country_modifier_inputs_available value = 1 }
	save_temporary_scope_value_as = { name = gui_cbp_trade_efficiency_route_quantity value = 100 }
	save_temporary_scope_value_as = { name = gui_cbp_trade_efficiency_sell_price value = 1.5 }
	save_temporary_scope_value_as = { name = gui_cbp_trade_efficiency_buy_price value = 1.0 }
	save_temporary_scope_value_as = { name = gui_cbp_trade_efficiency_export_cost_modifier value = 0.0 }
	save_temporary_scope_value_as = { name = gui_cbp_trade_efficiency_base_maintenance_amount value = 20 }
	save_temporary_scope_value_as = { name = gui_cbp_trade_efficiency_merchant_maintenance_efficiency value = 0 }
	save_temporary_scope_value_as = { name = gui_cbp_us20_trade_maintenance value = 0.70 }
	save_temporary_scope_value_as = { name = gui_cbp_us20_goods_amount_sent value = 100 }
	save_temporary_scope_value_as = { name = gui_cbp_us20_engine_goods_amount_received value = 100 }
}

cbp_debug_run_us17_owner_modifier_probe = {
	cbp_us17_owner_modifiers_clear_test_results = yes
	set_global_variable = test_cbp_us17_owner_modifiers_started
	debug_log = "ModeU5 TEST ENTERED scenario=us17_trade_owner_modifiers"
	debug_log = "ModeU5 US17 OWNER_MODIFIERS AUDIT START expected=owner_country_import_selling_merchant_maintenance_efficiency_plus_loaded_define_plus_maximum_only_cap"

	if = {
		limit = { NOT = { country_exists = c:FRA } }
		set_global_variable = test_cbp_us17_owner_modifiers_blocked
		debug_log = "ModeU5 TEST BLOCKED scenario=us17_trade_owner_modifiers reason=missing_FRA"
	}
	else = {
		save_temporary_scope_value_as = { name = test_cbp_us17_owner_modifiers_failure_count value = 0 }

		# Confirm that the capture helper reads the current trade-owner country and
		# derives base maintenance from the effective loaded define.
		save_temporary_scope_value_as = { name = cbp_trade_owner_trade_volume value = 40 }
		save_temporary_scope_value_as = { name = test_cbp_us17_expected_base_maintenance_unit_cost value = define:NCountry|MERCHANT_MAINTENANCE_COST }
		save_temporary_scope_value_as = {
			name = test_cbp_us17_expected_base_maintenance_amount
			value = {
				value = scope:cbp_trade_owner_trade_volume
				multiply = scope:test_cbp_us17_expected_base_maintenance_unit_cost
			}
		}
		c:FRA = {
			save_temporary_scope_as = cbp_trade_owner_country
			save_temporary_scope_value_as = { name = test_cbp_us17_expected_buying_efficiency value = modifier:import_efficiency }
			save_temporary_scope_value_as = { name = test_cbp_us17_expected_selling_efficiency value = modifier:selling_efficiency }
			save_temporary_scope_value_as = { name = test_cbp_us17_expected_merchant_maintenance_efficiency value = modifier:merchant_maintenance_efficiency }
			cbp_capture_trade_owner_country_modifier_inputs = yes
		}

		if = {
			limit = { NOT = { scope:cbp_trade_efficiency_country_modifier_inputs_available = 1 } }
			cbp_us17_owner_modifiers_note_failure = yes
			error_log = "ModeU5 US17 OWNER_MODIFIERS ASSERT FAIL assertion=country_modifier_inputs_available expected=1"
		}
		if = {
			limit = { NOT = { scope:gui_cbp_trade_efficiency_buying_efficiency = scope:test_cbp_us17_expected_buying_efficiency } }
			cbp_us17_owner_modifiers_note_failure = yes
			error_log = "ModeU5 US17 OWNER_MODIFIERS ASSERT FAIL assertion=import_efficiency_source expected=trade_owner_country_modifier"
		}
		if = {
			limit = { NOT = { scope:gui_cbp_trade_efficiency_selling_efficiency = scope:test_cbp_us17_expected_selling_efficiency } }
			cbp_us17_owner_modifiers_note_failure = yes
			error_log = "ModeU5 US17 OWNER_MODIFIERS ASSERT FAIL assertion=selling_efficiency_source expected=trade_owner_country_modifier"
		}
		if = {
			limit = { NOT = { scope:gui_cbp_trade_efficiency_merchant_maintenance_efficiency = scope:test_cbp_us17_expected_merchant_maintenance_efficiency } }
			cbp_us17_owner_modifiers_note_failure = yes
			error_log = "ModeU5 US17 OWNER_MODIFIERS ASSERT FAIL assertion=merchant_maintenance_efficiency_source expected=trade_owner_country_modifier"
		}
		if = {
			limit = { NOT = { scope:gui_cbp_trade_efficiency_base_maintenance_unit_cost = scope:test_cbp_us17_expected_base_maintenance_unit_cost } }
			cbp_us17_owner_modifiers_note_failure = yes
			error_log = "ModeU5 US17 OWNER_MODIFIERS ASSERT FAIL assertion=base_maintenance_define_source expected=define_NCountry_MERCHANT_MAINTENANCE_COST"
		}
		if = {
			limit = { NOT = { scope:gui_cbp_trade_efficiency_base_maintenance_amount = scope:test_cbp_us17_expected_base_maintenance_amount } }
			cbp_us17_owner_modifiers_note_failure = yes
			error_log = "ModeU5 US17 OWNER_MODIFIERS ASSERT FAIL assertion=base_maintenance_amount expected=trade_volume_times_loaded_define"
		}
		debug_log = "ModeU5 US17 OWNER_MODIFIERS AUDIT CHECK phase=country_modifier_capture expected=trade_owner_country inputs=import_selling_merchant_maintenance_efficiency base=trade_volume_times_loaded_define"

		# Lower-bound probe: -0.4 and -0.2 average to -0.3 and must remain negative.
		cbp_us17_owner_modifiers_seed_common_formula_inputs = yes
		save_temporary_scope_value_as = { name = gui_cbp_trade_efficiency_buying_efficiency value = -0.4 }
		save_temporary_scope_value_as = { name = gui_cbp_trade_efficiency_selling_efficiency value = -0.2 }
		cbp_compute_us17_us20_route_formula_from_owner_modifiers = yes
		if = {
			limit = {
				OR = {
					scope:gui_cbp_buying_selling_efficiency_clamped < -0.301
					scope:gui_cbp_buying_selling_efficiency_clamped > -0.299
				}
			}
			cbp_us17_owner_modifiers_note_failure = yes
			error_log = "ModeU5 US17 OWNER_MODIFIERS ASSERT FAIL assertion=negative_average_preserved expected=-0.30"
		}
		if = {
			limit = {
				OR = {
					scope:gui_cbp_trade_maintenance_efficiency_delta < -6.01
					scope:gui_cbp_trade_maintenance_efficiency_delta > -5.99
				}
			}
			cbp_us17_owner_modifiers_note_failure = yes
			error_log = "ModeU5 US17 OWNER_MODIFIERS ASSERT FAIL assertion=negative_efficiency_adds_maintenance expected=-6.00"
		}
		debug_log = "ModeU5 US17 OWNER_MODIFIERS AUDIT CHECK phase=maximum_only_lower_side expected=negative_average_preserved"

		# Upper-bound probe: 1.4 and 1.2 average to 1.3, capped at exactly 1.
		cbp_us17_owner_modifiers_seed_common_formula_inputs = yes
		save_temporary_scope_value_as = { name = gui_cbp_trade_efficiency_buying_efficiency value = 1.4 }
		save_temporary_scope_value_as = { name = gui_cbp_trade_efficiency_selling_efficiency value = 1.2 }
		cbp_compute_us17_us20_route_formula_from_owner_modifiers = yes
		if = {
			limit = { NOT = { scope:gui_cbp_buying_selling_efficiency_clamped = 1 } }
			cbp_us17_owner_modifiers_note_failure = yes
			error_log = "ModeU5 US17 OWNER_MODIFIERS ASSERT FAIL assertion=positive_average_capped expected=1"
		}
		debug_log = "ModeU5 US17 OWNER_MODIFIERS AUDIT CHECK phase=maximum_only_upper_side expected=cap_at_1"

		# Combined formula probe:
		# old price bonus = 35
		# capped average = 0.15
		# current maintenance factor = 1 - 0.20 = 0.80
		# adjusted base maintenance = 20 * 0.80 = 16
		# maintenance saving = 16 * 0.15 = 2.4
		# route money delta = -35 + 2.4 = -32.6
		cbp_us17_owner_modifiers_seed_common_formula_inputs = yes
		save_temporary_scope_value_as = { name = gui_cbp_trade_efficiency_buying_efficiency value = 0.20 }
		save_temporary_scope_value_as = { name = gui_cbp_trade_efficiency_selling_efficiency value = 0.10 }
		save_temporary_scope_value_as = { name = gui_cbp_trade_efficiency_merchant_maintenance_efficiency value = 0.20 }
		cbp_compute_us17_us20_route_formula_from_owner_modifiers = yes
		if = {
			limit = {
				OR = {
					scope:gui_cbp_trade_efficiency_merchant_maintenance_factor < 0.799
					scope:gui_cbp_trade_efficiency_merchant_maintenance_factor > 0.801
				}
			}
			cbp_us17_owner_modifiers_note_failure = yes
			error_log = "ModeU5 US17 OWNER_MODIFIERS ASSERT FAIL assertion=merchant_maintenance_factor expected=0.80"
		}
		if = {
			limit = {
				OR = {
					scope:gui_cbp_trade_maintenance_efficiency_delta < 2.39
					scope:gui_cbp_trade_maintenance_efficiency_delta > 2.41
				}
			}
			cbp_us17_owner_modifiers_note_failure = yes
			error_log = "ModeU5 US17 OWNER_MODIFIERS ASSERT FAIL assertion=maintenance_saving expected=2.40"
		}
		if = {
			limit = {
				OR = {
					scope:gui_cbp_trade_efficiency_route_money_delta < -32.61
					scope:gui_cbp_trade_efficiency_route_money_delta > -32.59
				}
			}
			cbp_us17_owner_modifiers_note_failure = yes
			error_log = "ModeU5 US17 OWNER_MODIFIERS ASSERT FAIL assertion=route_money_delta expected=-32.60"
		}
		debug_log = "ModeU5 US17 OWNER_MODIFIERS AUDIT CHECK phase=combined_owner_modifier_formula expected=old_bonus_35_current_maintenance_16_saving_2.4_delta_minus_32.6"

		if = {
			limit = { scope:test_cbp_us17_owner_modifiers_failure_count = 0 }
			set_global_variable = test_cbp_us17_owner_modifiers_passed
			debug_log = "ModeU5 TEST PASS scenario=us17_trade_owner_modifiers hard_failures=0 owner_inputs=import_selling_merchant_maintenance_efficiency base_cost=define_NCountry_MERCHANT_MAINTENANCE_COST clamp=maximum_only"
		}
		else = {
			set_global_variable = test_cbp_us17_owner_modifiers_failed
			debug_log = "ModeU5 TEST FAIL scenario=us17_trade_owner_modifiers outcome=HARD_FAILURE see_ASSERT_FAIL_entries"
		}
	}
}
