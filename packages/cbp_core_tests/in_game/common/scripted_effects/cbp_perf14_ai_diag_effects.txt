cbp_perf14_diag_run_ai_human_relevant_market_promotion = {
	cbp_enter_test_audit_runtime_mode = yes
	cbp_perf14_set_cmm_main_mode_for_test = { value = 1 }
	cbp_perf14_diag_reset_ui = yes

	save_temporary_scope_value_as = { name = test_cbp_perf14_ai_fixture_available value = 0 }

	if = {
		limit = { country_exists = c:ENG c:ENG = { is_ai = yes has_markets = yes } }
		save_temporary_scope_value_as = { name = test_cbp_perf14_ai_fixture_available value = 1 }
		set_variable = { name = test_cbp_perf14_ai_diag_fixture value = 1 }
		c:ENG = { save_temporary_scope_as = cbp_perf14_ai_country }
		c:ENG.capital.market = { save_temporary_scope_as = cbp_perf14_ai_market }
		debug_log = "ModeU5 PERF-14 AI_DIAG fixture ENG"
	}
	else_if = {
		limit = { country_exists = c:FRA c:FRA = { is_ai = yes has_markets = yes } }
		save_temporary_scope_value_as = { name = test_cbp_perf14_ai_fixture_available value = 1 }
		set_variable = { name = test_cbp_perf14_ai_diag_fixture value = 2 }
		c:FRA = { save_temporary_scope_as = cbp_perf14_ai_country }
		c:FRA.capital.market = { save_temporary_scope_as = cbp_perf14_ai_market }
		debug_log = "ModeU5 PERF-14 AI_DIAG fixture FRA"
	}
	else_if = {
		limit = { country_exists = c:POR c:POR = { is_ai = yes has_markets = yes } }
		save_temporary_scope_value_as = { name = test_cbp_perf14_ai_fixture_available value = 1 }
		set_variable = { name = test_cbp_perf14_ai_diag_fixture value = 3 }
		c:POR = { save_temporary_scope_as = cbp_perf14_ai_country }
		c:POR.capital.market = { save_temporary_scope_as = cbp_perf14_ai_market }
		debug_log = "ModeU5 PERF-14 AI_DIAG fixture POR"
	}

	if = {
		limit = { scope:test_cbp_perf14_ai_fixture_available > 0 }
		if = {
			limit = {
				has_global_variable_list = cbp_performance_relevant_markets
				is_target_in_global_variable_list = { name = cbp_performance_relevant_markets target = scope:cbp_perf14_ai_market }
			}
			set_variable = { name = test_cbp_perf14_ai_diag_before_relevant_marker value = 1 }
			debug_log = "ModeU5 PERF-14 AI_AUDIT before_relevant_marker YES"
		}
		else = { debug_log = "ModeU5 PERF-14 AI_AUDIT before_relevant_marker NO" }

		scope:cbp_perf14_ai_market = { save_temporary_scope_as = cbp_market }
		if = {
			limit = { cbp_market_detailed_accounting_promoted_trigger = yes }
			set_variable = { name = test_cbp_perf14_ai_diag_before_promoted_marker value = 1 }
			debug_log = "ModeU5 PERF-14 AI_AUDIT before_promoted_marker YES"
		}
		else = { debug_log = "ModeU5 PERF-14 AI_AUDIT before_promoted_marker NO" }

		cbp_clear_performance_relevant_markets = yes
		cbp_clear_detailed_accounting_promoted_markets = yes

		scope:cbp_perf14_ai_market = { save_temporary_scope_as = cbp_performance_relevant_market }
		cbp_mark_performance_relevant_market = yes

		if = {
			limit = {
				has_global_variable_list = cbp_performance_relevant_markets
				is_target_in_global_variable_list = { name = cbp_performance_relevant_markets target = scope:cbp_perf14_ai_market }
			}
			set_variable = { name = test_cbp_perf14_ai_diag_relevant_list value = 1 }
			debug_log = "ModeU5 PERF-14 AI_DIAG relevant_list YES"
		}
		else = { debug_log = "ModeU5 PERF-14 AI_DIAG relevant_list NO" }

		cbp_prepare_stock_mutation_accounting_mode = { country = scope:cbp_perf14_ai_country market = scope:cbp_perf14_ai_market }

		set_variable = { name = test_cbp_perf14_ai_diag_audit_goods_checked value = global_var:cbp_perf14_promotion_market_goods_checked }
		set_variable = { name = test_cbp_perf14_ai_diag_audit_country_sum_before value = global_var:cbp_perf14_promotion_country_sum_total_before }
		set_variable = { name = test_cbp_perf14_ai_diag_audit_market_aggregate_before value = global_var:cbp_perf14_promotion_aggregate_total_before }
		set_variable = { name = test_cbp_perf14_ai_diag_audit_missing_materialization_delta value = global_var:cbp_perf14_promotion_delta_total }
		set_variable = { name = test_cbp_perf14_ai_diag_audit_overmaterialized_delta value = { value = global_var:cbp_perf14_promotion_country_sum_total_before subtract = global_var:cbp_perf14_promotion_aggregate_total_before min = 0 } }
		if = { limit = { global_var:cbp_perf14_promotion_country_sum_total_before > 0 } set_variable = { name = test_cbp_perf14_ai_diag_audit_existing_country_stock value = 1 } debug_log = "ModeU5 PERF-14 AI_AUDIT existing_country_stock YES" }
		else = { debug_log = "ModeU5 PERF-14 AI_AUDIT existing_country_stock NO" }
		if = { limit = { global_var:cbp_perf14_promotion_country_sum_total_before > global_var:cbp_perf14_promotion_aggregate_total_before } debug_log = "ModeU5 PERF-14 AI_AUDIT country_sum_gt_market_aggregate YES" }
		else = { debug_log = "ModeU5 PERF-14 AI_AUDIT country_sum_gt_market_aggregate NO" }

		if = { limit = { cbp_stock_mutation_use_detailed_accounting_trigger = yes } set_variable = { name = test_cbp_perf14_ai_diag_detailed value = 1 } debug_log = "ModeU5 PERF-14 AI_DIAG detailed YES" }
		else = { debug_log = "ModeU5 PERF-14 AI_DIAG detailed NO" }
		if = { limit = { cbp_stock_mutation_use_market_level_fallback_trigger = yes } set_variable = { name = test_cbp_perf14_ai_diag_fallback value = 1 } debug_log = "ModeU5 PERF-14 AI_DIAG fallback YES" }
		else = { debug_log = "ModeU5 PERF-14 AI_DIAG fallback NO" }
		if = { limit = { cbp_stock_mutation_blocked_trigger = yes } set_variable = { name = test_cbp_perf14_ai_diag_blocked value = 1 } debug_log = "ModeU5 PERF-14 AI_DIAG blocked YES" }
		else = { debug_log = "ModeU5 PERF-14 AI_DIAG blocked NO" }
		if = { limit = { scope:cbp_stock_mutation_promotion_attempted > 0 } set_variable = { name = test_cbp_perf14_ai_diag_attempted value = 1 } debug_log = "ModeU5 PERF-14 AI_DIAG attempted YES" }
		else = { debug_log = "ModeU5 PERF-14 AI_DIAG attempted NO" }

		else = { debug_log = "ModeU5 PERF-14 AI_DIAG succeeded NO" }

		scope:cbp_perf14_ai_market = { save_temporary_scope_as = cbp_market }
		if = { limit = { cbp_market_detailed_accounting_promoted_trigger = yes } set_variable = { name = test_cbp_perf14_ai_diag_promoted value = 1 } debug_log = "ModeU5 PERF-14 AI_DIAG promoted YES" }
		else = { debug_log = "ModeU5 PERF-14 AI_DIAG promoted NO" }

		if = { limit = { has_global_variable = cbp_perf14_promotion_blocked_not_human_relevant global_var:cbp_perf14_promotion_blocked_not_human_relevant > 0 } set_variable = { name = test_cbp_perf14_ai_diag_reason_not_relevant value = 1 } debug_log = "ModeU5 PERF-14 AI_DIAG reason_not_relevant YES" }
		if = { limit = { has_global_variable = cbp_perf14_promotion_zero_capacity_failures global_var:cbp_perf14_promotion_zero_capacity_failures > 0 } set_variable = { name = test_cbp_perf14_ai_diag_reason_zero_capacity value = 1 } debug_log = "ModeU5 PERF-14 AI_DIAG reason_zero_capacity YES" }
		if = { limit = { has_global_variable = cbp_perf14_promotion_negative_capacity_failures global_var:cbp_perf14_promotion_negative_capacity_failures > 0 } set_variable = { name = test_cbp_perf14_ai_diag_reason_negative_capacity value = 1 } debug_log = "ModeU5 PERF-14 AI_DIAG reason_negative_capacity YES" }

		if = { limit = { has_global_variable = cbp_perf14_promotion_validation_failures global_var:cbp_perf14_promotion_validation_failures > 0 } set_variable = { name = test_cbp_perf14_ai_diag_reason_validation value = 1 } debug_log = "ModeU5 PERF-14 AI_DIAG reason_validation YES" }

	else = {
		debug_log = "ModeU5 PERF-14 AI_DIAG fixture NONE"
