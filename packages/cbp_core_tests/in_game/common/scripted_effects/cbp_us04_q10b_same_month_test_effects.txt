cbp_us04_q10b_begin = {
	debug_log = "ModeU5 TEST ENTERED scenario=us04_q10b_replace_pop_demand_same_month"
	if = {
		limit = { has_global_variable = cbp_us04_q10b_same_month_in_progress }
		set_global_variable = test_cbp_us04_q10b_same_month_blocked
		debug_log = "ModeU5 TEST BLOCKED scenario=us04_q10b_replace_pop_demand_same_month reason=probe_already_in_progress"
	}
	else = {
		cbp_us04_q10b_clear_state = yes
		set_global_variable = cbp_us04_q10b_same_month_in_progress
		capital.market = { save_scope_as = cbp_us04_q10b_market }
		set_global_variable = { name = cbp_us04_q9_replace_global_books value = 1.0 }
		debug_log = "ModeU5 US-04 VANILLA DEMAND DUMP q10b_same_month phase=begin source=1.0000 wait=one_day"
	}
}

cbp_us04_q10b_finalize = {
	if = {
		limit = { NOT = { has_global_variable = cbp_us04_q10b_same_month_in_progress } }
		set_global_variable = test_cbp_us04_q10b_same_month_blocked
		debug_log = "ModeU5 TEST BLOCKED scenario=us04_q10b_replace_pop_demand_same_month reason=finalize_without_in_progress"
	}
	else = {
		scope:cbp_us04_q10b_market = {
			save_temporary_scope_value_as = { name = cbp_us04_q10b_books_current value = "goods_demand_in_market(goods:books)" }
			save_temporary_scope_value_as = { name = cbp_us04_q10b_control_current value = "goods_demand_in_market(goods:wool)" }
		}
		set_global_variable = { name = cbp_us04_q10b_books_400 value = scope:cbp_us04_q10b_books_current }
		set_global_variable = { name = cbp_us04_q10b_control_400 value = scope:cbp_us04_q10b_control_current }
		set_variable = { name = cbp_us04_q10b_ui_books value = global_var:cbp_us04_q10b_books_400 }
		set_variable = { name = cbp_us04_q10b_ui_control value = global_var:cbp_us04_q10b_control_400 }
		debug_log = "ModeU5 US-04 VANILLA DEMAND DUMP q10b_same_month point=400 source=4.0000 books=[THIS.GetVariable('cbp_us04_q10b_ui_books').GetValue|4] control_wool=[THIS.GetVariable('cbp_us04_q10b_ui_control').GetValue|4]"

		save_temporary_scope_value_as = { name = cbp_us04_q10b_delta_099_vs_100 value = { value = global_var:cbp_us04_q10b_books_099 subtract = global_var:cbp_us04_q10b_books_100 } }
		save_temporary_scope_value_as = { name = cbp_us04_q10b_delta_101_vs_100 value = { value = global_var:cbp_us04_q10b_books_101 subtract = global_var:cbp_us04_q10b_books_100 } }
		save_temporary_scope_value_as = { name = cbp_us04_q10b_delta_120_vs_100 value = { value = global_var:cbp_us04_q10b_books_120 subtract = global_var:cbp_us04_q10b_books_100 } }
		save_temporary_scope_value_as = { name = cbp_us04_q10b_delta_400_vs_100 value = { value = global_var:cbp_us04_q10b_books_400 subtract = global_var:cbp_us04_q10b_books_100 } }
		set_global_variable = { name = cbp_us04_q10b_delta_099_vs_100 value = scope:cbp_us04_q10b_delta_099_vs_100 }
		set_global_variable = { name = cbp_us04_q10b_delta_101_vs_100 value = scope:cbp_us04_q10b_delta_101_vs_100 }
		set_global_variable = { name = cbp_us04_q10b_delta_120_vs_100 value = scope:cbp_us04_q10b_delta_120_vs_100 }
		set_global_variable = { name = cbp_us04_q10b_delta_400_vs_100 value = scope:cbp_us04_q10b_delta_400_vs_100 }

		set_variable = { name = cbp_us04_q10b_ui_delta_099 value = global_var:cbp_us04_q10b_delta_099_vs_100 }
		set_variable = { name = cbp_us04_q10b_ui_delta_101 value = global_var:cbp_us04_q10b_delta_101_vs_100 }
		set_variable = { name = cbp_us04_q10b_ui_delta_120 value = global_var:cbp_us04_q10b_delta_120_vs_100 }
		set_variable = { name = cbp_us04_q10b_ui_delta_400 value = global_var:cbp_us04_q10b_delta_400_vs_100 }

		debug_log = "ModeU5 TEST PASS scenario=us04_q10b_replace_pop_demand_same_month"
		remove_global_variable = cbp_us04_q10b_same_month_in_progress
	}
