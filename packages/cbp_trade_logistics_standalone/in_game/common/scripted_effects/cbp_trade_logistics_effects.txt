# Standalone US-17 / US-20 route reconciliation.
#
# Runtime ownership:
# monthly_country_pulse(country) -> every_trade -> trade.owner
#
# Accounting boundary:
# - US-17 mutates only trade-owner gold;
# - US-20 mutates only Vanilla destination-market goods supply;
# - no country x market accounting exists in this package.

cbp_trade_logistics_run_country_routes = {
	save_temporary_scope_as = cbp_trade_logistics_controller
	every_trade = {
		save_temporary_scope_as = cbp_trade_logistics_trade
		owner = { save_temporary_scope_as = cbp_trade_logistics_owner }
		to_market = { save_temporary_scope_as = cbp_trade_logistics_target_market }
		traded_goods = { save_temporary_scope_as = cbp_trade_logistics_good }

		if = {
			limit = {
				exists = scope:cbp_trade_logistics_owner
				exists = scope:cbp_trade_logistics_target_market
				exists = scope:cbp_trade_logistics_good
				scope:cbp_trade_logistics_owner = scope:cbp_trade_logistics_controller
			}
			cbp_trade_logistics_capture_route_inputs = yes
			cbp_trade_logistics_apply_us17 = yes
			cbp_trade_logistics_apply_us20 = yes
		}
	}
}

cbp_trade_logistics_capture_route_inputs = {
	save_temporary_scope_value_as = {
		name = cbp_trade_logistics_trade_volume
		value = { value = trade_volume min = 0 }
	}
	save_temporary_scope_value_as = {
		name = cbp_trade_logistics_goods_quantity
		value = {
			value = trade_volume
			divide = traded_goods:transport_cost
			min = 0
		}
	}
	save_temporary_scope_value_as = {
		name = cbp_trade_logistics_trade_maintenance
		value = { value = trade_maintenance min = 0 max = 1 }
	}
	save_temporary_scope_value_as = {
		name = cbp_trade_logistics_sell_price
		value = { value = sell_price min = 0 }
	}
	save_temporary_scope_value_as = {
		name = cbp_trade_logistics_buy_price
		value = { value = buy_price min = 0 }
	}
	save_temporary_scope_value_as = {
		name = cbp_trade_logistics_export_cost_modifier
		value = export_cost_modifier
	}

	scope:cbp_trade_logistics_owner = {
		save_temporary_scope_value_as = {
			name = cbp_trade_logistics_import_efficiency
			value = modifier:import_efficiency
		}
		save_temporary_scope_value_as = {
			name = cbp_trade_logistics_selling_efficiency
			value = modifier:selling_efficiency
		}
		save_temporary_scope_value_as = {
			name = cbp_trade_logistics_maintenance_efficiency
			value = modifier:merchant_maintenance_efficiency
		}
		save_temporary_scope_value_as = {
			name = cbp_trade_logistics_crown_share
			value = "estate_power(estate_type:crown_estate)"
		}
		save_temporary_scope_value_as = {
			name = cbp_trade_logistics_trade_income_modifier
			value = modifier:trade_income
		}
	}
}

cbp_trade_logistics_apply_us17 = {
	save_temporary_scope_value_as = {
		name = cbp_trade_logistics_efficiency_average
		value = {
			value = scope:cbp_trade_logistics_import_efficiency
			add = scope:cbp_trade_logistics_selling_efficiency
			divide = 2
			max = 1
		}
	}
	save_temporary_scope_value_as = {
		name = cbp_trade_logistics_old_price_bonus
		value = {
			value = scope:cbp_trade_logistics_goods_quantity
			multiply = scope:cbp_trade_logistics_sell_price
			multiply = scope:cbp_trade_logistics_selling_efficiency
			add = {
				value = scope:cbp_trade_logistics_goods_quantity
				multiply = scope:cbp_trade_logistics_buy_price
				multiply = scope:cbp_trade_logistics_import_efficiency
				multiply = {
					value = scope:cbp_trade_logistics_export_cost_modifier
					add = 1
				}
			}
		}
	}
	save_temporary_scope_value_as = {
		name = cbp_trade_logistics_maintenance_saving
		value = {
			value = scope:cbp_trade_logistics_trade_volume
			multiply = cbp_trade_logistics_base_merchant_maintenance_cost
			multiply = {
				value = 1
				add = {
					value = scope:cbp_trade_logistics_maintenance_efficiency
					multiply = -1
				}
				min = 0
			}
			multiply = scope:cbp_trade_logistics_efficiency_average
		}
	}
	save_temporary_scope_value_as = {
		name = cbp_trade_logistics_money_delta
		value = {
			value = scope:cbp_trade_logistics_old_price_bonus
			multiply = -1
			add = scope:cbp_trade_logistics_maintenance_saving
			multiply = scope:cbp_trade_logistics_crown_share
			multiply = scope:cbp_trade_logistics_trade_income_modifier
		}
	}

	if = {
		limit = { NOT = { scope:cbp_trade_logistics_money_delta = 0 } }
		scope:cbp_trade_logistics_owner = {
			add_gold = scope:cbp_trade_logistics_money_delta
		}
	}
}

cbp_trade_logistics_apply_us20 = {
	save_temporary_scope_value_as = {
		name = cbp_trade_logistics_delivery_loss
		value = {
			value = 1
			add = {
				value = scope:cbp_trade_logistics_trade_maintenance
				multiply = -1
			}
			min = 0
			max = 1
			multiply = 0.1
			max = scope:cbp_trade_logistics_goods_quantity
		}
	}
	if = {
		limit = { scope:cbp_trade_logistics_delivery_loss > 0 }
		save_temporary_scope_value_as = {
			name = cbp_trade_logistics_market_delta
			value = {
				value = scope:cbp_trade_logistics_delivery_loss
				multiply = -1
			}
		}
		scope:cbp_trade_logistics_target_market = {
			add_goods_supply = {
				goods = scope:cbp_trade_logistics_good
				amount = scope:cbp_trade_logistics_market_delta
			}
		}
	}
}
