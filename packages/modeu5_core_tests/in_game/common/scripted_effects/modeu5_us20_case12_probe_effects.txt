modeu5_us20_case12_probe_run_one = {
	modeu5_us17_us20_seed_reconciliation_fixture = yes
	save_temporary_scope_value_as = { name = modeu5_us20_test_origin_market_promoted value = $origin_promoted$ }
	save_temporary_scope_value_as = { name = modeu5_us20_test_target_market_promoted value = $target_promoted$ }
	save_temporary_scope_value_as = { name = modeu5_us20_test_trade_owner_present_in_target_market value = 0 }
	modeu5_compute_us17_us20_route_formula_from_current_values = yes
	if = {
		limit = { scope:modeu5_us20_goods_reconciliation_delta < 0 }
		save_temporary_scope_value_as = {
			name = modeu5_us20_goods_loss_quantity
			value = { value = scope:modeu5_us20_goods_reconciliation_delta multiply = -1 }
		}
		modeu5_classify_us20_route_market_accounting_surfaces = yes
		if = {
			limit = { scope:modeu5_us20_target_market_is_promoted = 0 }
			modeu5_apply_us20_destination_market_loss_reconciliation = yes
		}
		else = {
			modeu5_apply_us20_promoted_destination_loss_reconciliation = yes
		}
	}
	else = {
		modeu5_us20_case12_probe_note_failure = yes
		error_log = "ModeU5 US20 CASE12 ASSERT FAIL assertion=expected_negative_goods_delta category=hard_failure"
	}
}
