		}

modeu5_us20_case12_probe_run_one = {
	modeu5_us17_us20_seed_reconciliation_fixture = yes
	save_temporary_scope_value_as = { name = modeu5_us20_test_origin_market_promoted value = $origin_promoted$ }
	save_temporary_scope_value_as = { name = modeu5_us20_test_target_market_promoted value = 0 }
	modeu5_compute_us17_us20_route_formula_from_current_values = yes
	if = {
		limit = { scope:modeu5_us20_goods_reconciliation_delta < 0 }
		save_temporary_scope_value_as = {
			name = modeu5_us20_goods_loss_quantity
			value = { value = scope:modeu5_us20_goods_reconciliation_delta multiply = -1 }
		}
		modeu5_classify_us20_route_market_accounting_surfaces = yes
		modeu5_apply_us20_destination_market_loss_reconciliation = yes
	}
	else = {
		modeu5_us20_case12_probe_note_failure = yes
		error_log = "ModeU5 US20 CASE12 ASSERT FAIL assertion=expected_negative_goods_delta category=hard_failure"
	}
}

modeu5_debug_run_us20_case12_market_loss_probe = {
	modeu5_us20_case12_probe_clear = yes
	set_global_variable = modeu5_test_us20_case12_probe_started
	save_temporary_scope_value_as = { name = modeu5_us20_case12_probe_failure_count value = 0 }
	debug_log = "ModeU5 TEST ENTERED scenario=us20_case12_market_loss_probe"
	debug_log = "ModeU5 US20 CASE12 AUDIT START expected=case1_and_case2_destination_non_promoted_market_loss_only"

	if = {
		limit = { NOT = { country_exists = c:FRA } }
		modeu5_us20_case12_probe_note_failure = yes
		error_log = "ModeU5 US20 CASE12 ASSERT FAIL assertion=missing_FRA category=hard_failure"
	}
	else = {
		modeu5_reset_trade_efficiency_reconciliation_accumulator = yes

		debug_log = "ModeU5 US20 CASE12 AUDIT CHECK phase=case1 origin_promoted=0 destination_promoted=0 expected=market_loss_only"
		modeu5_us20_case12_probe_add_supply_fixture = yes
		modeu5_us20_case12_probe_run_one = { origin_promoted = 0 }
		modeu5_us20_case12_probe_cleanup_supply_fixture = yes

		debug_log = "ModeU5 US20 CASE12 AUDIT CHECK phase=case2 origin_promoted=1 destination_promoted=0 expected=market_loss_only"
		modeu5_us20_case12_probe_add_supply_fixture = yes
		modeu5_us20_case12_probe_run_one = { origin_promoted = 1 }
		modeu5_us20_case12_probe_cleanup_supply_fixture = yes

		if = { limit = { OR = { NOT = { has_global_variable = modeu5_us20_case_origin_unpromoted_destination_unpromoted_routes } NOT = { global_var:modeu5_us20_case_origin_unpromoted_destination_unpromoted_routes = 1 } } } modeu5_us20_case12_probe_note_failure = yes error_log = "ModeU5 US20 CASE12 ASSERT FAIL assertion=case1_classification_expected=1 category=hard_failure" }
		if = { limit = { OR = { NOT = { has_global_variable = modeu5_us20_case_origin_promoted_destination_unpromoted_routes } NOT = { global_var:modeu5_us20_case_origin_promoted_destination_unpromoted_routes = 1 } } } modeu5_us20_case12_probe_note_failure = yes error_log = "ModeU5 US20 CASE12 ASSERT FAIL assertion=case2_classification_expected=1 category=hard_failure" }
		if = { limit = { OR = { NOT = { has_global_variable = modeu5_us20_market_goods_supply_loss_routes } NOT = { global_var:modeu5_us20_market_goods_supply_loss_routes = 2 } } } modeu5_us20_case12_probe_note_failure = yes error_log = "ModeU5 US20 CASE12 ASSERT FAIL assertion=market_goods_supply_loss_routes_expected=2 category=hard_failure" }
		if = { limit = { has_global_variable = modeu5_us20_goods_loss_routes global_var:modeu5_us20_goods_loss_routes > 0 } modeu5_us20_case12_probe_note_failure = yes error_log = "ModeU5 US20 CASE12 ASSERT FAIL assertion=destination_non_promoted_should_not_emit_country_loss_route category=hard_failure" }
		if = { limit = { has_global_variable = modeu5_us20_market_level_remove_good_blocked_routes global_var:modeu5_us20_market_level_remove_good_blocked_routes > 0 } modeu5_us20_case12_probe_note_failure = yes error_log = "ModeU5 US20 CASE12 ASSERT FAIL assertion=market_goods_supply_should_not_block category=hard_failure" }
		if = { limit = { has_global_variable = modeu5_trade_efficiency_goods_delta_blocked_routes global_var:modeu5_trade_efficiency_goods_delta_blocked_routes > 0 } modeu5_us20_case12_probe_note_failure = yes error_log = "ModeU5 US20 CASE12 ASSERT FAIL assertion=goods_delta_should_not_block category=hard_failure" }
	}

	if = {
		limit = { scope:modeu5_us20_case12_probe_failure_count = 0 }
		set_global_variable = modeu5_test_us20_case12_probe_passed
		debug_log = "ModeU5 TEST PASS scenario=us20_case12_market_loss_probe hard_failures=0"
	}
	else = {
		set_global_variable = modeu5_test_us20_case12_probe_failed
		debug_log = "ModeU5 TEST FAIL scenario=us20_case12_market_loss_probe outcome=HARD_FAILURE see_ASSERT_FAIL_entries"
