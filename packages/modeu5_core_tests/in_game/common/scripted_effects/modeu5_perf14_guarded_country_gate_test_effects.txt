# PERF-14 guarded runner adjustment for country-gated performance mode.
#
# Performance-mode detailed stock mutation is country gated: the current policy
# requires both a human country and a human-relevant market. An AI country remains
# on the market-level fallback path even if its market is present in the
# performance-relevant market list.

modeu5_perf14_guarded_probe_ai_country_gate = {
	if = {
		limit = { country_exists = c:ENG c:ENG = { is_ai = yes has_markets = yes } }
		save_temporary_scope_value_as = { name = modeu5_perf14_ai_fixture_available value = 1 }
		c:ENG = { save_temporary_scope_as = modeu5_perf14_ai_country }
		c:ENG.capital.market = { save_temporary_scope_as = modeu5_perf14_ai_market }
	}
	else_if = {
		limit = { country_exists = c:FRA c:FRA = { is_ai = yes has_markets = yes } }
		save_temporary_scope_value_as = { name = modeu5_perf14_ai_fixture_available value = 1 }
		c:FRA = { save_temporary_scope_as = modeu5_perf14_ai_country }
		c:FRA.capital.market = { save_temporary_scope_as = modeu5_perf14_ai_market }
	}
	else_if = {
		limit = { country_exists = c:POR c:POR = { is_ai = yes has_markets = yes } }
		save_temporary_scope_value_as = { name = modeu5_perf14_ai_fixture_available value = 1 }
		c:POR = { save_temporary_scope_as = modeu5_perf14_ai_country }
		c:POR.capital.market = { save_temporary_scope_as = modeu5_perf14_ai_market }
	}
	if = {
		limit = { scope:modeu5_perf14_ai_fixture_available > 0 }
		modeu5_clear_performance_relevant_markets = yes
		modeu5_prepare_country_market_accounting_decision = { country = scope:modeu5_perf14_ai_country market = scope:modeu5_perf14_ai_market }
		if = {
			limit = {
				OR = {
					modeu5_detailed_country_market_accounting_enabled_trigger = yes
					NOT = { modeu5_market_level_fallback_required_trigger = yes }
					scope:modeu5_accounting_decision_country_is_human > 0
				}
			}
			debug_log = "ModeU5 PERF-14 FAIL_REASON reason=ai_decision_gate_unexpected_state"
			set_global_variable = modeu5_test_perf14_performance_mode_cmm_failed
		}
		modeu5_prepare_stock_mutation_accounting_mode = { country = scope:modeu5_perf14_ai_country market = scope:modeu5_perf14_ai_market }
		if = {
			limit = {
				OR = {
					modeu5_stock_mutation_use_detailed_accounting_trigger = yes
					NOT = { modeu5_stock_mutation_use_market_level_fallback_trigger = yes }
					modeu5_stock_mutation_blocked_trigger = yes
					scope:modeu5_stock_mutation_promotion_attempted > 0
				}
			}
			debug_log = "ModeU5 PERF-14 FAIL_REASON reason=ai_stock_mutation_fallback_unexpected_state"
			set_global_variable = modeu5_test_perf14_performance_mode_cmm_failed
		}
		modeu5_clear_detailed_accounting_promoted_markets = yes
		scope:modeu5_perf14_ai_market = { save_temporary_scope_as = modeu5_performance_relevant_market }
		modeu5_mark_performance_relevant_market = yes
		modeu5_prepare_stock_mutation_accounting_mode = { country = scope:modeu5_perf14_ai_country market = scope:modeu5_perf14_ai_market }
		if = {
			limit = {
				OR = {
					modeu5_stock_mutation_use_detailed_accounting_trigger = yes
					NOT = { modeu5_stock_mutation_use_market_level_fallback_trigger = yes }
					modeu5_stock_mutation_blocked_trigger = yes
					scope:modeu5_stock_mutation_promotion_attempted > 0
					modeu5_market_detailed_accounting_promoted_trigger = yes
				}
			}
			debug_log = "ModeU5 PERF-14 FAIL_REASON reason=ai_country_gate_human_relevant_market_fallback_unexpected_state"
			set_global_variable = modeu5_test_perf14_performance_mode_cmm_failed
		}
	}
	else = {
		set_global_variable = modeu5_test_perf14_performance_mode_cmm_blocked
		error_log = "ModeU5 PERF-14 performance mode CMM probe blocked because no AI country market fixture was available."
	}
}

modeu5_debug_run_perf14_performance_mode_cmm_test_guarded_country_gate = {
	modeu5_perf14_clear_test_results = yes
	modeu5_enter_test_audit_runtime_mode = yes
	set_global_variable = modeu5_test_perf14_performance_mode_cmm_started
	debug_log = "ModeU5 TEST ENTERED scenario=perf14_performance_mode_cmm"

	save_temporary_scope_as = modeu5_perf14_test_country
	save_temporary_scope_value_as = { name = modeu5_perf14_original_cmm_main_present value = 0 }
	save_temporary_scope_value_as = { name = modeu5_perf14_original_cmm_main_value value = 1 }
	save_temporary_scope_value_as = { name = modeu5_perf14_ai_fixture_available value = 0 }
	save_temporary_scope_value_as = { name = modeu5_perf14_nonpresent_market_fixture_available value = 0 }

	if = {
		limit = { has_variable_map = cmm is_key_in_variable_map = { name = cmm target = flag:no_void_economy__nve_no_void_economy_main } }
		save_temporary_scope_value_as = { name = modeu5_perf14_original_cmm_main_present value = 1 }
		save_temporary_scope_value_as = { name = modeu5_perf14_original_cmm_main_value value = "variable_map(cmm|flag:no_void_economy__nve_no_void_economy_main)" }
	}

	if = {
		limit = { has_markets = no }
		set_global_variable = modeu5_test_perf14_performance_mode_cmm_blocked
		error_log = "ModeU5 PERF-14 performance mode CMM probe blocked because the current country has no markets."
	}
	else = {
		capital.market = { save_temporary_scope_as = modeu5_perf14_test_market }
		modeu5_perf14_guarded_check_performance_mode_gate = yes
		modeu5_perf14_guarded_probe_stock_mutation_gate_promotes = yes
		modeu5_perf14_guarded_probe_market_runtime_gate = yes
		modeu5_perf14_guarded_probe_positive_promotion = yes
		modeu5_perf14_guarded_probe_idempotent_promotion = yes
		modeu5_perf14_guarded_probe_sparse_supplier_scan = yes
		modeu5_perf14_guarded_probe_partial_promotion = yes
		modeu5_perf14_guarded_probe_ai_country_gate = yes
		modeu5_perf14_guarded_probe_nonpresent_block = yes
		modeu5_perf14_guarded_probe_normal_mode = yes
		modeu5_perf14_guarded_probe_deactivated_mode = yes
		if = {
			limit = { NOT = { has_global_variable = modeu5_test_perf14_performance_mode_cmm_failed } NOT = { has_global_variable = modeu5_test_perf14_performance_mode_cmm_blocked } }
			set_global_variable = modeu5_test_perf14_performance_mode_cmm_passed
		}
	}

	modeu5_perf14_restore_cmm_main_mode_after_test = yes
	modeu5_perf14_log_probe_result = yes
}
