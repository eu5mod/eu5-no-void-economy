cbp_q8_7_probe_current_owner_market_workshape = {
	# Current live owner shape: market-center ownership workaround, filtered to the
	# Performance Mode relevant-market set. This mirrors the market-owner surface of
	# cbp_run_monthly_promoted_market_local_cycle without executing its mutating
	# local branch.
	every_country = {
		every_market_center_in_country = {
			save_temporary_scope_as = cbp_q8_7_shadow_comparison_market
			if = {
				limit = {
					has_global_variable_list = cbp_performance_relevant_markets
					is_target_in_global_variable_list = {
						name = cbp_performance_relevant_markets
						target = scope:cbp_q8_7_shadow_comparison_market
					}
				}
				cbp_q8_7_note_market_center_shadow_comparison_market = yes
				save_temporary_scope_as = cbp_market_country_cache_market
				remove_global_variable = cbp_market_country_cache_last_rebuild_failed
				cbp_rebuild_countries_present_in_market = yes
				cbp_q8_7_increment_current_owner_cache_rebuild_count = yes
				if = {
					limit = { has_global_variable = cbp_market_country_cache_last_rebuild_failed }
					cbp_q8_7_increment_current_owner_cache_failure_count = yes
				}
				if = {
					limit = { has_global_variable_list = cbp_countries_present_in_market }
					every_in_global_list = {
						variable = cbp_countries_present_in_market
						save_temporary_scope_as = cbp_q8_7_workshape_country
						cbp_q8_7_note_current_owner_country_pass = yes
					}
				}
			}
		}
	}
}

cbp_q8_7_probe_candidate_global_market_workshape = {
	# Candidate Q8.7 shape: one global market pass, filtered to the same relevant
	# market set, then market-local present-country work-cache rebuild. No generated
	# goods, no stock mutation, and no trade-owner work is executed here.
	every_market_in_world = {
		save_temporary_scope_as = cbp_q8_7_shadow_comparison_market
		if = {
			limit = {
				has_global_variable_list = cbp_performance_relevant_markets
				is_target_in_global_variable_list = {
					name = cbp_performance_relevant_markets
					target = scope:cbp_q8_7_shadow_comparison_market
				}
			}
			cbp_q8_7_note_global_shadow_comparison_market = yes
			save_temporary_scope_as = cbp_market_country_cache_market
			remove_global_variable = cbp_market_country_cache_last_rebuild_failed
			cbp_rebuild_countries_present_in_market = yes
			cbp_q8_7_increment_candidate_global_cache_rebuild_count = yes
			if = {
				limit = { has_global_variable = cbp_market_country_cache_last_rebuild_failed }
				cbp_q8_7_increment_candidate_global_cache_failure_count = yes
			}
			if = {
				limit = { has_global_variable_list = cbp_countries_present_in_market }
				every_in_global_list = {
					variable = cbp_countries_present_in_market
					save_temporary_scope_as = cbp_q8_7_workshape_country
					cbp_q8_7_note_candidate_global_country_pass = yes
				}
			}
		}
	}
}
