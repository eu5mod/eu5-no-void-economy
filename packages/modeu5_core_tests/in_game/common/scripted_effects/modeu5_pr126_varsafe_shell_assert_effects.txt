	save_temporary_scope_value_as = { name = modeu5_pr126_shell_promoted_normal value = var:modeu5_pr126_ui_shell_promoted_normal }
	save_temporary_scope_value_as = { name = modeu5_pr126_shell_iterations_normal value = var:modeu5_pr126_ui_shell_iterations_normal }
	save_temporary_scope_value_as = { name = modeu5_pr126_shell_promoted_performance value = var:modeu5_pr126_ui_shell_promoted_performance }
	save_temporary_scope_value_as = { name = modeu5_pr126_shell_iterations_performance value = var:modeu5_pr126_ui_shell_iterations_performance }

}

	if = {
