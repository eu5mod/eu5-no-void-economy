modeu5_q8_7_probe_current_owner_noop_dispatcher_shadow = {
	# Current owner reference: market-center ownership workaround, filtered to the
	# same performance-relevant market set. This shadows the live branch order but
	# does not call capacity refresh, US-00, US-10, validation, or stock operators.
	every_country = {
		every_market_center_in_country = {
			save_temporary_scope_as = modeu5_q8_7_noop_market
			if = {
				limit = {
					has_global_variable_list = modeu5_performance_relevant_markets
					is_target_in_global_variable_list = {
						name = modeu5_performance_relevant_markets
						target = scope:modeu5_q8_7_noop_market
					}
				}
				modeu5_q8_7_note_noop_current_market = yes
				save_temporary_scope_as = modeu5_market_country_cache_market
				remove_global_variable = modeu5_market_country_cache_last_rebuild_failed
				modeu5_rebuild_countries_present_in_market = yes
				modeu5_q8_7_note_noop_current_cache_rebuild = yes
				if = {
					limit = { has_global_variable = modeu5_market_country_cache_last_rebuild_failed }
					modeu5_q8_7_note_noop_current_cache_failure = yes
				}
				if = {
					limit = { has_global_variable_list = modeu5_countries_present_in_market }
					every_in_global_list = {
						variable = modeu5_countries_present_in_market
						save_temporary_scope_as = modeu5_q8_7_noop_country
						modeu5_q8_7_note_noop_current_present_country = yes

				}
			}
		}
	}
}
