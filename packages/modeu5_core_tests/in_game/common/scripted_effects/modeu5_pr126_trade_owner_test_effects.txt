# PR126 PR6 country trade-owner pass probe.
#
# This probe validates the country-scoped trade-owner pass surface and a controlled
# explicit inter-market transfer request. It does not switch the live monthly
# dispatcher. Native every_trade quantities are read as diagnostics only; direct
# scoped transport-cost conversion is intentionally fail-closed in PR6.1 because
# the generated literal-good helpers are the safe conversion path.
#
# Vanilla trade cannot have source_market == target_market. Same-market local
# stock consumption is therefore intentionally absent from this probe.

modeu5_pr126_trade_owner_capture_metrics = {
	set_variable = { name = modeu5_pr126_ui_trade_owner_pass_runs value = 0 }
	set_variable = { name = modeu5_pr126_ui_trade_owner_trades_seen value = 0 }
	set_variable = { name = modeu5_pr126_ui_trade_owner_inter_market value = 0 }
	set_variable = { name = modeu5_pr126_ui_trade_owner_owner_mismatches value = 0 }
	set_variable = { name = modeu5_pr126_ui_trade_owner_confirmed value = 0 }
	set_variable = { name = modeu5_pr126_ui_trade_owner_blocked value = 0 }
	set_variable = { name = modeu5_pr126_ui_trade_owner_explicit_requests value = 0 }
	set_variable = { name = modeu5_pr126_ui_trade_owner_inter_market_transfers value = 0 }
	set_variable = { name = modeu5_pr126_ui_trade_owner_requested value = 0 }
	set_variable = { name = modeu5_pr126_ui_trade_owner_confirmed_quantity value = 0 }
	set_variable = { name = modeu5_pr126_ui_trade_owner_transferred value = 0 }
	set_variable = { name = modeu5_pr126_ui_trade_owner_unsatisfied value = 0 }

	if = { limit = { has_global_variable = modeu5_country_trade_owner_pass_runs } set_variable = { name = modeu5_pr126_ui_trade_owner_pass_runs value = global_var:modeu5_country_trade_owner_pass_runs } }
	if = { limit = { has_global_variable = modeu5_country_trade_owner_trades_seen } set_variable = { name = modeu5_pr126_ui_trade_owner_trades_seen value = global_var:modeu5_country_trade_owner_trades_seen } }
	if = { limit = { has_global_variable = modeu5_country_trade_owner_inter_market_trades } set_variable = { name = modeu5_pr126_ui_trade_owner_inter_market value = global_var:modeu5_country_trade_owner_inter_market_trades } }
	if = { limit = { has_global_variable = modeu5_country_trade_owner_owner_mismatches } set_variable = { name = modeu5_pr126_ui_trade_owner_owner_mismatches value = global_var:modeu5_country_trade_owner_owner_mismatches } }
	if = { limit = { has_global_variable = modeu5_country_trade_owner_quantity_confirmed_trades } set_variable = { name = modeu5_pr126_ui_trade_owner_confirmed value = global_var:modeu5_country_trade_owner_quantity_confirmed_trades } }
	if = { limit = { has_global_variable = modeu5_country_trade_owner_quantity_blocked_trades } set_variable = { name = modeu5_pr126_ui_trade_owner_blocked value = global_var:modeu5_country_trade_owner_quantity_blocked_trades } }
	if = { limit = { has_global_variable = modeu5_country_trade_owner_explicit_requests } set_variable = { name = modeu5_pr126_ui_trade_owner_explicit_requests value = global_var:modeu5_country_trade_owner_explicit_requests } }
	if = { limit = { has_global_variable = modeu5_country_trade_owner_inter_market_transfers } set_variable = { name = modeu5_pr126_ui_trade_owner_inter_market_transfers value = global_var:modeu5_country_trade_owner_inter_market_transfers } }
	if = { limit = { has_global_variable = modeu5_country_trade_owner_requested_quantity } set_variable = { name = modeu5_pr126_ui_trade_owner_requested value = global_var:modeu5_country_trade_owner_requested_quantity } }
	if = { limit = { has_global_variable = modeu5_country_trade_owner_confirmed_trade_quantity } set_variable = { name = modeu5_pr126_ui_trade_owner_confirmed_quantity value = global_var:modeu5_country_trade_owner_confirmed_trade_quantity } }
	if = { limit = { has_global_variable = modeu5_country_trade_owner_transferred_quantity } set_variable = { name = modeu5_pr126_ui_trade_owner_transferred value = global_var:modeu5_country_trade_owner_transferred_quantity } }
	if = { limit = { has_global_variable = modeu5_country_trade_owner_unsatisfied_quantity } set_variable = { name = modeu5_pr126_ui_trade_owner_unsatisfied value = global_var:modeu5_country_trade_owner_unsatisfied_quantity } }
}

modeu5_pr126_trade_owner_select_two_markets = {
	remove_global_variable = modeu5_test_pr126_trade_owner_market_selection_failed
	set_local_variable = { name = modeu5_pr126_trade_owner_market_index value = 0 }

	every_market_present_in_country = {
		if = {
			limit = {
				scope:modeu5_pr126_trade_owner_test_country = {
					local_var:modeu5_pr126_trade_owner_market_index = 0
				}
			}
			save_temporary_scope_as = modeu5_pr126_trade_owner_source_market
			# Default target to source so later comparisons never dereference an unset
			# target. A second market replaces this below; otherwise the probe blocks.
			save_temporary_scope_as = modeu5_pr126_trade_owner_target_market
		}
		else_if = {
			limit = {
				scope:modeu5_pr126_trade_owner_test_country = {
					local_var:modeu5_pr126_trade_owner_market_index = 1
				}
			}
			save_temporary_scope_as = modeu5_pr126_trade_owner_target_market
		}
		scope:modeu5_pr126_trade_owner_test_country = {
			change_local_variable = { name = modeu5_pr126_trade_owner_market_index add = 1 }
		}
	}

	if = {
		limit = { NOT = { exists = scope:modeu5_pr126_trade_owner_source_market } }
		set_global_variable = modeu5_test_pr126_trade_owner_market_selection_failed
	}
	else_if = {
		limit = { scope:modeu5_pr126_trade_owner_source_market = scope:modeu5_pr126_trade_owner_target_market }
		set_global_variable = modeu5_test_pr126_trade_owner_market_selection_failed
	}
}

modeu5_debug_run_pr126_country_trade_owner_pass_test = {
	remove_global_variable = modeu5_test_pr126_country_trade_owner_pass_passed
	remove_global_variable = modeu5_test_pr126_country_trade_owner_pass_failed
	remove_global_variable = modeu5_test_pr126_country_trade_owner_pass_blocked
	remove_global_variable = modeu5_test_pr126_trade_owner_market_selection_failed

	remove_variable = modeu5_pr126_ui_trade_owner_pass_runs
	remove_variable = modeu5_pr126_ui_trade_owner_trades_seen
	remove_variable = modeu5_pr126_ui_trade_owner_inter_market
	remove_variable = modeu5_pr126_ui_trade_owner_owner_mismatches
	remove_variable = modeu5_pr126_ui_trade_owner_confirmed
	remove_variable = modeu5_pr126_ui_trade_owner_blocked
	remove_variable = modeu5_pr126_ui_trade_owner_explicit_requests
	remove_variable = modeu5_pr126_ui_trade_owner_inter_market_transfers
	remove_variable = modeu5_pr126_ui_trade_owner_requested
	remove_variable = modeu5_pr126_ui_trade_owner_confirmed_quantity
	remove_variable = modeu5_pr126_ui_trade_owner_transferred
	remove_variable = modeu5_pr126_ui_trade_owner_unsatisfied

	save_temporary_scope_value_as = { name = modeu5_perf14_original_cmm_main_present value = 0 }
	save_temporary_scope_value_as = { name = modeu5_perf14_original_cmm_main_value value = 0 }
	if = {
		limit = {
			has_variable_map = cmm
			is_key_in_variable_map = {
				name = cmm
				target = flag:no_void_economy__nve_no_void_economy_main
			}
		}
		save_temporary_scope_value_as = { name = modeu5_perf14_original_cmm_main_present value = 1 }
		save_temporary_scope_value_as = {
			name = modeu5_perf14_original_cmm_main_value
			value = "variable_map(cmm|flag:no_void_economy__nve_no_void_economy_main)"
		}
	}

	debug_log = "ModeU5 TEST ENTERED scenario=pr126_country_trade_owner_pass"

	save_temporary_scope_as = modeu5_pr126_trade_owner_test_country
	modeu5_perf14_set_cmm_main_mode_for_test = { value = 2 }
	modeu5_reset_country_trade_owner_pass_metrics = yes
	modeu5_pr126_trade_owner_select_two_markets = yes

	if = {
		limit = { has_global_variable = modeu5_test_pr126_trade_owner_market_selection_failed }
		set_global_variable = modeu5_test_pr126_country_trade_owner_pass_blocked
		debug_log = "ModeU5 TEST BLOCKED scenario=pr126_country_trade_owner_pass reason=not_enough_distinct_current_country_markets"
	}
	else = {
		modeu5_run_monthly_country_trade_owner_cycle = yes

		modeu5_perf14_clear_wheat_stock_in_market = {
			market = scope:modeu5_pr126_trade_owner_source_market
		}
		modeu5_perf14_clear_wheat_stock_in_market = {
			market = scope:modeu5_pr126_trade_owner_target_market
		}
		modeu5_debug_replace_country_wheat_capacity = {
			country = scope:modeu5_pr126_trade_owner_test_country
			market = scope:modeu5_pr126_trade_owner_source_market
			value = 100
		}
		modeu5_debug_replace_country_wheat_capacity = {
			country = scope:modeu5_pr126_trade_owner_test_country
			market = scope:modeu5_pr126_trade_owner_target_market
			value = 100
		}
		modeu5_add_stock = {
			country = scope:modeu5_pr126_trade_owner_test_country
			market = scope:modeu5_pr126_trade_owner_source_market
			good = wheat
			quantity = 10
			capacity_policy = allow_over_capacity
		}

		modeu5_process_country_trade_owner_explicit_request = {
			buyer_country = scope:modeu5_pr126_trade_owner_test_country
			source_market = scope:modeu5_pr126_trade_owner_source_market
			target_market = scope:modeu5_pr126_trade_owner_target_market
			good = wheat
			requested_quantity = 4
		}

		modeu5_pr126_trade_owner_capture_metrics = yes
	}

	if = {
		limit = {
			NOT = { has_global_variable = modeu5_test_pr126_country_trade_owner_pass_blocked }
			global_var:modeu5_country_trade_owner_pass_runs = 1
			global_var:modeu5_country_trade_owner_explicit_requests = 1
			global_var:modeu5_country_trade_owner_inter_market_transfers = 1
			global_var:modeu5_country_trade_owner_requested_quantity = 4
			global_var:modeu5_country_trade_owner_transferred_quantity >= 4
			global_var:modeu5_country_trade_owner_unsatisfied_quantity = 0
		}
		set_global_variable = modeu5_test_pr126_country_trade_owner_pass_passed
		debug_log = "ModeU5 TEST PASS scenario=pr126_country_trade_owner_pass"
	}
	else_if = {
		limit = {
			NOT = { has_global_variable = modeu5_test_pr126_country_trade_owner_pass_blocked }
		}
		set_global_variable = modeu5_test_pr126_country_trade_owner_pass_failed
		debug_log = "ModeU5 TEST FAIL scenario=pr126_country_trade_owner_pass"
	}

	modeu5_perf14_restore_cmm_main_mode_after_test = yes
}
