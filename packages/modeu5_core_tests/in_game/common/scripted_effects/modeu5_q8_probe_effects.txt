# Q8.2 — US-10 aggregate pending-request gate probe.
#
# This does not add the aggregate gate. It proves that a country-market scoped
# probe can safely inspect the existing per-good pending-request map and record
# a value for one canonical good. If this is insufficient in runtime logs, the
# next implementation must add a generated has-any-pending bucket.
modeu5_q8_probe_us10_pending_gate = {
	set_global_variable = modeu5_test_q8_2_us10_pending_gate_probe_started
	save_temporary_scope_as = modeu5_q8_probe_country
	capital.market = {
		save_temporary_scope_as = modeu5_q8_probe_market
	}

	save_temporary_scope_value_as = { name = modeu5_q8_probe_wheat_pending_requested value = 0 }

	if = {
		limit = {
			has_variable_map = modeu5_consumption_wheat_pending_requested_by_market
			is_key_in_variable_map = {
				name = modeu5_consumption_wheat_pending_requested_by_market
				target = scope:modeu5_q8_probe_market
			}
		}
		save_temporary_scope_value_as = {
			name = modeu5_q8_probe_wheat_pending_requested
			value = "variable_map(modeu5_consumption_wheat_pending_requested_by_market|scope:modeu5_q8_probe_market)"
		}
	}

	set_variable = { name = modeu5_q8_probe_wheat_pending_requested value = scope:modeu5_q8_probe_wheat_pending_requested }

	if = {
		limit = { scope:modeu5_q8_probe_wheat_pending_requested >= 0 }
		set_global_variable = modeu5_test_q8_2_us10_pending_gate_probe_passed
		debug_log = "ModeU5 Q8.2 PROBE us10_pending_gate PASS wheat_pending=[THIS.GetVariable('modeu5_q8_probe_wheat_pending_requested').GetValue|4]"
	}
	else = {
		set_global_variable = modeu5_test_q8_2_us10_pending_gate_probe_failed
		error_log = "ModeU5 Q8.2 PROBE us10_pending_gate FAIL"
	}
}

# Q8.7 — global market-local pass proof.
#
# This deliberately lives in the test package, not live runtime. It proves more
# than raw iterator exposure: a global/none-scope entry can enter each market,
# deduplicate the market set, and call the existing market-country work-cache
# rebuild from market scope without stock mutation or a dispatcher switch.
modeu5_q8_probe_global_market_iterator_exposure = {
	set_global_variable = modeu5_test_q8_7_global_market_probe_started
	debug_log = "ModeU5 TEST ENTERED scenario=q87_global_market_local_pass"

	remove_global_variable = modeu5_test_q8_7_global_market_count
	remove_global_variable = modeu5_test_q8_7_global_market_unique_count
	remove_global_variable = modeu5_test_q8_7_global_market_duplicate_count
	remove_global_variable = modeu5_test_q8_7_market_country_cache_rebuild_count
	remove_global_variable = modeu5_test_q8_7_market_country_cache_failed_count
	set_global_variable = { name = modeu5_test_q8_7_global_market_count value = 0 }
	set_global_variable = { name = modeu5_test_q8_7_global_market_unique_count value = 0 }
	set_global_variable = { name = modeu5_test_q8_7_global_market_duplicate_count value = 0 }
	set_global_variable = { name = modeu5_test_q8_7_market_country_cache_rebuild_count value = 0 }
	set_global_variable = { name = modeu5_test_q8_7_market_country_cache_failed_count value = 0 }

	if = {
		limit = { has_global_variable_list = modeu5_q8_7_global_market_seen_markets }
		clear_global_variable_list = modeu5_q8_7_global_market_seen_markets
	}

	every_market_in_world = {
		save_temporary_scope_as = modeu5_q8_7_global_market_candidate
		save_temporary_scope_as = modeu5_market_country_cache_market
		modeu5_q8_7_increment_global_market_count = yes
		modeu5_q8_7_note_seen_global_market = yes
		modeu5_rebuild_countries_present_in_market = yes
		modeu5_q8_7_increment_market_country_cache_rebuild_count = yes
		if = {
			limit = { has_global_variable = modeu5_market_country_cache_last_rebuild_failed }
			modeu5_q8_7_increment_market_country_cache_failed_count = yes
		}
	}

	if = {
		limit = {
			global_var:modeu5_test_q8_7_global_market_count > 0
			global_var:modeu5_test_q8_7_global_market_unique_count > 0
			global_var:modeu5_test_q8_7_global_market_duplicate_count < 1
			global_var:modeu5_test_q8_7_market_country_cache_rebuild_count > 0
			global_var:modeu5_test_q8_7_market_country_cache_failed_count < 1
		}
		set_global_variable = modeu5_test_q8_7_global_market_probe_passed
		debug_log = "ModeU5 Q8.7 PROBE global_market_local_pass PASS"
		debug_log = "ModeU5 TEST PASS scenario=q87_global_market_local_pass"
	}
	else = {
		set_global_variable = modeu5_test_q8_7_global_market_probe_failed
		error_log = "ModeU5 Q8.7 PROBE global_market_local_pass FAIL"
		error_log = "ModeU5 TEST FAIL scenario=q87_global_market_local_pass"
	}
}
