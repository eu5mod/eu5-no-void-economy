# PERF-14 guarded runner variant for the Q8.7 market-boundary spec.
#
# The Performance Mode detailed boundary is a human-relevant market. Once a market
# is relevant, AI-controlled countries present in that market are part of the
# market-local stock mutation surface. This test keeps the older guarded checks
# but replaces the previous random-AI-capital fixture with an AI country that is
# actually present in the same human-relevant market used by the core PERF-14
# promotion fixture.

cbp_perf14_guarded_probe_ai_country_in_human_relevant_market = {
	save_temporary_scope_value_as = { name = test_cbp_perf14_ai_fixture_available value = 0 }

	cbp_perf14_set_cmm_main_mode_for_test = { value = 1 }
	cbp_clear_detailed_accounting_promoted_markets = yes
	cbp_clear_performance_relevant_markets = yes
	cbp_mark_human_relevant_market_for_country = {
		country = scope:cbp_perf14_test_country
		market = scope:cbp_perf14_test_market
	}

	# Prefer the already-controlled PERF-14 market fixture. Earlier guarded probes
	# seed and promote this market successfully, so this boundary test stays focused
	# on the AI-in-relevant-market gate instead of accidentally selecting an arbitrary
	# human-relevant market with weak promotion materialisation data.
	scope:cbp_perf14_test_market = {
		save_temporary_scope_as = cbp_perf14_candidate_market
		save_temporary_scope_as = cbp_market_country_cache_market
	}
	cbp_rebuild_countries_present_in_market = yes
	if = {
		limit = { has_global_variable_list = cbp_countries_present_in_market }
		every_in_global_list = {
			variable = cbp_countries_present_in_market
			save_temporary_scope_as = cbp_perf14_candidate_country
			if = {
				limit = {
					scope:test_cbp_perf14_ai_fixture_available = 0
					scope:cbp_perf14_candidate_country = { is_ai = yes }
				}
				save_temporary_scope_value_as = { name = test_cbp_perf14_ai_fixture_available value = 1 }
				scope:cbp_perf14_candidate_country = { save_temporary_scope_as = cbp_perf14_ai_country }
				scope:cbp_perf14_candidate_market = { save_temporary_scope_as = cbp_perf14_ai_market }
			}
		}
	}

	if = {
		limit = { scope:test_cbp_perf14_ai_fixture_available > 0 }
		cbp_clear_detailed_accounting_promoted_markets = yes
		cbp_clear_performance_relevant_markets = yes
		cbp_mark_human_relevant_market_for_country = {
			country = scope:cbp_perf14_test_country
			market = scope:cbp_perf14_ai_market
		}

		cbp_prepare_country_market_accounting_decision = { country = scope:cbp_perf14_ai_country market = scope:cbp_perf14_ai_market }
		if = {
			limit = {
				OR = {
					NOT = { cbp_detailed_country_market_accounting_enabled_trigger = yes }
					cbp_market_level_fallback_required_trigger = yes
					scope:cbp_accounting_decision_country_is_human > 0
					NOT = { scope:cbp_accounting_decision_market_human_relevant > 0 }
				}
			}
			debug_log = "ModeU5 PERF-14 FAIL_REASON reason=ai_relevant_market_decision_unexpected_state"
			set_global_variable = test_cbp_perf14_performance_mode_cmm_failed
		}

		cbp_prepare_stock_mutation_accounting_mode = { country = scope:cbp_perf14_ai_country market = scope:cbp_perf14_ai_market }
		scope:cbp_perf14_ai_market = { save_temporary_scope_as = cbp_market }
		set_variable = { name = test_cbp_perf14_ai_debug_detailed value = scope:cbp_stock_mutation_use_detailed_accounting }
		set_variable = { name = test_cbp_perf14_ai_debug_fallback value = scope:cbp_stock_mutation_use_market_level_fallback }
		set_variable = { name = test_cbp_perf14_ai_debug_blocked value = scope:cbp_stock_mutation_blocked }
		set_variable = { name = test_cbp_perf14_ai_debug_market_relevant value = scope:cbp_accounting_decision_market_human_relevant }
		set_variable = { name = test_cbp_perf14_ai_debug_promotion_attempted value = scope:cbp_stock_mutation_promotion_attempted }
		set_variable = { name = test_cbp_perf14_ai_debug_promotion_succeeded value = scope:cbp_stock_mutation_promotion_succeeded }
		debug_log = "ModeU5 PERF-14 AI_RELEVANT_MARKET_GATE detailed=[THIS.GetVariable('cbp_perf14_ai_debug_detailed').GetValue|0] fallback=[THIS.GetVariable('cbp_perf14_ai_debug_fallback').GetValue|0] blocked=[THIS.GetVariable('cbp_perf14_ai_debug_blocked').GetValue|0] market_human_relevant=[THIS.GetVariable('cbp_perf14_ai_debug_market_relevant').GetValue|0] promotion_attempted=[THIS.GetVariable('cbp_perf14_ai_debug_promotion_attempted').GetValue|0] promotion_succeeded=[THIS.GetVariable('cbp_perf14_ai_debug_promotion_succeeded').GetValue|0] promotion_result=[GetGlobalVariable('cbp_perf14_promotion_last_result').GetValue|0] boundary=human_relevant_market country=ai"
		if = {
			limit = {
				OR = {
					NOT = { cbp_stock_mutation_use_detailed_accounting_trigger = yes }
					cbp_stock_mutation_use_market_level_fallback_trigger = yes
					cbp_stock_mutation_blocked_trigger = yes
					NOT = { cbp_market_detailed_accounting_promoted_trigger = yes }
				}
			}
			debug_log = "ModeU5 PERF-14 FAIL_REASON reason=ai_in_human_relevant_market_stock_mutation_unexpected_state"
			set_global_variable = test_cbp_perf14_performance_mode_cmm_failed
		}

		debug_log = "ModeU5 PERF-14 AI_RELEVANT_MARKET boundary=human_relevant_market country=ai result=checked"
	}
	else = {
		set_global_variable = test_cbp_perf14_performance_mode_cmm_blocked
		error_log = "ModeU5 PERF-14 performance mode CMM probe blocked because no AI country was present in the controlled human-relevant market fixture."
	}
}

cbp_debug_run_perf14_performance_mode_cmm_test_guarded_market_boundary = {
	cbp_perf14_clear_test_results = yes
	cbp_enter_test_audit_runtime_mode = yes
	set_global_variable = test_cbp_perf14_performance_mode_cmm_started
	debug_log = "ModeU5 TEST ENTERED scenario=perf14_performance_mode_cmm"

	save_temporary_scope_as = cbp_perf14_test_country
	save_temporary_scope_value_as = { name = test_cbp_perf14_original_cmm_main_present value = 0 }
	save_temporary_scope_value_as = { name = test_cbp_perf14_original_cmm_main_value value = 1 }
	save_temporary_scope_value_as = { name = test_cbp_perf14_ai_fixture_available value = 0 }
	save_temporary_scope_value_as = { name = test_cbp_perf14_nonpresent_market_fixture_available value = 0 }

	if = {
		limit = { has_variable_map = cmm is_key_in_variable_map = { name = cmm target = flag:no_void_economy__nve_no_void_economy_main } }
		save_temporary_scope_value_as = { name = test_cbp_perf14_original_cmm_main_present value = 1 }
		save_temporary_scope_value_as = { name = test_cbp_perf14_original_cmm_main_value value = "variable_map(cmm|flag:no_void_economy__nve_no_void_economy_main)" }
	}

	if = {
		limit = { has_markets = no }
		set_global_variable = test_cbp_perf14_performance_mode_cmm_blocked
		error_log = "ModeU5 PERF-14 performance mode CMM probe blocked because the current country has no markets."
	}
	else = {
		capital.market = { save_temporary_scope_as = cbp_perf14_test_market }
		cbp_perf14_guarded_check_performance_mode_gate = yes
		cbp_perf14_guarded_probe_stock_mutation_gate_promotes = yes
		cbp_perf14_guarded_probe_market_runtime_gate = yes
		cbp_perf14_guarded_probe_positive_promotion = yes
		cbp_perf14_guarded_probe_idempotent_promotion = yes
		cbp_perf14_guarded_probe_sparse_supplier_scan = yes
		cbp_perf14_guarded_probe_partial_promotion = yes
		cbp_perf14_guarded_probe_ai_country_in_human_relevant_market = yes
		cbp_perf14_guarded_probe_nonpresent_block = yes
		cbp_perf14_guarded_probe_normal_mode = yes
		cbp_perf14_guarded_probe_deactivated_mode = yes
		if = {
			limit = { NOT = { has_global_variable = test_cbp_perf14_performance_mode_cmm_failed } NOT = { has_global_variable = test_cbp_perf14_performance_mode_cmm_blocked } }
			set_global_variable = test_cbp_perf14_performance_mode_cmm_passed
		}
	}

	cbp_perf14_restore_cmm_main_mode_after_test = yes
	cbp_perf14_log_probe_result = yes
}
