namespace = modeu5_us20_probe

# Experimental US-20 promoted-market probe.
# This intentionally stays outside modeu5_revalidate_debug.1 so the stable
# baseline summary remains available even if this larger E2E probe needs work.
#
# The console launcher must not execute the logging effect directly. EU5 disables
# localization while processing a console game command, and a synchronous first
# debug_log can therefore trigger "Tried to localize with localization disabled".
# The launcher schedules a hidden continuation so probe logging starts in the
# ordinary event context instead.

modeu5_us20_probe.1 = {
	type = country_event
	title = modeu5_us20_probe.1.title
	desc = modeu5_us20_probe.1.desc
	outcome = neutral
	orphan = yes

	trigger = {
		always = no
	}

	option = {
		name = modeu5_us20_probe.1.a
		trigger_event_non_silently = { id = modeu5_us20_probe.10 days = 0 }
	}
	option = {
		name = modeu5_debug.close
	}
}
