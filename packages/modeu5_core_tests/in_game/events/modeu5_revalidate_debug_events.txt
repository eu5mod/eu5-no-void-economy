namespace = modeu5_revalidate_debug

# Revalidation execution contract:
# - Revalidation is a manual/debug console event chain only; never call it from
#   monthly/yearly on_actions or any runtime tick.
# - A full revalidation run is split into at most 10 scenario steps. The final
#   summary event is not a scenario step and must not run additional tests.
# - Continuations use days = 0 so the chain remains on the same in-game day and
#   does not cross a monthly tick between scenarios.
# - If adding an 11th scenario, split the suite into a separate manual entry
#   event instead of extending this chain.

modeu5_revalidate_debug.1 = {
	type = country_event
	title = modeu5_revalidate_debug.1.title
	desc = modeu5_revalidate_debug.1.desc
	outcome = neutral
	orphan = yes

	trigger = {
		always = no
	}

	option = {
		name = modeu5_revalidate_debug.1.a
		modeu5_revalidate_clear_results = yes
		set_global_variable = modeu5_revalidation_started
		# Step 1/10
		modeu5_revalidate_run_us02_capacity = yes
		trigger_event_non_silently = { id = modeu5_revalidate_debug.10 days = 0 }
	}
	option = {
		name = modeu5_debug.close
	}
}
