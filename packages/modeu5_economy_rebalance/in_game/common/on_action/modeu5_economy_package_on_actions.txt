# Snapshot package presence and ensure the permanent US-09 R.G.O. capacity modifier is applied.
# The load hook also supports saves created before this location modifier was introduced.
on_game_start = {
	on_actions = {
		modeu5_economy_rebalance_package_initialize
	}
}

on_game_load = {
	on_actions = {
		modeu5_economy_rebalance_package_initialize
	}
}

modeu5_economy_rebalance_package_initialize = {
	effect = {
		set_global_variable = cbp_economy_rebalance_loaded
		set_global_variable = {
			name = cbp_economy_package_version
			value = 100
		}
		modeu5_apply_us09_base_rgo_size_bonus = yes
	}
}
