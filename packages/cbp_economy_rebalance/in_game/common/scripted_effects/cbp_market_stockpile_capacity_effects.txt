# Synchronize vanilla market stockpile capacity with CBP's authoritative
# country x market capacity records.
#
# Scope contract:
# - $market$ is the market whose center location receives the modifier;
# - country capacity records should already have been refreshed for the market;
# - the market-center location is scope:cbp_market.location.

cbp_refresh_market_center_stockpile_capacity = {
	$market$ = { save_temporary_scope_as = cbp_market }

	cbp_prepare_promoted_market_country_cache = {
		market = scope:cbp_market
	}

	scope:cbp_market.location = {
		set_local_variable = {
			name = cbp_market_stockpile_capacity_accumulator
			value = 0
		}
		set_local_variable = {
			name = cbp_market_stockpile_capacity_country_count
			value = 0
		}
	}

	if = {
		limit = { has_global_variable_list = cbp_countries_present_in_market }
		every_in_global_list = {
			variable = cbp_countries_present_in_market
			save_temporary_scope_as = cbp_market_capacity_country

			if = {
				limit = {
					has_variable_map = cbp_stock_cap_by_market
					is_key_in_variable_map = {
						name = cbp_stock_cap_by_market
						target = scope:cbp_market
					}
				}
				save_temporary_scope_value_as = {
					name = cbp_market_current_country_capacity
					value = "variable_map(cbp_stock_cap_by_market|scope:cbp_market)"
				}
				scope:cbp_market.location = {
					change_local_variable = {
						name = cbp_market_stockpile_capacity_accumulator
						add = scope:cbp_market_current_country_capacity
					}
					change_local_variable = {
						name = cbp_market_stockpile_capacity_country_count
						add = 1
					}
				}
			}
		}
	}

	scope:cbp_market.location = {
		remove_location_modifier = cbp_market_stockpile_capacity
		add_location_modifier = {
			modifier = cbp_market_stockpile_capacity
			months = 2
			mode = replace
			size = local_var:cbp_market_stockpile_capacity_accumulator
			recalculate_immediately = yes
		}
		if = {
			limit = { has_global_variable = test_cbp_pr182_market_stockpile_probe_active }
			debug_log = "ModeU5 PR-182 STOCKPILE_REFRESH capacity=[THIS.GetVariable('cbp_market_stockpile_capacity_accumulator').GetValue|2] countries=[THIS.GetVariable('cbp_market_stockpile_capacity_country_count').GetValue|0]"
		}
		remove_local_variable = cbp_market_stockpile_capacity_accumulator
		remove_local_variable = cbp_market_stockpile_capacity_country_count
	}
}

cbp_refresh_market_center_stockpile_capacity_for_current_country = {
	every_market_center_in_country = {
		save_temporary_scope_as = cbp_market
		cbp_refresh_market_center_stockpile_capacity = {
			market = scope:cbp_market
		}
	}
}

cbp_refresh_all_market_center_stockpile_capacity = {
	every_country = {
		cbp_refresh_market_center_stockpile_capacity_for_current_country = yes
	}
}
