#   vanilla gross factor = 1 + sum(vanilla minting_income_factor sources)
#   CBP gross factor     = 1 + 1 + sum(2 * vanilla sources)
#                        = 2 * vanilla gross factor
# The second half of this identity is generated as exact-path vanilla-file
# overrides by tools/generate_us177_minting_overrides.py.
cbp_us177_double_minting_base = {
	requires_real = no
	minting_income_factor = 1.0
}
