# Snapshot package presence for a new campaign. Existing-save package changes are unsupported.
on_game_start = {
	on_actions = {
		cbp_war_rebalance_package_on_game_start
	}
}

cbp_war_rebalance_package_on_game_start = {
	effect = {
		set_global_variable = cbp_war_rebalance_loaded
		set_global_variable = {
			name = cbp_war_package_version
			value = 100
		}
	}
}
