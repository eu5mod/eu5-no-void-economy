# Snapshot package presence for a new campaign and repair missing markers on load.
# Existing-save package changes are still unsupported.
on_game_start = {
	on_actions = {
		cbp_trade_rebalance_package_on_game_start
	}
}

on_game_load = {
	on_actions = {
		cbp_trade_rebalance_package_on_game_load
	}
}

cbp_trade_rebalance_package_on_game_start = {
	effect = {
		set_global_variable = cbp_trade_rebalance_loaded
		set_global_variable = {
			name = cbp_trade_package_version
			value = 100
		}
	}
}

cbp_trade_rebalance_package_on_game_load = {
	effect = {
		set_global_variable = cbp_trade_rebalance_loaded
		set_global_variable = {
			name = cbp_trade_package_version
			value = 100
		}
	}
}
