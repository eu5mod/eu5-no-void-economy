	has_variable_map = cmm
	OR = {
		# CMM displays the registered default (enabled) without necessarily
		# materializing a map entry. Only an explicit zero disables US-17/US-20.
		NOT = {
			is_key_in_variable_map = {
				name = cmm
				target = flag:no_void_economy__cbp_general_gameplay_gameplay_trade_rework_settings
			}
		}
		AND = {
			is_key_in_variable_map = {
				name = cmm
				target = flag:no_void_economy__cbp_general_gameplay_gameplay_trade_rework_settings
			}
			"variable_map(cmm|flag:no_void_economy__cbp_general_gameplay_gameplay_trade_rework_settings)" = 1
		}
	}
}
