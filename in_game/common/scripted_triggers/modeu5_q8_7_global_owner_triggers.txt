# Q8.7 live global market-owner switch.
#
# Default is enabled for the final Q8.7 implementation PR. The old
# market-center owner remains available by setting
# modeu5_q8_7_live_global_market_owner_disabled from a debug/fallback event.

modeu5_q8_7_live_global_market_owner_enabled_trigger = {
	NOT = { has_global_variable = modeu5_q8_7_live_global_market_owner_disabled }
}

modeu5_q8_7_live_global_market_owner_disabled_trigger = {
	has_global_variable = modeu5_q8_7_live_global_market_owner_disabled
}
