cbp_initialization_complete_trigger = {
	has_global_variable = cbp_stock_schema_version
	has_global_variable = cbp_initialization_state
	global_var:cbp_stock_schema_version = cbp_current_stock_schema_version
	global_var:cbp_initialization_state = 2
}

cbp_initialization_not_started_trigger = {
	NOT = { has_global_variable = cbp_stock_schema_version }
	NOT = { has_global_variable = cbp_initialization_state }
}

cbp_initialization_failed_trigger = {
	has_global_variable = cbp_initialization_state
	global_var:cbp_initialization_state = -1
}

cbp_initialization_in_progress_trigger = {
	has_global_variable = cbp_initialization_state
	global_var:cbp_initialization_state = 1
}
