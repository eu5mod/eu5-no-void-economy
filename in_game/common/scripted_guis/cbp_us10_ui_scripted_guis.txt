cbp_us10_ui_refresh_on_click = {
	scope = country

	is_shown = {
		always = yes
	}

	is_valid = {
		always = yes
	}

	effect = {
		cbp_us10_ui_prepare_current_market_table_for_production_ui = yes
	}
}

cbp_us10_ui_select_market_on_click = {
	scope = country

	is_shown = {
		always = yes
	}

	is_valid = {
		always = yes
	}

	effect = {
		if = {
			limit = {
				scope:cbp_us10_ui_clicked_market ?= { always = yes }
			}
			cbp_us10_ui_set_selected_market = { market = scope:cbp_us10_ui_clicked_market }
		}
		cbp_us10_ui_prepare_current_market_table_for_production_ui = yes
	}
}

cbp_us10_ui_next_market_on_click = {
	scope = country

	is_shown = {
		always = yes
	}

	is_valid = {
		always = yes
	}

	effect = {
		cbp_us10_ui_select_next_market = yes
		cbp_us10_ui_prepare_current_market_table_for_production_ui = yes
	}
}
