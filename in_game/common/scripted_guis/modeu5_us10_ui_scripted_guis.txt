modeu5_us10_ui_refresh_on_click = {
	scope = country

	is_shown = {
		always = yes
	}

	is_valid = {
		always = yes
	}

	effect = {
		modeu5_us10_ui_prepare_current_market_table_for_production_ui = yes
	}
}

modeu5_us10_ui_select_market_on_click = {
	scope = country

	is_shown = {
		always = yes
	}

	is_valid = {
		always = yes
	}

	effect = {
		if = {
			limit = {
				scope:modeu5_us10_ui_clicked_market ?= { always = yes }
			}
			modeu5_us10_ui_set_selected_market = { market = scope:modeu5_us10_ui_clicked_market }
		}
		modeu5_us10_ui_prepare_current_market_table_for_production_ui = yes
	}
}

modeu5_us10_ui_next_market_on_click = {
	scope = country

	is_shown = {
		always = yes
	}

	is_valid = {
		always = yes
	}

	effect = {
		modeu5_us10_ui_select_next_market = yes
		modeu5_us10_ui_prepare_current_market_table_for_production_ui = yes
	}
}
