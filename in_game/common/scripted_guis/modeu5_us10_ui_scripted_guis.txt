modeu5_us10_ui_refresh_on_click = {
	scope = country

	is_shown = {
		always = yes
	}

	is_valid = {
		always = yes
	}

	effect = {
		modeu5_us10_ui_prepare_current_market_table = yes
	}
}

modeu5_us10_ui_select_market_on_click = {
	scope = country

	is_shown = {
		always = yes
	}

	is_valid = {
		always = yes
	}

	effect = {
		# Safety fallback: do not reference a GUI-provided market scope here.
		# The previous implementation used scope:modeu5_us10_ui_clicked_market,
		# which Jomini validates as an unset event target.
		modeu5_us10_ui_prepare_current_market_table = yes
	}
}

modeu5_us10_ui_next_market_on_click = {
	scope = country

	is_shown = {
		always = yes
	}

	is_valid = {
		always = yes
	}

	effect = {
		modeu5_us10_ui_select_next_market = yes
		modeu5_us10_ui_prepare_current_market_table = yes
	}
}
