# US-05 ModeU5 Economic Base values.
#
# This file deliberately exposes the replacement base without attempting a
# monthly gold/modifier reconciliation. The remaining integration boundary is
# the direct Stability / legitimacy-producing Court-Government Power cost call
# site described by TECH-01 #044.

modeu5_slider_cost_base = {
	value = 0
	if = {
		limit = { has_variable = cbp_country_wealth_endpoint }
		value = var:cbp_country_wealth_endpoint
		add = monthly_trade_income
	}
}

modeu5_slider_cost_base_wealth_component = {
	value = 0
	if = {
		limit = { has_variable = cbp_country_wealth_endpoint }
		value = var:cbp_country_wealth_endpoint
	}
}

modeu5_slider_cost_base_trade_income_component = {
	value = monthly_trade_income
}
