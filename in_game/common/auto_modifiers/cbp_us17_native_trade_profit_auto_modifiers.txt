# US-17 keeps Selling Efficiency on its native price surface.
# Import and Export Efficiency are cancelled from price formation and transferred
# into the reciprocal Merchant Maintenance formula:
#
#   effective maintenance = 1 - 1 / (1 + M / 2 + (I + Ex) * 5)
#
# The persisted maintenance correction is the difference between that target and
# the reconstructed non-CBP Merchant Maintenance Efficiency baseline.

cbp_us17_import_efficiency_cancellation = {
	requires_real = no
	potential_trigger = { cbp_trade_rework_enabled_trigger = yes }
	limit = { has_variable = cbp_us17_native_import_correction }
	scales_with = { value = var:cbp_us17_native_import_correction }
	import_efficiency = 1
}

cbp_us17_export_efficiency_cancellation = {
	requires_real = no
	potential_trigger = { cbp_trade_rework_enabled_trigger = yes }
	limit = { has_variable = cbp_us17_native_export_correction }
	scales_with = { value = var:cbp_us17_native_export_correction }
	export_efficiency = 1
}

cbp_us17_merchant_maintenance_reconciliation = {
	requires_real = no
	potential_trigger = { cbp_trade_rework_enabled_trigger = yes }
	limit = { has_variable = cbp_us17_native_maintenance_correction }
	scales_with = { value = var:cbp_us17_native_maintenance_correction }
	merchant_maintenance_efficiency = 1
}
