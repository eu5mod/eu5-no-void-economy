# US-17 keeps trade profit inside the Vanilla engine so route selection, AI,
# tooltips, and treasury accounting all consume the same result. The persisted
# scales are refreshed from country scope; each modifier contributes exactly the
# correction calculated from the unmodified country baseline.

cbp_us17_selling_efficiency_cancellation = {
	requires_real = no
	potential_trigger = { cbp_trade_rework_enabled_trigger = yes }
	limit = { has_variable = cbp_us17_native_selling_correction }
	scales_with = { value = var:cbp_us17_native_selling_correction }
	selling_efficiency = 1
}

	limit = { has_variable = cbp_us17_native_import_correction }
	scales_with = { value = var:cbp_us17_native_import_correction }
