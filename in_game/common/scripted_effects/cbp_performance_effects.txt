# ModeU5 sparse traversal helpers and debug-only performance counters.
#
# PERF-02 builds the country -> native market side. PERF-03 adds a
# market -> countries-present work cache in cbp_market_country_cache_effects.
# PERF-13 counters are intentionally inert in normal runtime unless a
# debug/test path calls them. They do not mutate stock.

cbp_clear_performance_relevant_markets = {
	remove_global_variable = cbp_performance_human_country_count
	remove_global_variable = cbp_performance_relevant_market_native_iterations
	remove_global_variable = cbp_performance_relevant_market_added_count

	set_global_variable = { name = cbp_performance_human_country_count value = 0 }
	set_global_variable = { name = cbp_performance_relevant_market_native_iterations value = 0 }
	set_global_variable = { name = cbp_performance_relevant_market_added_count value = 0 }

	if = {
		limit = { has_global_variable_list = cbp_performance_relevant_markets }
		clear_global_variable_list = cbp_performance_relevant_markets
	}
}

cbp_prepare_performance_relevant_market_counters = {
	if = {
		limit = { NOT = { has_global_variable = cbp_performance_human_country_count } }
		set_global_variable = { name = cbp_performance_human_country_count value = 0 }
	}
	if = {
		limit = { NOT = { has_global_variable = cbp_performance_relevant_market_native_iterations } }
		set_global_variable = { name = cbp_performance_relevant_market_native_iterations value = 0 }
	}
	if = {
		limit = { NOT = { has_global_variable = cbp_performance_relevant_market_added_count } }
		set_global_variable = { name = cbp_performance_relevant_market_added_count value = 0 }
	}
}

cbp_mark_performance_relevant_market = {
	cbp_prepare_performance_relevant_market_counters = yes

	if = {
		limit = {
			NOT = { has_global_variable_list = cbp_performance_relevant_markets }
		}
		add_to_global_variable_list = {
			name = cbp_performance_relevant_markets
			target = scope:cbp_performance_relevant_market
		}
		set_global_variable = {
			name = cbp_performance_relevant_market_added_count
			value = {
				value = global_var:cbp_performance_relevant_market_added_count
				add = 1
			}
		}
	}
	else_if = {
		limit = {
			NOT = {
				is_target_in_global_variable_list = {
					name = cbp_performance_relevant_markets
					target = scope:cbp_performance_relevant_market
				}
			}
		}
		add_to_global_variable_list = {
			name = cbp_performance_relevant_markets
			target = scope:cbp_performance_relevant_market
		}
		set_global_variable = {
			name = cbp_performance_relevant_market_added_count
			value = {
				value = global_var:cbp_performance_relevant_market_added_count
				add = 1
			}
		}
	}
}

cbp_mark_human_relevant_market_for_country = {
	$country$ = { save_temporary_scope_as = cbp_performance_country }
	$market$ = { save_temporary_scope_as = cbp_performance_relevant_market }

	if = {
		limit = {
			cbp_performance_mode_enabled_trigger = yes
			scope:cbp_performance_country = { is_ai = no }
		}
		cbp_mark_performance_relevant_market = yes
	}
}

cbp_mark_human_relevant_market_if_present = {
	$country$ = { save_temporary_scope_as = cbp_performance_country }
	$market$ = {
		save_temporary_scope_as = cbp_performance_target_market
		save_temporary_scope_as = cbp_performance_relevant_market
	}

	save_temporary_scope_value_as = { name = cbp_performance_target_market_present value = 0 }

	if = {
		limit = {
			cbp_performance_mode_enabled_trigger = yes
			scope:cbp_performance_country = { is_ai = no }
		}
		scope:cbp_performance_country = {
			set_local_variable = { name = cbp_performance_target_market_present value = 0 }
			every_market_present_in_country = {
				save_temporary_scope_as = cbp_performance_iterated_market
				if = {
					limit = {
						scope:cbp_performance_iterated_market = scope:cbp_performance_target_market
					}
					scope:cbp_performance_country = {
						set_local_variable = { name = cbp_performance_target_market_present value = 1 }
					}
				}
			}
			save_temporary_scope_value_as = {
				name = cbp_performance_target_market_present
				value = local_var:cbp_performance_target_market_present
			}
		}
		if = {
			limit = { scope:cbp_performance_target_market_present > 0 }
			cbp_mark_performance_relevant_market = yes
		}
	}
}

cbp_add_country_present_markets_to_performance_relevant_list = {
	cbp_prepare_performance_relevant_market_counters = yes

	scope:cbp_performance_country = {
		every_market_present_in_country = {
			save_temporary_scope_as = cbp_performance_relevant_market
			set_global_variable = {
				name = cbp_performance_relevant_market_native_iterations
				value = {
					value = global_var:cbp_performance_relevant_market_native_iterations
					add = 1
				}
			}
			cbp_mark_performance_relevant_market = yes
		}
	}
}

	$country$ = { save_temporary_scope_as = cbp_performance_country }

	if = {

			cbp_performance_mode_enabled_trigger = yes
			scope:cbp_performance_country = { is_ai = no }

}

cbp_rebuild_current_country_relevant_markets = {
	save_temporary_scope_as = cbp_performance_country
	cbp_clear_performance_relevant_markets = yes
	cbp_add_country_present_markets_to_performance_relevant_list = yes
}

cbp_rebuild_performance_relevant_markets_for_country = {
	$country$ = { save_temporary_scope_as = cbp_performance_country }
	cbp_clear_performance_relevant_markets = yes
	cbp_add_country_present_markets_to_performance_relevant_list = yes
}

cbp_rebuild_human_relevant_markets = {
	cbp_clear_performance_relevant_markets = yes

	every_country = {
		limit = { is_ai = no }
		save_temporary_scope_as = cbp_performance_country
		set_global_variable = {
			name = cbp_performance_human_country_count
			value = {
				value = global_var:cbp_performance_human_country_count
				add = 1
			}
		}
		cbp_add_country_present_markets_to_performance_relevant_list = yes
	}
}

cbp_prepare_human_relevant_full_ledger_markets = {
	if = {
		limit = { cbp_accounting_human_relevant_persistence_trigger = yes }
		save_temporary_scope_value_as = {
			name = cbp_human_relevant_full_ledger_stamp
			value = {
				value = current_year
				multiply = 12
				add = current_month
			}
		}
		if = {
			limit = {
				OR = {
					NOT = { has_global_variable = cbp_human_relevant_full_ledger_stamp }
					global_var:cbp_human_relevant_full_ledger_stamp != scope:cbp_human_relevant_full_ledger_stamp
				}
			}
			set_global_variable = {
				name = cbp_human_relevant_full_ledger_stamp
				value = scope:cbp_human_relevant_full_ledger_stamp
			}
			cbp_rebuild_human_relevant_markets = yes
		}
	}
}

	cbp_rebuild_countries_present_in_market = yes

	if = {
		limit = { has_global_variable_list = cbp_countries_present_in_market }
		every_in_global_list = {
			variable = cbp_countries_present_in_market
			save_temporary_scope_as = cbp_country
			scope:cbp_promotion_market = { save_temporary_scope_as = cbp_market }
			cbp_recalculate_country_market_capacity_shared = yes

				name = cbp_perf14_promotion_capacity_country_count
				value = {
					value = global_var:cbp_perf14_promotion_capacity_country_count
					add = 1
				}
			}
		}
	}
}

cbp_promote_market_to_detailed_accounting = {
	$market$ = {
		save_temporary_scope_as = cbp_promotion_market
		save_temporary_scope_as = cbp_market
	}

	cbp_reset_market_promotion_metrics = yes

	if = {
		limit = { cbp_market_detailed_accounting_promoted_trigger = yes }
		set_global_variable = { name = cbp_perf14_promotion_last_result value = 1 }
		set_global_variable = { name = cbp_perf14_promotion_noop_already_promoted value = 1 }
	}
	else_if = {
		limit = {
			OR = {
				NOT = { has_global_variable_list = cbp_performance_relevant_markets }
				NOT = {
					is_target_in_global_variable_list = {
						name = cbp_performance_relevant_markets
						target = scope:cbp_promotion_market
					}
				}
			}
		}
		set_global_variable = { name = cbp_perf14_promotion_blocked_not_human_relevant value = 1 }
	}
	else = {
		cbp_refresh_market_promotion_country_capacities = yes
		cbp_promote_market_to_detailed_accounting_all_goods = yes

		if = {
			limit = { NOT = { has_global_variable = cbp_perf14_promotion_failure_detected } }
			cbp_mark_market_detailed_accounting_promoted = yes
			set_global_variable = { name = cbp_perf14_promotion_last_result value = 1 }
		}
		else_if = {
			limit = { global_var:cbp_perf14_promotion_failure_detected <= 0 }
			cbp_mark_market_detailed_accounting_promoted = yes
			set_global_variable = { name = cbp_perf14_promotion_last_result value = 1 }
		}
	}
}
