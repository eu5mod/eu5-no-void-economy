# US-05 monthly application of per-expense reconciliation ratios.
#
# This effect runs after PR #166 refreshes cbp_country_wealth_endpoint. Court and
# Diplomatic modifiers are refreshed independently so their target bases can
# diverge without changing the runtime architecture.

modeu5_us05_refresh_safe_slider_cost_reconciliation = {
	remove_country_modifier = modeu5_us05_court_cost_reconciliation
	remove_country_modifier = modeu5_us05_diplomatic_cost_reconciliation

	if = {
		limit = {
			has_variable = cbp_country_wealth_endpoint
			modeu5_us05_vanilla_economic_base > 0
		}
		add_country_modifier = {
			modifier = modeu5_us05_court_cost_reconciliation
			years = -1
			mode = replace
			size = modeu5_us05_court_cost_modifier
		}
		add_country_modifier = {
			modifier = modeu5_us05_diplomatic_cost_reconciliation
			years = -1
			mode = replace
			size = modeu5_us05_diplomatic_cost_modifier
		}
	}

	set_variable = {
		name = cbp_us05_court_cost_modifier
		value = modeu5_us05_court_cost_modifier
	}
	set_variable = {
		name = cbp_us05_court_target_base
		value = modeu5_us05_court_target_base
	}
	set_variable = {
		name = cbp_us05_diplomatic_cost_modifier
		value = modeu5_us05_diplomatic_cost_modifier
	}
	set_variable = {
		name = cbp_us05_diplomatic_target_base
		value = modeu5_us05_diplomatic_target_base
	}
	set_variable = {
		name = cbp_us05_vanilla_economic_base
		value = modeu5_us05_vanilla_economic_base
	}
}
