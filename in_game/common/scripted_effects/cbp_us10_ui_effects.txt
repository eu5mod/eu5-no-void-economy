# ModeU5 US-10-UI market table read-model helpers.
#
# These helpers feed the player-facing ModeU5 stock tab and deterministic
# debug probes. They read canonical country x market x good records and expose
# a generated per-good UI row for every ModeU5 stock good. Rows with no selected
# market production stay hidden in the GUI.

cbp_us10_ui_clear_selected_market_work_list = {

		limit = { has_global_variable_list = gui_cbp_us10_ui_selected_market }
		clear_global_variable_list = gui_cbp_us10_ui_selected_market
	}
}

cbp_us10_ui_rebuild_country_market_selector = {
	save_temporary_scope_as = cbp_us10_ui_country
	cbp_us10_ui_clear_market_selector_work_list = yes

	set_variable = { name = gui_cbp_us10_ui_market_count value = 0 }
	set_variable = { name = gui_cbp_us10_ui_market_iterator_index value = 0 }

	every_market_present_in_country = {
		save_temporary_scope_as = cbp_us10_ui_candidate_market
		scope:cbp_us10_ui_country = {
			set_variable = {
				name = gui_cbp_us10_ui_market_count
				value = { value = var:gui_cbp_us10_ui_market_count add = 1 }
			}
		}
		if = {
			limit = { NOT = { has_global_variable_list = gui_cbp_us10_ui_market_selector_markets } }
			add_to_global_variable_list = { name = gui_cbp_us10_ui_market_selector_markets target = scope:cbp_us10_ui_candidate_market }
		}
		else_if = {
			limit = {
				NOT = {
					is_target_in_global_variable_list = { name = gui_cbp_us10_ui_market_selector_markets target = scope:cbp_us10_ui_candidate_market }
				}
			}
			add_to_global_variable_list = { name = gui_cbp_us10_ui_market_selector_markets target = scope:cbp_us10_ui_candidate_market }
		}
	}
}

cbp_us10_ui_set_selected_market = {
	$market$ = { save_temporary_scope_as = cbp_us10_ui_candidate_market }
	cbp_us10_ui_clear_selected_market_work_list = yes
	add_to_global_variable_list = { name = gui_cbp_us10_ui_selected_market target = scope:cbp_us10_ui_candidate_market }
}

cbp_us10_ui_select_first_market_from_selector = {
	save_temporary_scope_as = cbp_us10_ui_country
	cbp_us10_ui_clear_selected_market_work_list = yes
	set_variable = { name = gui_cbp_us10_ui_selected_market_found value = 0 }
	set_variable = { name = gui_cbp_us10_ui_market_iterator_index value = 0 }
	set_variable = { name = gui_cbp_us10_ui_selected_market_index value = 0 }

	if = {
		limit = { has_global_variable_list = gui_cbp_us10_ui_market_selector_markets }
		every_in_global_list = {
			variable = gui_cbp_us10_ui_market_selector_markets
			save_temporary_scope_as = cbp_us10_ui_candidate_market
			scope:cbp_us10_ui_country = {
				set_variable = { name = gui_cbp_us10_ui_market_iterator_index value = { value = var:gui_cbp_us10_ui_market_iterator_index add = 1 } }
			}
			if = {
				limit = { scope:cbp_us10_ui_country = { var:gui_cbp_us10_ui_selected_market_found = 0 } }
				add_to_global_variable_list = { name = gui_cbp_us10_ui_selected_market target = scope:cbp_us10_ui_candidate_market }
				scope:cbp_us10_ui_country = {
					set_variable = { name = gui_cbp_us10_ui_selected_market_found value = 1 }
					set_variable = { name = gui_cbp_us10_ui_selected_market_index value = var:gui_cbp_us10_ui_market_iterator_index }
				}
			}
		}
	}
}

cbp_us10_ui_validate_or_select_market = {
	save_temporary_scope_as = cbp_us10_ui_country
	cbp_us10_ui_rebuild_country_market_selector = yes
	set_variable = { name = gui_cbp_us10_ui_selected_market_valid value = 0 }
	set_variable = { name = gui_cbp_us10_ui_market_iterator_index value = 0 }
	set_variable = { name = gui_cbp_us10_ui_selected_market_index value = 0 }

	if = {
		limit = { has_global_variable_list = gui_cbp_us10_ui_market_selector_markets has_global_variable_list = gui_cbp_us10_ui_selected_market }
		every_in_global_list = {
			variable = gui_cbp_us10_ui_market_selector_markets
			save_temporary_scope_as = cbp_us10_ui_candidate_market
			scope:cbp_us10_ui_country = {
				set_variable = { name = gui_cbp_us10_ui_market_iterator_index value = { value = var:gui_cbp_us10_ui_market_iterator_index add = 1 } }
			}
			if = {
				limit = { is_target_in_global_variable_list = { name = gui_cbp_us10_ui_selected_market target = scope:cbp_us10_ui_candidate_market } }
				scope:cbp_us10_ui_country = {
					set_variable = { name = gui_cbp_us10_ui_selected_market_valid value = 1 }
					set_variable = { name = gui_cbp_us10_ui_selected_market_index value = var:gui_cbp_us10_ui_market_iterator_index }
				}
			}
		}
	}

	if = {
		limit = { var:gui_cbp_us10_ui_selected_market_valid = 0 }
		cbp_us10_ui_select_first_market_from_selector = yes
	}
}

cbp_us10_ui_select_next_market = {
	save_temporary_scope_as = cbp_us10_ui_country
	cbp_us10_ui_rebuild_country_market_selector = yes
	cbp_us10_ui_clear_next_selected_market_work_list = yes

	if = {
		limit = { OR = { NOT = { has_global_variable_list = gui_cbp_us10_ui_selected_market } NOT = { has_global_variable_list = gui_cbp_us10_ui_market_selector_markets } } }
		cbp_us10_ui_select_first_market_from_selector = yes
	}
	else = {
		set_variable = { name = gui_cbp_us10_ui_seen_current_market value = 0 }
		set_variable = { name = gui_cbp_us10_ui_selected_market_found value = 0 }
		set_variable = { name = gui_cbp_us10_ui_market_iterator_index value = 0 }

		every_in_global_list = {
			variable = gui_cbp_us10_ui_market_selector_markets
			save_temporary_scope_as = cbp_us10_ui_candidate_market
			scope:cbp_us10_ui_country = {
				set_variable = { name = gui_cbp_us10_ui_market_iterator_index value = { value = var:gui_cbp_us10_ui_market_iterator_index add = 1 } }
			}
			if = {
				limit = { scope:cbp_us10_ui_country = { var:gui_cbp_us10_ui_seen_current_market = 1 var:gui_cbp_us10_ui_selected_market_found = 0 } }
				add_to_global_variable_list = { name = gui_cbp_us10_ui_next_selected_market target = scope:cbp_us10_ui_candidate_market }
				scope:cbp_us10_ui_country = {
					set_variable = { name = gui_cbp_us10_ui_selected_market_found value = 1 }
					set_variable = { name = gui_cbp_us10_ui_selected_market_index value = var:gui_cbp_us10_ui_market_iterator_index }
				}
			}
			if = {
				limit = { is_target_in_global_variable_list = { name = gui_cbp_us10_ui_selected_market target = scope:cbp_us10_ui_candidate_market } }
				scope:cbp_us10_ui_country = { set_variable = { name = gui_cbp_us10_ui_seen_current_market value = 1 } }
			}
		}

		if = {
			limit = { var:gui_cbp_us10_ui_selected_market_found = 0 }
			cbp_us10_ui_select_first_market_from_selector = yes
		}
		else = {
			cbp_us10_ui_clear_selected_market_work_list = yes
			every_in_global_list = {
				variable = gui_cbp_us10_ui_next_selected_market
				save_temporary_scope_as = cbp_us10_ui_candidate_market
				add_to_global_variable_list = { name = gui_cbp_us10_ui_selected_market target = scope:cbp_us10_ui_candidate_market }
			}
		}
	}
}

cbp_us10_ui_compute_selected_market_capacity = {
	save_temporary_scope_value_as = { name = gui_cbp_us10_ui_selected_market_capacity_value value = 0 }
	scope:cbp_us10_ui_selected_market_scope = { save_temporary_scope_as = cbp_market_country_cache_market }
	cbp_rebuild_countries_present_in_market = yes

	if = {
		limit = { has_global_variable_list = cbp_countries_present_in_market }
		every_in_global_list = {
			variable = cbp_countries_present_in_market
			if = {
				limit = { has_variable_map = cbp_stock_cap_by_market is_key_in_variable_map = { name = cbp_stock_cap_by_market target = scope:cbp_us10_ui_selected_market_scope } }
				save_temporary_scope_value_as = { name = gui_cbp_us10_ui_country_market_capacity value = { value = "variable_map(cbp_stock_cap_by_market|scope:cbp_us10_ui_selected_market_scope)" min = 0 } }
				save_temporary_scope_value_as = {
					name = gui_cbp_us10_ui_selected_market_capacity_value
					value = {
						value = scope:gui_cbp_us10_ui_selected_market_capacity_value
						add = scope:gui_cbp_us10_ui_country_market_capacity
					}
				}
			}
		}
	}

	scope:cbp_us10_ui_country = {
		set_variable = { name = gui_cbp_us10_ui_selected_market_capacity value = scope:gui_cbp_us10_ui_selected_market_capacity_value }
	}
}

cbp_us10_ui_capture_good_row = {
	cbp_read_country_stock_record = {
		country = scope:cbp_us10_ui_country
		market = scope:cbp_us10_ui_selected_market_scope
		good = $good$
	}
	cbp_read_void_economy_record = {
		country = scope:cbp_us10_ui_country
		market = scope:cbp_us10_ui_selected_market_scope
		good = $good$
	}
	scope:cbp_us10_ui_country = {
		set_variable = { name = gui_cbp_us10_ui_$key$_country_stock value = scope:cbp_read_country_stock }
		set_variable = { name = gui_cbp_us10_ui_$key$_country_capacity value = scope:cbp_read_country_stock_cap }
		set_variable = { name = gui_cbp_us10_ui_$key$_market_stock value = scope:cbp_read_market_stock }
		set_variable = { name = gui_cbp_us10_ui_$key$_market_capacity value = scope:gui_cbp_us10_ui_selected_market_capacity_value }
		set_variable = { name = gui_cbp_us10_ui_$key$_produced value = scope:cbp_us00_produced }
		set_variable = { name = gui_cbp_us10_ui_$key$_added value = scope:cbp_us00_added }
		set_variable = { name = gui_cbp_us10_ui_$key$_rejected value = scope:cbp_us00_rejected }
		set_variable = { name = gui_cbp_us10_ui_$key$_country_over_cap value = scope:cbp_read_over_capacity }
		save_temporary_scope_value_as = { name = gui_cbp_us10_ui_$key$_market_over_cap_value value = { value = scope:cbp_read_market_stock subtract = scope:gui_cbp_us10_ui_selected_market_capacity_value min = 0 } }
		set_variable = { name = gui_cbp_us10_ui_$key$_market_over_cap value = scope:gui_cbp_us10_ui_$key$_market_over_cap_value }
		save_temporary_scope_value_as = { name = gui_cbp_us10_ui_$key$_overproduction_percent_value value = { value = scope:cbp_us00_read_overproduction_ratio multiply = 100 min = 0 } }
		set_variable = { name = gui_cbp_us10_ui_$key$_overproduction_percent value = scope:gui_cbp_us10_ui_$key$_overproduction_percent_value }
		save_temporary_scope_value_as = { name = gui_cbp_us10_ui_$key$_production_penalty_percent_value value = { value = scope:cbp_us00_read_production_penalty multiply = 100 } }
		set_variable = { name = gui_cbp_us10_ui_$key$_production_penalty_percent value = scope:gui_cbp_us10_ui_$key$_production_penalty_percent_value }
		if = {
			limit = { scope:cbp_us00_read_overproduction_ratio > 0.05 }
			set_variable = { name = gui_cbp_us10_ui_$key$_overproduction_warning value = 1 }
		}
		else = { set_variable = { name = gui_cbp_us10_ui_$key$_overproduction_warning value = 0 } }
	}
}
