# ModeU5 Q8.6 market-sliced verifier.
#
# This is a bounded debug/audit verifier slice. It collects candidate markets
# from already-bounded scheduling surfaces and verifies only that slice by
# rebuilding the current-market country work cache.
#
# Guardrails:
# - no stock mutation;
# - no stock repair;
# - no global market pass;
# - no durable per-market country-list storage;
# - candidate lists are scheduling/debug state only.

modeu5_clear_market_sliced_verifier_state = {
	remove_global_variable = cbp_market_sliced_verifier_candidates_built
	remove_global_variable = cbp_market_sliced_verifier_candidates_checked
	remove_global_variable = cbp_market_sliced_verifier_candidates_passed
	remove_global_variable = cbp_market_sliced_verifier_candidates_failed
	remove_global_variable = cbp_market_sliced_verifier_last_run_skipped
	remove_global_variable = cbp_market_sliced_verifier_last_run_empty
	remove_global_variable = cbp_market_sliced_verifier_last_run_failed
	remove_global_variable = cbp_market_sliced_verifier_last_run_passed

	set_global_variable = { name = cbp_market_sliced_verifier_candidates_built value = 0 }
	set_global_variable = { name = cbp_market_sliced_verifier_candidates_checked value = 0 }
	set_global_variable = { name = cbp_market_sliced_verifier_candidates_passed value = 0 }
	set_global_variable = { name = cbp_market_sliced_verifier_candidates_failed value = 0 }

	if = {
		limit = { has_global_variable_list = cbp_market_sliced_verifier_candidate_markets }
		clear_global_variable_list = cbp_market_sliced_verifier_candidate_markets
	}
}

modeu5_add_market_to_market_sliced_verifier_candidates = {
	if = {
		limit = {
			exists = scope:modeu5_market_sliced_verifier_candidate_market
			NOT = { has_global_variable_list = cbp_market_sliced_verifier_candidate_markets }
		}
		add_to_global_variable_list = {
			name = cbp_market_sliced_verifier_candidate_markets
			target = scope:modeu5_market_sliced_verifier_candidate_market
		}
		set_global_variable = {
			name = cbp_market_sliced_verifier_candidates_built
			value = {
				value = global_var:cbp_market_sliced_verifier_candidates_built
				add = 1
			}
		}
	}
	else_if = {
		limit = {
			exists = scope:modeu5_market_sliced_verifier_candidate_market
			NOT = {
				is_target_in_global_variable_list = {
					name = cbp_market_sliced_verifier_candidate_markets
					target = scope:modeu5_market_sliced_verifier_candidate_market
				}
			}
		}
		add_to_global_variable_list = {
			name = cbp_market_sliced_verifier_candidate_markets
			target = scope:modeu5_market_sliced_verifier_candidate_market
		}
		set_global_variable = {
			name = cbp_market_sliced_verifier_candidates_built
			value = {
				value = global_var:cbp_market_sliced_verifier_candidates_built
				add = 1
			}
		}
	}
}

modeu5_add_dirty_markets_to_market_sliced_verifier_candidates = {
	if = {
		limit = { has_global_variable_list = cbp_market_country_cache_dirty_markets }
		every_in_global_list = {
			variable = cbp_market_country_cache_dirty_markets
			save_temporary_scope_as = modeu5_market_sliced_verifier_candidate_market
			modeu5_add_market_to_market_sliced_verifier_candidates = yes
		}
	}
}

modeu5_add_promoted_markets_to_market_sliced_verifier_candidates = {
	if = {
		limit = { has_global_variable_list = cbp_promoted_markets_this_cycle }
		every_in_global_list = {
			variable = cbp_promoted_markets_this_cycle
			save_temporary_scope_as = modeu5_market_sliced_verifier_candidate_market
			modeu5_add_market_to_market_sliced_verifier_candidates = yes
		}
	}
}

modeu5_prepare_market_sliced_verifier_candidates = {
	modeu5_clear_market_sliced_verifier_state = yes
	modeu5_add_dirty_markets_to_market_sliced_verifier_candidates = yes
	modeu5_add_promoted_markets_to_market_sliced_verifier_candidates = yes
}

modeu5_verify_market_sliced_verifier_candidate = {
	set_global_variable = {
		name = cbp_market_sliced_verifier_candidates_checked
		value = {
			value = global_var:cbp_market_sliced_verifier_candidates_checked
			add = 1
		}
	}

	save_temporary_scope_as = modeu5_market_country_cache_market
	modeu5_rebuild_countries_present_in_market = yes

	if = {
		limit = { has_global_variable = cbp_market_country_cache_last_rebuild_failed }
		set_global_variable = {
			name = cbp_market_sliced_verifier_candidates_failed
			value = {
				value = global_var:cbp_market_sliced_verifier_candidates_failed
				add = 1
			}
		}
	}
	else = {
		set_global_variable = {
			name = cbp_market_sliced_verifier_candidates_passed
			value = {
				value = global_var:cbp_market_sliced_verifier_candidates_passed
				add = 1
			}
		}
	}
}

modeu5_run_market_sliced_verifier_candidates = {
	if = {
		limit = { modeu5_market_sliced_verifier_allowed_trigger = yes }
		modeu5_prepare_market_sliced_verifier_candidates = yes

		if = {
			limit = { has_global_variable_list = cbp_market_sliced_verifier_candidate_markets }
			every_in_global_list = {
				variable = cbp_market_sliced_verifier_candidate_markets
				modeu5_verify_market_sliced_verifier_candidate = yes
			}
			if = {
				limit = { global_var:cbp_market_sliced_verifier_candidates_failed > 0 }
				set_global_variable = cbp_market_sliced_verifier_last_run_failed
				error_log = "ModeU5 Q8.6 market-sliced verifier failed for at least one candidate market."
			}
			else = {
				set_global_variable = cbp_market_sliced_verifier_last_run_passed
				debug_log = "ModeU5 Q8.6 market-sliced verifier PASS candidates=[GetGlobalVariable('modeu5_market_sliced_verifier_candidates_checked').GetValue|0]"
			}
		}
		else = {
			set_global_variable = cbp_market_sliced_verifier_last_run_empty
			debug_log = "ModeU5 Q8.6 market-sliced verifier SKIP no_candidate_markets"
		}
	}
	else = {
		set_global_variable = cbp_market_sliced_verifier_last_run_skipped
	}
}
