# PR126 promoted-market orchestration helper surface.
#
# The early PR126 helpers below name the B/C/D entry points that stacked layers
# call from the promoted-market shell. PR7 also adds the live promoted dispatcher
# at the bottom of this file. Each helper delegates to the current domain owner:
# capacity, market-country cache, generated good adapters, US-00, or US-10.

# B: prepare the current promoted market's country work cache. The cache remains
# a rebuilt work list, never durable per-market storage and never a stock source.
modeu5_prepare_promoted_market_country_cache = {
	$market$ = {
		save_temporary_scope_as = modeu5_market
		save_temporary_scope_as = modeu5_market_country_cache_market
	}

	modeu5_rebuild_countries_present_in_market = yes
}

# B: refresh one selected country-market capacity record from the existing
# country location pool and the selected market's trade-capacity contribution.
# This is the promoted-market-compatible capacity entry point: it avoids the
# full owned-location pool rebuild and expects lifecycle hooks/startup to own
# that persistent country pool.
modeu5_prepare_promoted_country_market_capacity = {
	$country$ = { save_temporary_scope_as = modeu5_country }
	$market$ = { save_temporary_scope_as = modeu5_market }

	scope:modeu5_country = {
		modeu5_calculate_country_storage_capacity_pool = yes
	}
	modeu5_recalculate_country_market_capacity_from_prepared_pool_shared = yes
}

# B: refresh capacities for every country currently present in one promoted
# market. Later dispatcher PRs can call this after market promotion; current
# monthly order is not rewired in this PR.
modeu5_prepare_promoted_market_capacity_cache = {
	$market$ = { save_temporary_scope_as = modeu5_market }

	modeu5_prepare_promoted_market_country_cache = {
		market = scope:modeu5_market
	}

	if = {
		limit = { has_global_variable_list = cbp_countries_present_in_market }
		every_in_global_list = {
			variable = cbp_countries_present_in_market
			save_temporary_scope_as = modeu5_country
			modeu5_prepare_promoted_country_market_capacity = {
				country = scope:modeu5_country
				market = scope:modeu5_market
			}
		}
	}
}

# C: generated-good adapter bridge for scoped US-00 market-good work. This is a
# naming layer over the existing generated adapter; it is not called by the

# PR4: promoted-market dispatcher shell.
#
# This shell is intentionally not wired into `modeu5_run_monthly_stock_cycle`
# yet. It proves the outer work-list shape that future PR126 layers will use:
# current country -> every_market_present_in_country -> promoted market list.
# Normal mode promotes all current-country markets. Performance mode promotes
# only markets already present in `modeu5_performance_relevant_markets`.

modeu5_clear_promoted_market_cycle_work_list = {
	remove_global_variable = cbp_promoted_market_candidate_count
	remove_global_variable = cbp_promoted_market_count
	remove_global_variable = cbp_promoted_market_rejected_count
	remove_global_variable = cbp_promoted_market_shell_iterations

	set_global_variable = { name = cbp_promoted_market_candidate_count value = 0 }
	set_global_variable = { name = cbp_promoted_market_count value = 0 }
	set_global_variable = { name = cbp_promoted_market_rejected_count value = 0 }
	set_global_variable = { name = cbp_promoted_market_shell_iterations value = 0 }

	if = {
		limit = { has_global_variable_list = cbp_promoted_markets_this_cycle }
		clear_global_variable_list = cbp_promoted_markets_this_cycle
	}
}

modeu5_note_promoted_market_cycle_candidate = {
	set_global_variable = {
		name = cbp_promoted_market_candidate_count
		value = {
			value = global_var:cbp_promoted_market_candidate_count
			add = 1
		}
	}
}

modeu5_note_promoted_market_cycle_rejected = {
	set_global_variable = {
		name = cbp_promoted_market_rejected_count
		value = {
			value = global_var:cbp_promoted_market_rejected_count
			add = 1
		}
	}
}

modeu5_add_current_market_to_promoted_market_cycle_work_list = {
	if = {
		limit = {
			NOT = { has_global_variable_list = cbp_promoted_markets_this_cycle }
		}
		add_to_global_variable_list = {
			name = cbp_promoted_markets_this_cycle
			target = scope:modeu5_promoted_market_candidate
		}
		set_global_variable = {
			name = cbp_promoted_market_count
			value = {
				value = global_var:cbp_promoted_market_count
				add = 1
			}
		}
	}
	else_if = {
		limit = {
			NOT = {
				is_target_in_global_variable_list = {
					name = cbp_promoted_markets_this_cycle
					target = scope:modeu5_promoted_market_candidate
				}
			}
		}
		add_to_global_variable_list = {
			name = cbp_promoted_markets_this_cycle
			target = scope:modeu5_promoted_market_candidate
		}
		set_global_variable = {
			name = cbp_promoted_market_count
			value = {
				value = global_var:cbp_promoted_market_count
				add = 1
			}
		}
	}
}

modeu5_prepare_promoted_market_work_list_for_current_country = {
	save_temporary_scope_as = modeu5_promoted_market_cycle_country
	modeu5_clear_promoted_market_cycle_work_list = yes

	every_market_present_in_country = {
		save_temporary_scope_as = modeu5_promoted_market_candidate
		modeu5_note_promoted_market_cycle_candidate = yes

		if = {
			limit = { modeu5_nve_normal_mode_trigger = yes }
			modeu5_add_current_market_to_promoted_market_cycle_work_list = yes
		}
		else_if = {
			limit = {
				modeu5_performance_mode_enabled_trigger = yes
				has_global_variable_list = cbp_performance_relevant_markets
				is_target_in_global_variable_list = {
					name = cbp_performance_relevant_markets
					target = scope:modeu5_promoted_market_candidate
				}
			}
			modeu5_add_current_market_to_promoted_market_cycle_work_list = yes
		}
		else = {
			modeu5_note_promoted_market_cycle_rejected = yes
		}
	}
}

modeu5_run_monthly_promoted_market_cycle = {
	# Test-only shell in PR4. Future PRs wire B/C/D logic under this iteration.
	save_temporary_scope_as = modeu5_promoted_market_cycle_country
	modeu5_prepare_promoted_market_work_list_for_current_country = yes

	if = {
		limit = { has_global_variable_list = cbp_promoted_markets_this_cycle }
		every_in_global_list = {
			variable = cbp_promoted_markets_this_cycle
			save_temporary_scope_as = modeu5_promoted_market
			set_global_variable = {
				name = cbp_promoted_market_shell_iterations
				value = {
					value = global_var:cbp_promoted_market_shell_iterations
					add = 1
				}
			}
		}
	}
}

modeu5_run_promoted_market_local_branch_market_good = {
	$country$ = {
		save_temporary_scope_as = modeu5_local_branch_country
		save_temporary_scope_as = modeu5_country
	}
	$market$ = {
		save_temporary_scope_as = modeu5_market
		save_temporary_scope_as = modeu5_market_country_cache_market
	}
	goods:$good$ = { save_temporary_scope_as = modeu5_good }

	modeu5_prepare_promoted_market_country_cache = {
		market = scope:modeu5_market
	}
	modeu5_note_promoted_market_local_branch_country_cache_rebuild = yes

	if = {
		limit = { has_global_variable_list = cbp_countries_present_in_market }
		every_in_global_list = {
			variable = cbp_countries_present_in_market
			save_temporary_scope_as = modeu5_promoted_market_capacity_country
			modeu5_prepare_promoted_country_market_capacity = {
				country = scope:modeu5_promoted_market_capacity_country
				market = scope:modeu5_market
			}
			modeu5_note_promoted_market_local_branch_capacity_country = yes
		}
	}

	scope:modeu5_local_branch_country = {
		save_temporary_scope_as = modeu5_country
		save_temporary_scope_as = modeu5_reconciliation_controller
	}

	modeu5_note_promoted_market_local_branch_us00 = yes
	modeu5_run_scoped_us00_market_good = {
		country = scope:modeu5_local_branch_country
		market = scope:modeu5_market
		good = $good$
	}

	save_temporary_scope_value_as = { name = cbp_satisfied_quantity value = 0 }
	save_temporary_scope_value_as = { name = cbp_unsatisfied_quantity value = 0 }
	modeu5_note_promoted_market_local_branch_us10 = yes
	modeu5_run_scoped_us10_monthly_market_good = {
		country = scope:modeu5_local_branch_country
		market = scope:modeu5_market
		good = $good$
	}
	set_global_variable = {
		name = cbp_promoted_market_local_branch_us10_satisfied_quantity
		value = {
			value = global_var:cbp_promoted_market_local_branch_us10_satisfied_quantity
			add = scope:cbp_satisfied_quantity
		}
	}

	scope:modeu5_market = {
		save_temporary_scope_as = modeu5_market_country_cache_market
	}
	modeu5_note_promoted_market_local_branch_validation = yes
	modeu5_validate_scoped_market_good_from_prepared_country_cache = {
		country = scope:modeu5_local_branch_country
		market = scope:modeu5_market
		good = $good$
	}
	if = {
		limit = { scope:cbp_validation_failed_after_rebuild > 0 }
		set_global_variable = {
			name = cbp_promoted_market_local_branch_validation_failures
			value = {
				value = global_var:cbp_promoted_market_local_branch_validation_failures
				add = 1
			}
		}
	}

	set_global_variable = {
		name = cbp_promoted_market_local_branch_markets_processed
		value = {
			value = global_var:cbp_promoted_market_local_branch_markets_processed
			add = 1
		}
	}
}

modeu5_prepare_monthly_promoted_market_live_dispatcher_metrics = {
	save_temporary_scope_value_as = {
		name = cbp_promoted_market_live_dispatcher_stamp
		value = {
			value = current_year
			multiply = 12
			add = current_month
		}
	}

	if = {
		limit = {
			OR = {
				NOT = { has_global_variable = cbp_promoted_market_live_dispatcher_stamp }
				global_var:cbp_promoted_market_live_dispatcher_stamp != scope:cbp_promoted_market_live_dispatcher_stamp
			}
		}
		set_global_variable = {
			name = cbp_promoted_market_live_dispatcher_stamp
			value = scope:cbp_promoted_market_live_dispatcher_stamp
		}
		modeu5_reset_promoted_market_live_dispatcher_metrics = yes
	}
	else_if = {
		limit = { NOT = { has_global_variable = cbp_promoted_market_live_dispatcher_runs } }
		modeu5_reset_promoted_market_live_dispatcher_metrics = yes
	}
}

modeu5_run_promoted_market_live_local_branch_market_all_goods = {
	$country$ = {
		save_temporary_scope_as = modeu5_local_branch_country
		save_temporary_scope_as = modeu5_reconciliation_controller
	}
	$market$ = {
		save_temporary_scope_as = modeu5_market
		save_temporary_scope_as = modeu5_market_country_cache_market
	}

	modeu5_pr71_prepare_active_good_metrics = yes

	modeu5_prepare_promoted_market_country_cache = {
		market = scope:modeu5_market
	}
	modeu5_note_promoted_market_live_country_cache_rebuild = yes

	if = {
		limit = { has_global_variable_list = cbp_countries_present_in_market }

		# Q4.1 loop merge: capacity refresh is fused with the US-00 country pass.
		# Keep US-10 in a second pass so every present country's US-00 facts are
		# updated before any same-market consumption scans the market stock.
		every_in_global_list = {
			variable = cbp_countries_present_in_market
			save_temporary_scope_as = modeu5_country
			save_temporary_scope_as = modeu5_promoted_market_capacity_country
			modeu5_prepare_promoted_country_market_capacity = {
				country = scope:modeu5_promoted_market_capacity_country
				market = scope:modeu5_market
			}
			modeu5_note_promoted_market_live_capacity_country = yes
			modeu5_note_promoted_market_live_us00_country_pass = yes
			modeu5_pr71_process_us00_monthly_market_active_goods = yes
		}
		every_in_global_list = {
			variable = cbp_countries_present_in_market
			save_temporary_scope_as = modeu5_country
			modeu5_note_promoted_market_live_us10_country_pass = yes
			modeu5_pr71_process_us10_monthly_market_pending_goods = yes
		}
	}

	modeu5_note_promoted_market_live_local_market_processed = yes
}

modeu5_run_monthly_promoted_market_local_cycle = {
	save_temporary_scope_as = modeu5_promoted_market_cycle_country
	modeu5_prepare_monthly_promoted_market_live_dispatcher_metrics = yes
	modeu5_note_promoted_market_live_dispatcher_run = yes

	every_market_center_in_country = {
		save_temporary_scope_as = modeu5_market
		save_temporary_scope_as = modeu5_market_country_cache_market
		modeu5_mark_monthly_market_seen = yes
		modeu5_prepare_market_runtime_accounting_mode = {
			market = scope:modeu5_market
		}
		if = {
			limit = { modeu5_market_runtime_use_detailed_accounting_trigger = yes }
			modeu5_note_promoted_market_live_detailed_market = yes
			modeu5_run_promoted_market_live_local_branch_market_all_goods = {
				country = scope:modeu5_promoted_market_cycle_country
				market = scope:modeu5_market
			}
		}
		else_if = {
			limit = { modeu5_market_runtime_use_vanilla_fallback_trigger = yes }
			modeu5_note_promoted_market_live_fallback_market = yes
			modeu5_note_us00_vanilla_fallback_market = yes
			modeu5_note_us10_vanilla_fallback_market = yes
		}
		else_if = {
			limit = { modeu5_market_runtime_blocked_trigger = yes }
			modeu5_note_promoted_market_live_blocked_market = yes
			modeu5_note_runtime_blocked_market = yes
		}
	}
}
