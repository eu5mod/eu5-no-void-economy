modeu5_clear_promoted_market_cycle_work_list = {
	remove_global_variable = modeu5_promoted_market_candidate_count
	remove_global_variable = modeu5_promoted_market_count
	remove_global_variable = modeu5_promoted_market_rejected_count
	remove_global_variable = modeu5_promoted_market_shell_iterations
	set_global_variable = { name = modeu5_promoted_market_candidate_count value = 0 }
	set_global_variable = { name = modeu5_promoted_market_count value = 0 }
	set_global_variable = { name = modeu5_promoted_market_rejected_count value = 0 }
	set_global_variable = { name = modeu5_promoted_market_shell_iterations value = 0 }
	if = { limit = { has_global_variable_list = modeu5_promoted_markets_this_cycle } clear_global_variable_list = modeu5_promoted_markets_this_cycle }
}

modeu5_add_current_market_to_promoted_market_cycle_work_list = {
	if = { limit = { NOT = { has_global_variable_list = modeu5_promoted_markets_this_cycle } } add_to_global_variable_list = { name = modeu5_promoted_markets_this_cycle target = scope:modeu5_promoted_market_candidate } set_global_variable = { name = modeu5_promoted_market_count value = { value = global_var:modeu5_promoted_market_count add = 1 } } }
	else_if = { limit = { NOT = { is_target_in_global_variable_list = { name = modeu5_promoted_markets_this_cycle target = scope:modeu5_promoted_market_candidate } } } add_to_global_variable_list = { name = modeu5_promoted_markets_this_cycle target = scope:modeu5_promoted_market_candidate } set_global_variable = { name = modeu5_promoted_market_count value = { value = global_var:modeu5_promoted_market_count add = 1 } } }
}

modeu5_prepare_promoted_market_work_list_for_current_country = {
	save_temporary_scope_as = modeu5_promoted_market_cycle_country
	modeu5_clear_promoted_market_cycle_work_list = yes
	every_market_present_in_country = {
		save_temporary_scope_as = modeu5_promoted_market_candidate
		modeu5_note_promoted_market_cycle_candidate = yes
		if = { limit = { modeu5_nve_normal_mode_trigger = yes } modeu5_add_current_market_to_promoted_market_cycle_work_list = yes }
		else_if = { limit = { modeu5_performance_mode_enabled_trigger = yes has_global_variable_list = modeu5_performance_relevant_markets is_target_in_global_variable_list = { name = modeu5_performance_relevant_markets target = scope:modeu5_promoted_market_candidate } } modeu5_add_current_market_to_promoted_market_cycle_work_list = yes }
		else = { modeu5_note_promoted_market_cycle_rejected = yes }
	}
}

modeu5_run_monthly_promoted_market_cycle = {
	save_temporary_scope_as = modeu5_promoted_market_cycle_country

	$country$ = { save_temporary_scope_as = modeu5_local_branch_country save_temporary_scope_as = modeu5_country }
	$market$ = { save_temporary_scope_as = modeu5_market save_temporary_scope_as = modeu5_market_country_cache_market }

	modeu5_prepare_promoted_market_country_cache = { market = scope:modeu5_market }
	modeu5_note_promoted_market_local_branch_country_cache_rebuild = yes
	if = { limit = { has_global_variable_list = modeu5_countries_present_in_market } every_in_global_list = { variable = modeu5_countries_present_in_market save_temporary_scope_as = modeu5_promoted_market_capacity_country modeu5_prepare_promoted_country_market_capacity = { country = scope:modeu5_promoted_market_capacity_country market = scope:modeu5_market } modeu5_note_promoted_market_local_branch_capacity_country = yes } }
	scope:modeu5_local_branch_country = { save_temporary_scope_as = modeu5_country save_temporary_scope_as = modeu5_reconciliation_controller }
	modeu5_note_promoted_market_local_branch_us00 = yes
	modeu5_run_scoped_us00_market_good = { country = scope:modeu5_local_branch_country market = scope:modeu5_market good = $good$ }
	save_temporary_scope_value_as = { name = modeu5_satisfied_quantity value = 0 }
	save_temporary_scope_value_as = { name = modeu5_unsatisfied_quantity value = 0 }
	modeu5_note_promoted_market_local_branch_us10 = yes
	modeu5_run_scoped_us10_monthly_market_good = { country = scope:modeu5_local_branch_country market = scope:modeu5_market good = $good$ }
	set_global_variable = { name = modeu5_promoted_market_local_branch_us10_satisfied_quantity value = { value = global_var:modeu5_promoted_market_local_branch_us10_satisfied_quantity add = scope:modeu5_satisfied_quantity } }
	scope:modeu5_market = { save_temporary_scope_as = modeu5_market_country_cache_market }
	modeu5_note_promoted_market_local_branch_validation = yes
	modeu5_validate_scoped_market_good_from_prepared_country_cache = { country = scope:modeu5_local_branch_country market = scope:modeu5_market good = $good$ }
	if = { limit = { scope:modeu5_validation_failed_after_rebuild > 0 } set_global_variable = { name = modeu5_promoted_market_local_branch_validation_failures value = { value = global_var:modeu5_promoted_market_local_branch_validation_failures add = 1 } } }

}
