# PR126 PR6 country trade-owner pass.
#
# This file intentionally does not switch the monthly dispatcher yet. It adds a
# country-scoped trade pass surface that can be tested before PR7 wires it into
# the live monthly order.
#
# TECH-01 047 confirms every_trade as a country-scope iterator. TECH-01 056/138
# are treated as runtime-confirmed for the actual moved-goods quantity formula,
# but the implementation must not read `traded_goods:transport_cost` directly
# from generic trade scope.
#
#   goods_quantity = trade_volume / traded_goods:transport_cost
#
# PR6.1 implements this through generated literal-good transport-cost helpers.
# The trade owner is the trade-scope `owner` scope link, not an implicit copy of
# the country that happens to run the iterator. The country pulse/controller still
# schedules the pass, but attribution must come from trade.owner.
#
# Vanilla trade always crosses distinct source and target markets. The situation
# `source_market == target_market` is impossible under vanilla mechanics and is
# deliberately not represented here. Same-market stock consumption belongs to the
# local non-trade demand branch, not to the country trade-owner pass.
#
# PR6 still does not mutate stock from native vanilla trade objects because the
# stock-affecting generated per-good dispatch is deliberately deferred; stock
# consequences continue to go through explicit ModeU5 requests, existing US-10
# inter-market handlers, and central stock operators.
#
# #107 extension: US-17/#105, #120, and US-20/#118 route reconciliation is wired
# into the existing every_trade body after route scopes and quantity are captured.
# This preserves the #161 Q8.7 boundary: no market-scope every_trade loop is
# introduced. The hook is dormant unless the CMM trade-rework setting enables
# modeu5_trade_rework_enabled_trigger.
#
# US-17 owner-modifier rule:
# - buying_efficiency, selling_efficiency, and merchant_maintenance_cost are read
#   from the saved trade owner country;
# - the scheduler country and market-center owner are never used as substitutes;
# - unresolved route prices/base-maintenance/income surfaces remain fail-closed.

modeu5_capture_country_trade_owner_trade_quantity = {
	save_temporary_scope_value_as = {
		name = cbp_trade_owner_trade_volume
		value = {
			value = trade_volume
			min = 0
		}
	}

	# Do not divide by `traded_goods:transport_cost` here. The safe current
	# implementation is the generated literal-good helper pattern:
	#
	#   modeu5_compute_goods_quantity_from_trade_capacity_good_<good>
	#
	# The generated dispatcher matches the saved traded-good scope to a literal
	# good and then uses the existing static transport-cost denominator with
	# the same 0.0001 clamp as the current implementation.
	modeu5_compute_trade_owner_goods_quantity_from_traded_good = yes

	if = {
		limit = {
			scope:test_cbp_trade_owner_quantity_matched > 0
			scope:test_cbp_trade_owner_goods_quantity > 0
		}
		modeu5_note_country_trade_owner_quantity_confirmed_trade = yes
	}
	else = {
		modeu5_note_country_trade_owner_quantity_blocked_trade = yes
	}
}

# Country-scoped native trade-owner pass. It classifies trades reached from the
# current country iterator, but attributes each trade through the trade-scope
# `owner` link. The current country is retained only as the controller/scheduler
# so tests can detect whether country-scoped every_trade ever yields a trade whose
# owner differs from the country that scheduled the pass.
modeu5_run_monthly_country_trade_owner_cycle = {
	save_temporary_scope_as = modeu5_trade_owner_controller_country
	modeu5_note_country_trade_owner_pass_run = yes
	if = {
		limit = { modeu5_trade_rework_enabled_trigger = yes }
		modeu5_prepare_trade_efficiency_reconciliation_runtime_metrics_once = yes
		modeu5_reset_trade_efficiency_reconciliation_accumulator = yes
	}

	every_trade = {
		save_temporary_scope_as = modeu5_trade_owner_trade
		modeu5_note_country_trade_owner_trade_seen = yes

		owner = { save_temporary_scope_as = modeu5_trade_owner_country }
		from_market = { save_temporary_scope_as = modeu5_trade_owner_source_market }
		to_market = { save_temporary_scope_as = modeu5_trade_owner_target_market }
		traded_goods = { save_temporary_scope_as = modeu5_trade_owner_good }
		modeu5_capture_country_trade_owner_trade_quantity = yes
		if = {
			limit = { exists = scope:modeu5_trade_owner_country }
			scope:modeu5_trade_owner_country = {
				if = {
					limit = { modeu5_trade_rework_enabled_trigger = yes }
					# Country modifier inputs are live and confirmed here. Route prices,
					# export cost, base maintenance and visible income/profit remain
					# explicit fail-closed inputs until their route surfaces are proven.
					modeu5_capture_trade_owner_country_modifier_inputs = yes
					save_temporary_scope_value_as = { name = cbp_trade_efficiency_formula_inputs_available value = 0 }
					modeu5_run_us17_us20_route_reconciliation_from_owner_modifiers = yes
				}
			}
		}

		if = {
			limit = {
				NOT = { scope:modeu5_trade_owner_country = scope:modeu5_trade_owner_controller_country }
			}
			modeu5_note_country_trade_owner_owner_mismatch = yes
		}

		modeu5_note_country_trade_owner_inter_market_trade = yes
	}
}

# Explicit-request handler for tests and future ModeU5-owned trade requests.
# PR6 only models inter-market trade transfer here. Same-market local consumption
# is intentionally absent because vanilla trade cannot have source_market equal
# to target_market.
modeu5_process_country_trade_owner_explicit_request = {
	$buyer_country$ = { save_temporary_scope_as = modeu5_country_trade_owner_buyer_country }
	$source_market$ = { save_temporary_scope_as = modeu5_country_trade_owner_source_market }
	$target_market$ = { save_temporary_scope_as = modeu5_country_trade_owner_target_market }
	goods:$good$ = { save_temporary_scope_as = modeu5_good }

	modeu5_note_country_trade_owner_explicit_request = { requested_quantity = $requested_quantity$ }
	modeu5_note_country_trade_owner_inter_market_trade = yes
	modeu5_handoff_scoped_inter_market_transfer = {
		buyer_country = scope:modeu5_country_trade_owner_buyer_country
		source_market = scope:modeu5_country_trade_owner_source_market
		target_market = scope:modeu5_country_trade_owner_target_market
		good = $good$
		requested_quantity = $requested_quantity$
	}
	modeu5_note_country_trade_owner_inter_market_transfer = yes
	modeu5_note_country_trade_owner_outcome_quantities = {
		transferred_quantity = scope:cbp_transferred_quantity
		unsatisfied_quantity = scope:cbp_unsatisfied_quantity
	}
}
