# Q8.7 live global market-owner dispatcher.
#
# This is the runtime migration layer after the #155/#158/#159/#160 proof stack.
# It moves the market-local owner from the market-center workaround to a
# once-per-month native every_market_in_world pass while preserving the existing
# mutating market-local branch internals.
#
# Boundaries:
# - US-00 and US-10 stay inside the existing generated-good live branch.
# - US-00 remains a full present-country pass before US-10 starts.
# - every_trade remains owned by the country trade-owner pass.
# - US-04 signed reconciliation runs after trade and before audit validation.
# - old market-center owner remains available through the fallback switch.

cbp_enable_q8_7_live_global_market_owner = {
	remove_global_variable = cbp_q8_7_live_global_market_owner_disabled
}

cbp_disable_q8_7_live_global_market_owner = {
	set_global_variable = cbp_q8_7_live_global_market_owner_disabled
}

cbp_reset_q8_7_live_global_owner_metrics = {
	remove_global_variable = cbp_q8_7_live_global_owner_runs
	remove_global_variable = cbp_q8_7_live_global_owner_skip_runs
	remove_global_variable = cbp_q8_7_live_global_owner_markets_seen
	remove_global_variable = cbp_q8_7_live_global_owner_markets_detailed
	remove_global_variable = cbp_q8_7_live_global_owner_markets_fallback
	remove_global_variable = cbp_q8_7_live_global_owner_markets_blocked
	remove_global_variable = cbp_q8_7_live_global_owner_us00_before_us10_guard
	remove_global_variable = cbp_q8_7_live_global_owner_no_market_scope_trade_guard

	set_global_variable = { name = cbp_q8_7_live_global_owner_runs value = 0 }
	set_global_variable = { name = cbp_q8_7_live_global_owner_skip_runs value = 0 }
	set_global_variable = { name = cbp_q8_7_live_global_owner_markets_seen value = 0 }
	set_global_variable = { name = cbp_q8_7_live_global_owner_markets_detailed value = 0 }
	set_global_variable = { name = cbp_q8_7_live_global_owner_markets_fallback value = 0 }
	set_global_variable = { name = cbp_q8_7_live_global_owner_markets_blocked value = 0 }
	set_global_variable = { name = cbp_q8_7_live_global_owner_us00_before_us10_guard value = 1 }
	set_global_variable = { name = cbp_q8_7_live_global_owner_no_market_scope_trade_guard value = 1 }
}

cbp_note_q8_7_live_global_owner_run = {
	set_global_variable = {
		name = cbp_q8_7_live_global_owner_runs
		value = {
			value = global_var:cbp_q8_7_live_global_owner_runs
			add = 1
		}
	}
}

}

cbp_q8_7_run_global_market_local_owner_market = {
	$country$ = {
		save_temporary_scope_as = cbp_q8_7_live_global_owner_controller
	}
	$market$ = {
		save_temporary_scope_as = cbp_market
		save_temporary_scope_as = cbp_market_country_cache_market
	}

	cbp_mark_monthly_market_seen = yes
	cbp_note_q8_7_live_global_owner_market_seen = yes

	cbp_prepare_market_runtime_accounting_mode = {
		market = scope:cbp_market
	}
	if = {
		limit = { cbp_market_runtime_use_detailed_accounting_trigger = yes }
		cbp_note_promoted_market_live_detailed_market = yes
		cbp_note_q8_7_live_global_owner_detailed_market = yes
		cbp_run_promoted_market_live_local_branch_market_all_goods = {
			country = scope:cbp_q8_7_live_global_owner_controller
			market = scope:cbp_market
		}
	}
	else_if = {
		limit = { cbp_market_runtime_use_vanilla_fallback_trigger = yes }
		cbp_note_promoted_market_live_fallback_market = yes
		cbp_note_q8_7_live_global_owner_fallback_market = yes
		cbp_note_us00_vanilla_fallback_market = yes
		cbp_note_us10_vanilla_fallback_market = yes
	}
	else_if = {
		limit = { cbp_market_runtime_blocked_trigger = yes }
		cbp_note_promoted_market_live_blocked_market = yes
		cbp_note_q8_7_live_global_owner_blocked_market = yes
		cbp_note_runtime_blocked_market = yes
	}
}

cbp_run_monthly_q8_7_global_market_local_cycle_once = {
	$country$ = {
		save_temporary_scope_as = cbp_q8_7_live_global_owner_controller
	}

	save_temporary_scope_value_as = {
		name = cbp_q8_7_live_global_owner_month_stamp
		value = {
			value = current_year
			multiply = 12
			add = current_month
		}
	}

	if = {
		limit = {
			OR = {
				NOT = { has_global_variable = cbp_q8_7_live_global_owner_month_stamp }
				global_var:cbp_q8_7_live_global_owner_month_stamp != scope:cbp_q8_7_live_global_owner_month_stamp
			}
		}
		set_global_variable = {
			name = cbp_q8_7_live_global_owner_month_stamp
			value = scope:cbp_q8_7_live_global_owner_month_stamp
		}
		cbp_reset_q8_7_live_global_owner_metrics = yes
		cbp_prepare_monthly_promoted_market_live_dispatcher_metrics = yes
		cbp_note_promoted_market_live_dispatcher_run = yes
		cbp_note_q8_7_live_global_owner_run = yes

		every_market_in_world = {
			save_temporary_scope_as = cbp_market
			save_temporary_scope_as = cbp_market_country_cache_market
			cbp_q8_7_run_global_market_local_owner_market = {
				country = scope:cbp_q8_7_live_global_owner_controller
				market = scope:cbp_market
			}
		}
	}
	else = {
		cbp_note_q8_7_live_global_owner_skip_run = yes
	}
}

cbp_run_monthly_stock_cycle_q8_7_owner_switch = {
	if = {
		limit = { cbp_stock_runtime_ready_trigger = yes }
		cbp_prepare_performance_mode_human_relevant_markets = yes
		cbp_run_monthly_capacity_refresh_for_current_country = yes
		cbp_prepare_monthly_market_seen_registry = yes
		cbp_prepare_human_relevant_full_ledger_markets = yes
		save_temporary_scope_as = cbp_country
		if = {
			limit = { cbp_q8_7_live_global_market_owner_enabled_trigger = yes }
			cbp_run_monthly_q8_7_global_market_local_cycle_once = {
				country = scope:cbp_country
			}
		}
		else = {
			cbp_run_monthly_promoted_market_local_cycle = yes
		}
		cbp_run_monthly_country_trade_owner_cycle = yes
		cbp_note_promoted_market_live_trade_owner_pass = yes
		cbp_run_monthly_us04_reconciliation_for_current_country = yes
		if = {
			limit = { cbp_audit_enabled_trigger = yes }
			cbp_run_monthly_stock_reconciliation_once = yes
		}
	}
	else = {
		if = {
			limit = { cbp_debug_capture_enabled_trigger = yes }
			set_variable = {
				name = cbp_debug_last_monthly_capacity_refresh_gate_passed
				value = 0
			}
		}
	}
}
