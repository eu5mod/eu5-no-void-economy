# PR126 PR6 country trade-owner pass.
#
# This file intentionally does not switch the monthly dispatcher yet. It adds a
# country-scoped trade pass surface that can be tested before PR7 wires it into
# the live monthly order.
#
# TECH-01 047 confirms every_trade as a country-scope iterator. TECH-01 056/138
# confirm that native trade_volume is already the traded-goods quantity.
#
#   goods_quantity = trade_volume
# The trade owner is the trade-scope `owner` scope link, not an implicit copy of
# the country that happens to run the iterator. The country pulse/controller still
# schedules the pass, but attribution must come from trade.owner.
#
# Vanilla trade always crosses distinct source and target markets. The situation
# `source_market == target_market` is impossible under vanilla mechanics and is
# deliberately not represented here. Same-market stock consumption belongs to the
# local non-trade demand branch, not to the country trade-owner pass.
#
# PR6 still does not mutate stock from native vanilla trade objects because the
# stock-affecting generated per-good dispatch is deliberately deferred; stock
# consequences continue to go through explicit ModeU5 requests, existing US-10
# inter-market handlers, and central stock operators.
#
# #107 extension: US-17/#105, #120, and US-20/#118 route reconciliation is wired
# into the existing every_trade body after route scopes and quantity are captured.
# This preserves the #161 Q8.7 boundary: no market-scope every_trade loop is
# introduced. The hook is dormant unless the CMM trade-rework setting enables
# cbp_trade_rework_enabled_trigger.
#
# US-17 is refreshed once from country scope before the route loop. The route
# scope selects import or export efficiency through is_export and applies the
# operation-aware maintenance delta. US-20 remains the goods-only follow-up.

cbp_reset_country_trade_owner_pass_metrics = {
	remove_global_variable = cbp_country_trade_owner_pass_runs
	remove_global_variable = cbp_country_trade_owner_trades_seen
	remove_global_variable = cbp_country_trade_owner_inter_market_trades
	remove_global_variable = cbp_country_trade_owner_owner_mismatches
	remove_global_variable = cbp_country_trade_owner_quantity_confirmed_trades
	remove_global_variable = cbp_country_trade_owner_quantity_blocked_trades
	remove_global_variable = cbp_country_trade_owner_explicit_requests
	remove_global_variable = cbp_country_trade_owner_inter_market_transfers
	remove_global_variable = cbp_country_trade_owner_requested_quantity
	remove_global_variable = cbp_country_trade_owner_confirmed_trade_quantity
	remove_global_variable = cbp_country_trade_owner_transferred_quantity
	remove_global_variable = cbp_country_trade_owner_unsatisfied_quantity

	set_global_variable = { name = cbp_country_trade_owner_pass_runs value = 0 }
	set_global_variable = { name = cbp_country_trade_owner_trades_seen value = 0 }
	set_global_variable = { name = cbp_country_trade_owner_inter_market_trades value = 0 }
	set_global_variable = { name = cbp_country_trade_owner_owner_mismatches value = 0 }
	set_global_variable = { name = cbp_country_trade_owner_quantity_confirmed_trades value = 0 }
	set_global_variable = { name = cbp_country_trade_owner_quantity_blocked_trades value = 0 }
	set_global_variable = { name = cbp_country_trade_owner_explicit_requests value = 0 }
	set_global_variable = { name = cbp_country_trade_owner_inter_market_transfers value = 0 }
	set_global_variable = { name = cbp_country_trade_owner_requested_quantity value = 0 }
	set_global_variable = { name = cbp_country_trade_owner_confirmed_trade_quantity value = 0 }
	set_global_variable = { name = cbp_country_trade_owner_transferred_quantity value = 0 }
	set_global_variable = { name = cbp_country_trade_owner_unsatisfied_quantity value = 0 }
}

	set_global_variable = {
		name = cbp_country_trade_owner_pass_runs
		value = { value = global_var:cbp_country_trade_owner_pass_runs add = 1 }
	}
}

	save_temporary_scope_value_as = { name = cbp_country_trade_owner_unsatisfied_delta value = { value = $unsatisfied_quantity$ min = 0 } }
	set_global_variable = {
		name = cbp_country_trade_owner_transferred_quantity
		value = { value = global_var:cbp_country_trade_owner_transferred_quantity add = scope:cbp_country_trade_owner_transferred_delta }
	}
	set_global_variable = {
		name = cbp_country_trade_owner_unsatisfied_quantity
		value = { value = global_var:cbp_country_trade_owner_unsatisfied_quantity add = scope:cbp_country_trade_owner_unsatisfied_delta }
	}
}

cbp_capture_country_trade_owner_trade_quantity = {
	save_temporary_scope_value_as = {
		name = cbp_trade_owner_trade_volume
		value = {
			value = trade_volume
			min = 0
		}
	}
	save_temporary_scope_value_as = { name = cbp_trade_owner_quantity_matched value = 0 }
	save_temporary_scope_value_as = { name = cbp_trade_owner_goods_quantity value = 0 }
	if = {
		limit = { exists = scope:cbp_trade_owner_good }
		save_temporary_scope_value_as = { name = cbp_trade_owner_goods_quantity value = scope:cbp_trade_owner_trade_volume }
		save_temporary_scope_value_as = { name = cbp_trade_owner_quantity_matched value = 1 }
	}

	if = {
		limit = {
			scope:cbp_trade_owner_quantity_matched > 0
			scope:cbp_trade_owner_goods_quantity > 0
		}
		cbp_note_country_trade_owner_quantity_confirmed_trade = yes
	}
	else = {
		cbp_note_country_trade_owner_quantity_blocked_trade = yes
	}
}

# Country-scoped native trade-owner pass. It classifies trades reached from the
# current country iterator, but attributes each trade through the trade-scope
# `owner` link. The current country is retained only as the controller/scheduler
# so tests can detect whether country-scoped every_trade ever yields a trade whose
# owner differs from the country that scheduled the pass.
cbp_run_monthly_country_trade_owner_cycle = {
	save_temporary_scope_as = cbp_trade_owner_controller_country
	cbp_note_country_trade_owner_pass_run = yes
	cbp_refresh_us17_native_profit_modifiers_for_current_country = yes
	if = {
		limit = { cbp_trade_rework_enabled_trigger = yes }
		cbp_prepare_trade_efficiency_reconciliation_runtime_metrics_once = yes
		cbp_reset_trade_efficiency_reconciliation_accumulator = yes
	}

	every_trade = {
		save_temporary_scope_as = cbp_trade_owner_trade
		cbp_note_country_trade_owner_trade_seen = yes

		owner = { save_temporary_scope_as = cbp_trade_owner_country }
		from_market = { save_temporary_scope_as = cbp_trade_owner_source_market }
		to_market = { save_temporary_scope_as = cbp_trade_owner_target_market }
		traded_goods = { save_temporary_scope_as = cbp_trade_owner_good }
		cbp_capture_country_trade_owner_trade_quantity = yes
		if = {
			limit = { exists = scope:cbp_trade_owner_country }
			cbp_run_us17_operation_aware_route_profit_reconciliation = yes
			scope:cbp_trade_owner_country = {
				if = {
					limit = { cbp_trade_rework_enabled_trigger = yes }
					save_temporary_scope_value_as = { name = cbp_trade_efficiency_formula_inputs_available value = 0 }
					cbp_run_us20_route_loss_reconciliation = yes
				}
			}
		}

		if = {
			limit = {
				NOT = { scope:cbp_trade_owner_country = scope:cbp_trade_owner_controller_country }
			}
			cbp_note_country_trade_owner_owner_mismatch = yes
		}

		cbp_note_country_trade_owner_inter_market_trade = yes
	}
}

# Explicit-request handler for tests and future ModeU5-owned trade requests.
# PR6 only models inter-market trade transfer here. Same-market local consumption
# is intentionally absent because vanilla trade cannot have source_market equal
# to target_market.
cbp_process_country_trade_owner_explicit_request = {
	$buyer_country$ = { save_temporary_scope_as = cbp_country_trade_owner_buyer_country }
	$source_market$ = { save_temporary_scope_as = cbp_country_trade_owner_source_market }
	$target_market$ = { save_temporary_scope_as = cbp_country_trade_owner_target_market }
	goods:$good$ = { save_temporary_scope_as = cbp_good }

	cbp_note_country_trade_owner_explicit_request = { requested_quantity = $requested_quantity$ }
	cbp_note_country_trade_owner_inter_market_trade = yes
	cbp_handoff_scoped_inter_market_transfer = {
		buyer_country = scope:cbp_country_trade_owner_buyer_country
		source_market = scope:cbp_country_trade_owner_source_market
		target_market = scope:cbp_country_trade_owner_target_market
		good = $good$
		requested_quantity = $requested_quantity$
	}
	cbp_note_country_trade_owner_inter_market_transfer = yes
	cbp_note_country_trade_owner_outcome_quantities = {
		transferred_quantity = scope:cbp_transferred_quantity
		unsatisfied_quantity = scope:cbp_unsatisfied_quantity
	}
}
