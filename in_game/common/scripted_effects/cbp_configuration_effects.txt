# ModeU5 package selection is owned by the launcher playset.
# Community Mod Framework configures diagnostic/performance behavior.
# Local developer runtime mode is generated from .cbp.local.env by
# tools/generate_local_runtime_config.sh. It must not depend on the EU5 engine
# --debug_mode launch argument.
# Package presence is captured for a new campaign; existing-save package changes are unsupported.

cbp_refresh_nve_main_mode_from_cmm_country_scope = {
	if = {
		limit = {
			NOT = { has_variable_map = cmm }
		}
		cbp_enter_nve_performance_mode = yes
	}
	else_if = {
		limit = {
			NOT = {
				is_key_in_variable_map = {
					name = cmm
					target = flag:no_void_economy__nve_general_gameplay_country_level_stocks_countries_have_own_stocks_settings
				}
			}
		}
		cbp_enter_nve_performance_mode = yes
	}
	else_if = {
		limit = {
			"variable_map(cmm|flag:no_void_economy__nve_general_gameplay_country_level_stocks_countries_have_own_stocks_settings)" = 2
		}
		cbp_enter_nve_normal_mode = yes
	}
	else_if = {
		limit = {
			"variable_map(cmm|flag:no_void_economy__nve_general_gameplay_country_level_stocks_countries_have_own_stocks_settings)" = 3
		}
		cbp_enter_nve_deactivated_mode = yes
	}
	else = {
		cbp_enter_nve_performance_mode = yes
	}
}

cbp_prepare_performance_mode_human_relevant_markets = {
	if = {
		limit = { cbp_performance_mode_enabled_trigger = yes }
		save_temporary_scope_value_as = {
			name = cbp_performance_human_relevant_markets_stamp
			value = {
				value = current_year
				multiply = 12
				add = current_month
			}
		}
		if = {
			limit = {
				OR = {
					NOT = { has_global_variable = cbp_performance_human_relevant_markets_stamp }
					global_var:cbp_performance_human_relevant_markets_stamp != scope:cbp_performance_human_relevant_markets_stamp
					NOT = { has_global_variable_list = cbp_performance_relevant_markets }
				}
			}
			set_global_variable = {
				name = cbp_performance_human_relevant_markets_stamp
				value = scope:cbp_performance_human_relevant_markets_stamp
			}
			cbp_rebuild_human_relevant_markets = yes
		}
	}
}

cbp_prepare_country_market_accounting_decision = {
	$country$ = { save_temporary_scope_as = cbp_accounting_country }
	$market$ = { save_temporary_scope_as = cbp_accounting_market }

	save_temporary_scope_value_as = { name = cbp_detailed_country_market_accounting_enabled value = 0 }
	save_temporary_scope_value_as = { name = cbp_market_level_fallback_required value = 0 }
	save_temporary_scope_value_as = { name = cbp_detailed_country_market_stock_mutation_allowed value = 0 }
	save_temporary_scope_value_as = { name = cbp_accounting_decision_country_is_human value = 0 }
	save_temporary_scope_value_as = { name = cbp_accounting_decision_market_human_relevant value = 0 }

	if = {
		limit = { cbp_nve_deactivated_trigger = yes }
		save_temporary_scope_value_as = { name = cbp_market_level_fallback_required value = 0 }
	}
	else_if = {
		limit = { cbp_nve_normal_mode_trigger = yes }
		save_temporary_scope_value_as = { name = cbp_detailed_country_market_accounting_enabled value = 1 }
	}
	else = {
		scope:cbp_accounting_country = {
			if = {
				limit = { is_ai = no }
				save_temporary_scope_value_as = { name = cbp_accounting_decision_country_is_human value = 1 }
			}
		}
		if = {
			limit = { scope:cbp_accounting_decision_country_is_human > 0 }
			cbp_mark_human_relevant_market_if_present = {
				country = scope:cbp_accounting_country
				market = scope:cbp_accounting_market
			}
		}
		if = {
			limit = {
				has_global_variable_list = cbp_performance_relevant_markets
				is_target_in_global_variable_list = {
					name = cbp_performance_relevant_markets
					target = scope:cbp_accounting_market
				}
			}
			save_temporary_scope_value_as = { name = cbp_accounting_decision_market_human_relevant value = 1 }
		}

		if = {
			limit = {
				scope:cbp_accounting_decision_country_is_human > 0
				scope:cbp_accounting_decision_market_human_relevant > 0
			}
			save_temporary_scope_value_as = { name = cbp_detailed_country_market_accounting_enabled value = 1 }
		}
		else = {
			save_temporary_scope_value_as = { name = cbp_market_level_fallback_required value = 1 }
		}
	}

	if = {
		limit = { cbp_detailed_country_market_accounting_enabled_trigger = yes }
		scope:cbp_accounting_market = { save_temporary_scope_as = cbp_market }
		if = {
			limit = { cbp_detailed_country_market_stock_mutation_allowed_trigger = yes }
			save_temporary_scope_value_as = { name = cbp_detailed_country_market_stock_mutation_allowed value = 1 }
		}
	}
}

cbp_prepare_stock_mutation_accounting_mode = {
	$country$ = { save_temporary_scope_as = cbp_stock_mutation_country }
	$market$ = { save_temporary_scope_as = cbp_stock_mutation_market }

	save_temporary_scope_value_as = { name = cbp_stock_mutation_use_detailed_accounting value = 0 }
	save_temporary_scope_value_as = { name = cbp_stock_mutation_use_market_level_fallback value = 0 }
	save_temporary_scope_value_as = { name = cbp_stock_mutation_blocked value = 0 }
	save_temporary_scope_value_as = { name = cbp_stock_mutation_promotion_attempted value = 0 }
	save_temporary_scope_value_as = { name = cbp_stock_mutation_promotion_succeeded value = 0 }
	save_temporary_scope_value_as = { name = cbp_stock_mutation_gate_result value = 0 }

	cbp_prepare_country_market_accounting_decision = {
		country = scope:cbp_stock_mutation_country
		market = scope:cbp_stock_mutation_market
	}
	scope:cbp_stock_mutation_market = { save_temporary_scope_as = cbp_market }

	if = {
		limit = { cbp_nve_deactivated_trigger = yes }
		save_temporary_scope_value_as = { name = cbp_stock_mutation_blocked value = 1 }
		save_temporary_scope_value_as = { name = cbp_stock_mutation_gate_result value = 3 }
	}
	else_if = {
		limit = { cbp_nve_normal_mode_trigger = yes }
		save_temporary_scope_value_as = { name = cbp_stock_mutation_use_detailed_accounting value = 1 }
		save_temporary_scope_value_as = { name = cbp_stock_mutation_gate_result value = 1 }
	}
	else_if = {
		limit = {
			cbp_performance_mode_enabled_trigger = yes
			scope:cbp_accounting_decision_market_human_relevant > 0
		}
		if = {
			limit = { cbp_detailed_country_market_stock_mutation_allowed_trigger = yes }
			save_temporary_scope_value_as = { name = cbp_stock_mutation_use_detailed_accounting value = 1 }
			save_temporary_scope_value_as = { name = cbp_stock_mutation_gate_result value = 1 }
		}
		else = {
			save_temporary_scope_value_as = { name = cbp_stock_mutation_promotion_attempted value = 1 }
			cbp_promote_market_to_detailed_accounting = {
				market = scope:cbp_stock_mutation_market
			}
			scope:cbp_stock_mutation_market = { save_temporary_scope_as = cbp_market }
			if = {
				limit = { cbp_detailed_country_market_stock_mutation_allowed_trigger = yes }
				save_temporary_scope_value_as = { name = cbp_stock_mutation_use_detailed_accounting value = 1 }
				save_temporary_scope_value_as = { name = cbp_stock_mutation_promotion_succeeded value = 1 }
				save_temporary_scope_value_as = { name = cbp_stock_mutation_gate_result value = 1 }
			}
			else = {
				save_temporary_scope_value_as = { name = cbp_stock_mutation_use_market_level_fallback value = 1 }
				save_temporary_scope_value_as = { name = cbp_stock_mutation_gate_result value = 2 }
			}
		}
	}
	else_if = {
		limit = { cbp_market_level_fallback_required_trigger = yes }
		save_temporary_scope_value_as = { name = cbp_stock_mutation_use_market_level_fallback value = 1 }
		save_temporary_scope_value_as = { name = cbp_stock_mutation_gate_result value = 2 }
	}
	else = {
		save_temporary_scope_value_as = { name = cbp_stock_mutation_blocked value = 1 }
		save_temporary_scope_value_as = { name = cbp_stock_mutation_gate_result value = 3 }
	}
}

cbp_apply_debug_mode_cmm_defaults = {
	# Intentionally no-op.
	# ModeU5 debug runtime is now controlled by generated local config / CMM,
	# not by EU5 engine --debug_mode.
}

cbp_initialize_configuration_state_effect = {
    if = {
        limit = {
            NOT = { has_global_variable = cbp_core_package_loaded }
        }
        set_global_variable = cbp_core_package_loaded
    }

    set_global_variable = {
        name = cbp_core_package_version
        value = 100
    }

    cbp_apply_generated_local_runtime_mode = yes
    cbp_enter_nve_performance_mode = yes
    cbp_prepare_performance_mode_human_relevant_markets = yes

    set_global_variable = {
        name = cbp_debug_level
        value = 0
    }

    cbp_enter_minimal_accounting_persistence = yes
}

cbp_refresh_configuration_from_cmm_country_scope = {
	cbp_apply_debug_mode_cmm_defaults = yes
	cbp_refresh_nve_main_mode_from_cmm_country_scope = yes
	cbp_prepare_performance_mode_human_relevant_markets = yes

	if = {
		limit = {
			has_variable_map = cmm
			is_key_in_variable_map = {
				name = cmm
				target = flag:no_void_economy__nve_debug_audit_debug_audit_debug_messages_settings
			}
			"variable_map(cmm|flag:no_void_economy__nve_debug_audit_debug_audit_debug_messages_settings)" = 3
		}
		cbp_enter_debug_runtime_mode = yes
		set_global_variable = {
			name = cbp_debug_level
			value = 2
		}
	}
	else_if = {
		limit = {
			has_variable_map = cmm
			is_key_in_variable_map = {
				name = cmm
				target = flag:no_void_economy__nve_debug_audit_debug_audit_debug_messages_settings
			}
			"variable_map(cmm|flag:no_void_economy__nve_debug_audit_debug_audit_debug_messages_settings)" = 2
		}
		cbp_enter_debug_runtime_mode = yes
		set_global_variable = {
			name = cbp_debug_level
			value = 1
		}
	}
	else = {
		cbp_apply_generated_local_runtime_mode = yes
		set_global_variable = {
			name = cbp_debug_level
			value = 0
		}
	}

	if = {
		limit = {
			has_variable_map = cmm
			is_key_in_variable_map = {
				name = cmm
				target = flag:no_void_economy__nve_debug_audit_debug_audit_monthly_stock_check_settings

			"variable_map(cmm|flag:no_void_economy__nve_debug_audit_debug_audit_monthly_stock_check_settings)" = 2

		if = {

		}
		else = {
			cbp_enter_audit_runtime_mode = yes
		}

	if = {
		limit = {
			has_variable_map = cmm
			is_key_in_variable_map = {
				name = cmm
				target = flag:no_void_economy__nve_debug_audit_debug_audit_save_mode_settings

		}
