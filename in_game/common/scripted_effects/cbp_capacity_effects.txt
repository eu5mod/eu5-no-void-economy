# US-02 storage-capacity calculation.
#
# Capacity is rebuilt from current world state and persisted once per
# country-market. The expensive country-owned location scan is performed once
# per country as a pooled location/rank capacity calculation, then divided
# across markets present in that country. Each market keeps its own
# merchant/trade-capacity contribution. Generated per-good stock adapters read
# the shared capacity maps. These effects never mutate stock.
#
# Q8.3 adds a monthly country capacity-pool stamp. The public
# cbp_calculate_country_storage_capacity_pool entry point prepares the
# country-wide pool facts once per country/month and reloads them for later
# country-market capacity records.
#
# PERF layer B adds a country-owned map keyed by market:
#
#   cbp_capacity_monthly_stamp_by_market[country, market] = year * 12 + month
#
# The public prepared-pool country-market entry point is now idempotent within a
# month. Forced lifecycle/topology refreshes keep using the raw entry point and
# overwrite both the capacity record and its stamp.

cbp_recalculate_country_market_capacity = {
	$country$ = { save_temporary_scope_as = cbp_country }
	$market$ = { save_temporary_scope_as = cbp_market }
	goods:$good$ = { save_temporary_scope_as = cbp_good }

	cbp_recalculate_country_market_capacity_shared = yes
}

cbp_recalculate_saved_country_market_storage_capacities = {
	cbp_recalculate_country_market_capacity_shared = yes
}

# Forced all-market refresh used by initialization and topology hooks. This path
# deliberately bypasses the monthly country-market gate.
cbp_recalculate_saved_country_storage_capacities = {
	scope:cbp_country = {
		cbp_prepare_country_storage_capacity_pool_once_per_month = yes
		every_market_present_in_country = {
			save_temporary_scope_as = cbp_market
			cbp_recalculate_country_market_capacity_from_prepared_pool_raw_shared = yes
		}
	}
}

# Monthly all-market refresh. The country pool is prepared once, then each
# country-market record independently decides whether this month is already
# current.
cbp_ensure_saved_country_storage_capacities_current_month = {
	scope:cbp_country = {
		cbp_prepare_country_storage_capacity_pool_once_per_month = yes
		every_market_present_in_country = {
			save_temporary_scope_as = cbp_market
			cbp_recalculate_country_market_capacity_from_prepared_pool_shared = yes
		}
	}
}

cbp_recalculate_country_storage_capacities = {
	$country$ = { save_temporary_scope_as = cbp_country }
	scope:cbp_country = {
		cbp_rebuild_country_location_capacity_pool = yes
	}
	cbp_recalculate_saved_country_storage_capacities = yes
}

cbp_run_monthly_capacity_refresh_for_current_country = {
	save_temporary_scope_as = cbp_country
	cbp_ensure_saved_country_storage_capacities_current_month = yes

	save_temporary_scope_value_as = {
		name = cbp_monthly_capacity_refresh_stamp
		value = {
			value = current_year
			multiply = 12
			add = current_month
		}
	}
	set_variable = {
		name = cbp_debug_last_monthly_capacity_refresh_stamp
		value = scope:cbp_monthly_capacity_refresh_stamp
	}
	set_variable = {
		name = cbp_debug_last_monthly_capacity_refresh_gate_passed
		value = 1
	}
}

cbp_calculate_location_rank_storage_capacity = {
	save_temporary_scope_value_as = {
		name = cbp_capacity_current_location_rank_contribution
		value = cbp_storage_capacity_rural_settlement
	}

	if = {
		limit = { is_capital = yes }
		save_temporary_scope_value_as = {
			name = cbp_capacity_current_location_rank_contribution
			value = cbp_storage_capacity_capital
		}
	}
	else_if = {
		limit = { location_rank = location_rank:megalopolis }
		save_temporary_scope_value_as = {
			name = cbp_capacity_current_location_rank_contribution
			value = cbp_storage_capacity_megalopolis
		}
	}
	else_if = {
		limit = { location_rank = location_rank:city }
		save_temporary_scope_value_as = {
			name = cbp_capacity_current_location_rank_contribution
			value = cbp_storage_capacity_city
		}
	}
	else_if = {
		limit = { location_rank = location_rank:town }
		save_temporary_scope_value_as = {
			name = cbp_capacity_current_location_rank_contribution
			value = cbp_storage_capacity_town
		}
	}
}

# Reusable CORE-03 numerator. It captures only the local contribution carried by
# a transferred location. Country-market merchant capacity is recalculated at
# the country-market level and is not duplicated into one location numerator.
cbp_calculate_location_storage_capacity = {
	$country$ = { save_temporary_scope_as = cbp_capacity_country }
	$location$ = { save_temporary_scope_as = cbp_capacity_location }

	scope:cbp_capacity_country = {
		set_local_variable = { name = cbp_location_capacity_base_accumulator value = 0 }
	}

	scope:cbp_capacity_location = {
		cbp_calculate_location_rank_storage_capacity = yes
		scope:cbp_capacity_country = {
			change_local_variable = {
				name = cbp_location_capacity_base_accumulator
				add = scope:cbp_capacity_current_location_rank_contribution
			}
		}
	}

	scope:cbp_capacity_country = {
		save_temporary_scope_value_as = {
			name = cbp_location_capacity_base
			value = local_var:cbp_location_capacity_base_accumulator
		}
		save_temporary_scope_value_as = {
			name = cbp_location_capacity_total
			value = local_var:cbp_location_capacity_base_accumulator
		}
	}
}

cbp_clear_country_storage_capacity_pool_cache = {
	scope:cbp_country = {
		remove_variable = cbp_capacity_pool_monthly_stamp
		remove_variable = cbp_capacity_pool_cached_location_rank_capacity
		remove_variable = cbp_capacity_pool_cached_location_count
		remove_variable = cbp_capacity_pool_cached_market_count
		remove_variable = cbp_capacity_pool_cached_base
		remove_variable = cbp_capacity_pool_cached_total
		remove_variable = cbp_capacity_pool_cached_location_rank_per_market
	}
}

cbp_rebuild_country_location_capacity_pool = {
	scope:cbp_country = {
		cbp_clear_country_storage_capacity_pool_cache = yes
		set_local_variable = { name = cbp_capacity_pool_location_rank_accumulator value = 0 }
		set_local_variable = { name = cbp_capacity_pool_owned_location_count value = 0 }

		every_owned_location = {
			cbp_calculate_location_rank_storage_capacity = yes
			scope:cbp_country = {
				change_local_variable = {
					name = cbp_capacity_pool_location_rank_accumulator
					add = scope:cbp_capacity_current_location_rank_contribution
				}
				change_local_variable = {
					name = cbp_capacity_pool_owned_location_count
					add = 1
				}
			}
		}

		set_variable = {
			name = cbp_location_capacity_pool_total
			value = local_var:cbp_capacity_pool_location_rank_accumulator
		}
		set_variable = {
			name = cbp_location_capacity_pool_location_count
			value = local_var:cbp_capacity_pool_owned_location_count
		}
	}
}

cbp_calculate_country_storage_capacity_pool_raw = {
	scope:cbp_country = {
		set_local_variable = { name = cbp_capacity_pool_market_count value = 0 }

		every_market_present_in_country = {
			scope:cbp_country = {
				change_local_variable = { name = cbp_capacity_pool_market_count add = 1 }
			}
		}

		save_temporary_scope_value_as = { name = cbp_capacity_pool_location_rank_capacity value = 0 }
		if = {
			limit = { has_variable = cbp_location_capacity_pool_total }
			save_temporary_scope_value_as = {
				name = cbp_capacity_pool_location_rank_capacity
				value = var:cbp_location_capacity_pool_total
			}
		}

		save_temporary_scope_value_as = { name = cbp_capacity_pool_location_count value = 0 }
		if = {
			limit = { has_variable = cbp_location_capacity_pool_location_count }
			save_temporary_scope_value_as = {
				name = cbp_capacity_pool_location_count
				value = var:cbp_location_capacity_pool_location_count
			}
		}

		save_temporary_scope_value_as = {
			name = cbp_capacity_pool_market_count
			value = local_var:cbp_capacity_pool_market_count
		}
		save_temporary_scope_value_as = {
			name = cbp_capacity_pool_base
			value = scope:cbp_capacity_pool_location_rank_capacity
		}
		save_temporary_scope_value_as = {
			name = cbp_capacity_pool_total
			value = scope:cbp_capacity_pool_location_rank_capacity
		}

		save_temporary_scope_value_as = { name = cbp_capacity_pool_location_rank_per_market value = 0 }

		if = {
			limit = { scope:cbp_capacity_pool_market_count > 0 }
			save_temporary_scope_value_as = {
				name = cbp_capacity_pool_location_rank_per_market
				value = {
					value = scope:cbp_capacity_pool_location_rank_capacity
					divide = scope:cbp_capacity_pool_market_count
				}
			}
		}
	}
}

cbp_store_country_storage_capacity_pool_cache = {
	scope:cbp_country = {
		set_variable = { name = cbp_capacity_pool_monthly_stamp value = scope:cbp_capacity_pool_current_month_stamp }
		set_variable = { name = cbp_capacity_pool_cached_location_rank_capacity value = scope:cbp_capacity_pool_location_rank_capacity }
		set_variable = { name = cbp_capacity_pool_cached_location_count value = scope:cbp_capacity_pool_location_count }
		set_variable = { name = cbp_capacity_pool_cached_market_count value = scope:cbp_capacity_pool_market_count }
		set_variable = { name = cbp_capacity_pool_cached_base value = scope:cbp_capacity_pool_base }
		set_variable = { name = cbp_capacity_pool_cached_total value = scope:cbp_capacity_pool_total }
		set_variable = { name = cbp_capacity_pool_cached_location_rank_per_market value = scope:cbp_capacity_pool_location_rank_per_market }
	}
}

cbp_load_country_storage_capacity_pool_cache = {
	scope:cbp_country = {
		save_temporary_scope_value_as = { name = cbp_capacity_pool_location_rank_capacity value = var:cbp_capacity_pool_cached_location_rank_capacity }
		save_temporary_scope_value_as = { name = cbp_capacity_pool_location_count value = var:cbp_capacity_pool_cached_location_count }
		save_temporary_scope_value_as = { name = cbp_capacity_pool_market_count value = var:cbp_capacity_pool_cached_market_count }
		save_temporary_scope_value_as = { name = cbp_capacity_pool_base value = var:cbp_capacity_pool_cached_base }
		save_temporary_scope_value_as = { name = cbp_capacity_pool_total value = var:cbp_capacity_pool_cached_total }
		save_temporary_scope_value_as = { name = cbp_capacity_pool_location_rank_per_market value = var:cbp_capacity_pool_cached_location_rank_per_market }
	}
}

cbp_calculate_country_storage_capacity_pool = {
	scope:cbp_country = {
		save_temporary_scope_value_as = {
			name = cbp_capacity_pool_current_month_stamp
			value = {
				value = current_year
				multiply = 12
				add = current_month
			}
		}

		if = {
			limit = {
				OR = {
					NOT = { has_variable = cbp_capacity_pool_monthly_stamp }
					var:cbp_capacity_pool_monthly_stamp != scope:cbp_capacity_pool_current_month_stamp
					NOT = { has_variable = cbp_capacity_pool_cached_location_rank_per_market }
				}
			}
			cbp_calculate_country_storage_capacity_pool_raw = yes
			cbp_store_country_storage_capacity_pool_cache = yes
		}
		else = {
			cbp_load_country_storage_capacity_pool_cache = yes
		}
	}
}

cbp_prepare_country_storage_capacity_pool_once_per_month = {
	cbp_calculate_country_storage_capacity_pool = yes
}

cbp_apply_country_storage_capacity_pool_to_current_market = {
	scope:cbp_country = {
		save_temporary_scope_value_as = {
			name = cbp_capacity_current_market_trade_contribution
			value = {
				value = "scope:cbp_market.merchant_capacity(scope:cbp_country)"
				multiply = cbp_storage_capacity_per_trade_capacity
				min = 0
			}
		}
		save_temporary_scope_value_as = {
			name = cbp_capacity_base
			value = {
				value = scope:cbp_capacity_current_market_trade_contribution
				add = scope:cbp_capacity_pool_location_rank_per_market
			}
		}
		save_temporary_scope_value_as = {
			name = cbp_capacity_building
			value = 0
		}
		save_temporary_scope_value_as = {
			name = cbp_capacity_foreign
			value = 0
		}
		save_temporary_scope_value_as = {
			name = cbp_capacity_trade_capacity
			value = scope:cbp_capacity_current_market_trade_contribution
		}
		save_temporary_scope_value_as = {
			name = cbp_capacity_location_rank_capacity
			value = scope:cbp_capacity_pool_location_rank_per_market
		}
		save_temporary_scope_value_as = {
			name = cbp_capacity_total
			value = {
				value = scope:cbp_capacity_current_market_trade_contribution
				add = scope:cbp_capacity_pool_location_rank_per_market
			}
		}
		save_temporary_scope_value_as = {
			name = cbp_capacity_location_count
			value = scope:cbp_capacity_pool_location_count
		}
		save_temporary_scope_value_as = {
			name = cbp_capacity_market_count
			value = scope:cbp_capacity_pool_market_count
		}
		save_temporary_scope_value_as = {
			name = cbp_capacity_country_pool_total
			value = scope:cbp_capacity_pool_total
		}
	}
}

cbp_calculate_country_market_capacity = {
	cbp_rebuild_country_location_capacity_pool = yes
	cbp_calculate_country_storage_capacity_pool = yes
	cbp_apply_country_storage_capacity_pool_to_current_market = yes
}

cbp_load_capacity_breakdown = {
	scope:cbp_country = {
		if = {
			limit = {
				has_variable_map = cbp_stock_cap_by_market
				is_key_in_variable_map = {
					name = cbp_stock_cap_by_market
					target = scope:cbp_market
				}
			}
			save_temporary_scope_value_as = {
				name = cbp_country_stock_cap_raw
				value = "variable_map(cbp_stock_cap_by_market|scope:cbp_market)"
			}
		}
		else = {
			save_temporary_scope_value_as = { name = cbp_country_stock_cap_raw value = 0 }
		}
		if = {
			limit = {
				has_variable_map = cbp_base_capacity_by_market
				is_key_in_variable_map = {
					name = cbp_base_capacity_by_market
					target = scope:cbp_market
				}
			}
			save_temporary_scope_value_as = {
				name = cbp_country_base_capacity_raw
				value = "variable_map(cbp_base_capacity_by_market|scope:cbp_market)"
			}
		}
		else = {
			save_temporary_scope_value_as = { name = cbp_country_base_capacity_raw value = 0 }
		}
		if = {
			limit = {
				has_variable_map = cbp_building_capacity_by_market
				is_key_in_variable_map = {
					name = cbp_building_capacity_by_market
					target = scope:cbp_market
				}
			}
			save_temporary_scope_value_as = {
				name = cbp_country_building_capacity_raw
				value = "variable_map(cbp_building_capacity_by_market|scope:cbp_market)"
			}
		}
		else = {
			save_temporary_scope_value_as = { name = cbp_country_building_capacity_raw value = 0 }
		}
		if = {
			limit = {
				has_variable_map = cbp_foreign_capacity_by_market
				is_key_in_variable_map = {
					name = cbp_foreign_capacity_by_market
					target = scope:cbp_market
				}
			}
			save_temporary_scope_value_as = {
				name = cbp_country_foreign_capacity_raw
				value = "variable_map(cbp_foreign_capacity_by_market|scope:cbp_market)"
			}
		}
		else = {
			save_temporary_scope_value_as = { name = cbp_country_foreign_capacity_raw value = 0 }
		}
	}
}

cbp_store_capacity_record = {
	scope:cbp_country = {
		if = {
			limit = {
				has_variable_map = cbp_stock_cap_by_market
				is_key_in_variable_map = {
					name = cbp_stock_cap_by_market
					target = scope:cbp_market
				}
			}
			remove_from_variable_map = {
				name = cbp_stock_cap_by_market
				key = scope:cbp_market
			}
		}
		if = {
			limit = { scope:cbp_capacity_total > 0 }
			add_to_variable_map = {
				name = cbp_stock_cap_by_market
				key = scope:cbp_market
				value = scope:cbp_capacity_total
			}
		}

		if = {
			limit = {
				has_variable_map = cbp_base_capacity_by_market
				is_key_in_variable_map = {
					name = cbp_base_capacity_by_market
					target = scope:cbp_market
				}
			}
			remove_from_variable_map = {
				name = cbp_base_capacity_by_market
				key = scope:cbp_market
			}
		}
		if = {
			limit = { scope:cbp_capacity_base > 0 }
			add_to_variable_map = {
				name = cbp_base_capacity_by_market
				key = scope:cbp_market
				value = scope:cbp_capacity_base
			}
		}

		if = {
			limit = {
				has_variable_map = cbp_building_capacity_by_market
				is_key_in_variable_map = {
					name = cbp_building_capacity_by_market
					target = scope:cbp_market
				}
			}
			remove_from_variable_map = {
				name = cbp_building_capacity_by_market
				key = scope:cbp_market
			}
		}
		if = {
			limit = { scope:cbp_capacity_building > 0 }
			add_to_variable_map = {
				name = cbp_building_capacity_by_market
				key = scope:cbp_market
				value = scope:cbp_capacity_building
			}
		}

		if = {
			limit = {
				has_variable_map = cbp_foreign_capacity_by_market
				is_key_in_variable_map = {
					name = cbp_foreign_capacity_by_market
					target = scope:cbp_market
				}
			}
			remove_from_variable_map = {
				name = cbp_foreign_capacity_by_market
				key = scope:cbp_market
			}
		}
		if = {
			limit = { scope:cbp_capacity_foreign > 0 }
			add_to_variable_map = {
				name = cbp_foreign_capacity_by_market
				key = scope:cbp_market
				value = scope:cbp_capacity_foreign
			}
		}
	}
}

# Always read the country-owned stamp map through the saved country scope. The
# idempotent helper is called from both country and market iterator scopes.
cbp_load_country_market_capacity_monthly_stamp = {
	save_temporary_scope_value_as = { name = cbp_capacity_stored_month_stamp value = -1 }
	scope:cbp_country = {
		if = {
			limit = {
				has_variable_map = cbp_capacity_monthly_stamp_by_market
				is_key_in_variable_map = {
					name = cbp_capacity_monthly_stamp_by_market
					target = scope:cbp_market
				}
			}
			save_temporary_scope_value_as = {
				name = cbp_capacity_stored_month_stamp
				value = "variable_map(cbp_capacity_monthly_stamp_by_market|scope:cbp_market)"
			}
		}
	}
}

cbp_store_country_market_capacity_current_month_stamp = {
	cbp_prepare_country_market_capacity_current_month_stamp = yes
	scope:cbp_country = {
		if = {
			limit = {
				has_variable_map = cbp_capacity_monthly_stamp_by_market
				is_key_in_variable_map = {
					name = cbp_capacity_monthly_stamp_by_market
					target = scope:cbp_market
				}
			}
			remove_from_variable_map = {
				name = cbp_capacity_monthly_stamp_by_market

			name = cbp_capacity_monthly_stamp_by_market

}

# Idempotent prepared-pool entry point used by both the current-country monthly
# preparation pass and the detailed promoted-market pass.
cbp_recalculate_country_market_capacity_from_prepared_pool_shared = {
	cbp_prepare_country_market_capacity_current_month_stamp = yes
	cbp_load_country_market_capacity_monthly_stamp = yes
	if = {
		limit = { scope:cbp_capacity_stored_month_stamp != scope:cbp_capacity_current_month_stamp }
		cbp_recalculate_country_market_capacity_from_prepared_pool_raw_shared = yes
	}
}

cbp_rebuild_and_refresh_country_storage_capacities = {
	save_temporary_scope_as = cbp_country
	cbp_rebuild_country_location_capacity_pool = yes
	cbp_recalculate_saved_country_storage_capacities = yes
}

cbp_handle_location_capacity_changed = {
	if = {
		limit = { cbp_stock_runtime_ready_trigger = yes }
		owner ?= {
			cbp_rebuild_and_refresh_country_storage_capacities = yes
		}
	}
}

cbp_handle_capital_moved_capacity = {
	if = {
		limit = { cbp_stock_runtime_ready_trigger = yes }
		cbp_rebuild_and_refresh_country_storage_capacities = yes
	}
}

cbp_finalize_country_market_capacity = {
	save_temporary_scope_value_as = {
		name = cbp_capacity_stock
		value = {
			value = scope:cbp_country_stock_before_raw
			min = 0
		}
