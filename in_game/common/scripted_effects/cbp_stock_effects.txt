# ModeU5 centralized stock mutation effects.
#
# Public calls use a literal goods token so each expanded map name remains static:
#
# cbp_add_stock = {
#     country = root
#     market = root.capital.market
#     good = wheat
#     quantity = 50
#     capacity_policy = enforce
#     only_add_at_country_level = yes
# }
#
# Supported capacity policies are enforce and allow_over_capacity. The latter is
# reserved for CORE-02, CORE-03, separately approved migrations, and explicit
# reconciliation/restoration offsets such as US-04 below-baseline consumption.
#
# only_*_at_country_level = yes keeps the vanilla market-supply ledger untouched.
# only_*_at_country_level = no also applies the matching add_goods_supply delta
# from this central wrapper.

cbp_select_capacity_policy_enforce = {
	save_temporary_scope_value_as = {
		name = cbp_capacity_policy
		value = 0
	}
}

cbp_add_stock = {
	$country$ = { save_temporary_scope_as = cbp_country }
	$market$ = { save_temporary_scope_as = cbp_market }
	goods:$good$ = { save_temporary_scope_as = cbp_good }

	save_temporary_scope_value_as = {
		name = cbp_quantity_requested_raw
		value = $quantity$
	}
	save_temporary_scope_value_as = {
		name = cbp_quantity_requested
		value = {
			value = scope:cbp_quantity_requested_raw
			min = 0
		}
	}
	if = {
		limit = { scope:cbp_quantity_requested_raw < 0 }
		error_log = "ModeU5 cbp_add_stock received a negative quantity; request normalized to zero."
	}

	save_temporary_scope_value_as = { name = cbp_stock_operator_contract_valid value = 0 }
	save_temporary_scope_value_as = { name = cbp_stock_operator_apply_vanilla_market_delta value = 0 }
	cbp_select_stock_operator_country_only_$only_add_at_country_level$ = yes
	if = {
		limit = { scope:cbp_stock_operator_contract_valid > 0 }
		cbp_select_capacity_policy_$capacity_policy$ = yes
		cbp_add_stock_good_$good$ = yes
		if = {
			limit = {
				scope:cbp_stock_operator_apply_vanilla_market_delta > 0
				scope:cbp_actual_added_quantity > 0
			}
			scope:cbp_market = {
				add_goods_supply = {
					goods = goods:$good$
					amount = scope:cbp_actual_added_quantity
				}
			}
		}
	}
	else = {
		save_temporary_scope_value_as = { name = cbp_actual_added_quantity value = 0 }
		save_temporary_scope_value_as = { name = cbp_rejected_quantity value = scope:cbp_quantity_requested }
		save_temporary_scope_value_as = { name = cbp_consistency_validation_required value = 1 }
		error_log = "ModeU5 STOCK CONTRACT ERROR operator=cbp_add_stock reason=missing_or_invalid_only_add_at_country_level"
	}
}

cbp_add_stock_calculate = {
	save_temporary_scope_value_as = {
		name = cbp_country_stock_before
		value = {
			value = scope:cbp_country_stock_before_raw
			min = 0
		}
	}
	save_temporary_scope_value_as = {
		name = cbp_country_stock_cap
		value = {
			value = scope:cbp_country_stock_cap_raw
			min = 0
		}
	}
	save_temporary_scope_value_as = {
		name = cbp_available_capacity
		value = {
			value = scope:cbp_country_stock_cap
			subtract = scope:cbp_country_stock_before
			min = 0
		}
	}
	save_temporary_scope_value_as = {
		name = cbp_over_capacity_before
		value = {
			value = scope:cbp_country_stock_before
			subtract = scope:cbp_country_stock_cap
			min = 0
		}
	}

	if = {
		limit = { scope:cbp_capacity_policy = 1 }
		save_temporary_scope_value_as = {
			name = cbp_actual_added_quantity
			value = scope:cbp_quantity_requested
		}
	}
	else = {
		save_temporary_scope_value_as = {
			name = cbp_actual_added_quantity
			value = {
				value = scope:cbp_quantity_requested
				max = scope:cbp_available_capacity
			}
		}
	}

	save_temporary_scope_value_as = {
		name = cbp_rejected_quantity
		value = {
			value = scope:cbp_quantity_requested
			subtract = scope:cbp_actual_added_quantity
		}
	}
	save_temporary_scope_value_as = {
		name = cbp_country_stock_after
		value = {
			value = scope:cbp_country_stock_before
			add = scope:cbp_actual_added_quantity
		}
	}
	save_temporary_scope_value_as = {
		name = cbp_market_stock_after
		value = {
			value = scope:cbp_market_stock_before
			add = scope:cbp_actual_added_quantity
		}
	}
	save_temporary_scope_value_as = {
		name = cbp_over_capacity_after
		value = {
			value = scope:cbp_country_stock_after
			subtract = scope:cbp_country_stock_cap
			min = 0
		}
	}
	save_temporary_scope_value_as = {
		name = cbp_over_capacity_created
		value = {
			value = scope:cbp_over_capacity_after
			subtract = scope:cbp_over_capacity_before
			min = 0
		}
	}
	save_temporary_scope_value_as = {
		name = cbp_stock_difference_after
		value = {
			value = scope:cbp_market_stock_after
			subtract = scope:cbp_market_stock_before
			subtract = scope:cbp_country_stock_after
			add = scope:cbp_country_stock_before
		}
	}
	save_temporary_scope_value_as = {
		name = cbp_consistency_validation_required
		value = 0
	}

	if = {
		limit = { scope:cbp_country_stock_before_raw < 0 }
		error_log = "ModeU5 cbp_add_stock found a negative country stock; the country record is normalized and consistency validation is required."
		save_temporary_scope_value_as = {
			name = cbp_consistency_validation_required
			value = 1
		}
	}
	if = {
		limit = { scope:cbp_country_stock_cap_raw < 0 }
		error_log = "ModeU5 cbp_add_stock found a negative capacity; capacity is treated as zero."
	}
	if = {
		limit = { scope:cbp_market_stock_before < 0 }
		error_log = "ModeU5 cbp_add_stock found a negative market aggregate; consistency validation is required."
		save_temporary_scope_value_as = {
			name = cbp_consistency_validation_required
			value = 1
		}
	}

	save_temporary_scope_value_as = {
		name = cbp_country_record_write_required
		value = 0
	}
	if = {
		limit = {
			OR = {
				scope:cbp_actual_added_quantity > 0
				scope:cbp_country_stock_before_raw < 0
			}
		}
		save_temporary_scope_value_as = {
			name = cbp_country_record_write_required
			value = 1
		}
	}
	save_temporary_scope_value_as = {
		name = cbp_market_record_write_required
		value = 0
	}
	if = {
		limit = { scope:cbp_actual_added_quantity > 0 }
		save_temporary_scope_value_as = {
			name = cbp_market_record_write_required
			value = 1
		}
	}

}

cbp_remove_stock = {
	$country$ = { save_temporary_scope_as = cbp_country }
	$market$ = { save_temporary_scope_as = cbp_market }
	goods:$good$ = { save_temporary_scope_as = cbp_good }

	save_temporary_scope_value_as = {
		name = cbp_quantity_requested_raw
		value = $quantity$
	}
	save_temporary_scope_value_as = {
		name = cbp_quantity_requested
		value = {
			value = scope:cbp_quantity_requested_raw
			min = 0
		}
	}
	if = {
		limit = { scope:cbp_quantity_requested_raw < 0 }
		error_log = "ModeU5 cbp_remove_stock received a negative quantity; request normalized to zero."
	}

	save_temporary_scope_value_as = { name = cbp_stock_operator_contract_valid value = 0 }
	save_temporary_scope_value_as = { name = cbp_stock_operator_apply_vanilla_market_delta value = 0 }
	cbp_select_stock_operator_country_only_$only_remove_at_country_level$ = yes
	if = {
		limit = { scope:cbp_stock_operator_contract_valid > 0 }
		cbp_select_remove_reason_$reason$ = yes
		cbp_remove_stock_good_$good$ = yes
		if = {
			limit = {
				scope:cbp_stock_operator_apply_vanilla_market_delta > 0
				scope:cbp_actual_removed_quantity > 0
			}
			save_temporary_scope_value_as = {
				name = cbp_stock_operator_vanilla_market_delta
				value = {
					value = scope:cbp_actual_removed_quantity
					multiply = -1
				}
			}
			scope:cbp_market = {
				add_goods_supply = {
					goods = goods:$good$
					amount = scope:cbp_stock_operator_vanilla_market_delta
				}
			}
		}
	}
	else = {
		save_temporary_scope_value_as = { name = cbp_actual_removed_quantity value = 0 }
		save_temporary_scope_value_as = { name = cbp_unsatisfied_quantity value = scope:cbp_quantity_requested }
		save_temporary_scope_value_as = { name = cbp_consistency_validation_required value = 1 }
		error_log = "ModeU5 STOCK CONTRACT ERROR operator=cbp_remove_stock reason=missing_or_invalid_only_remove_at_country_level"
	}
}

		}

	save_temporary_scope_value_as = {

			name = cbp_actual_removed_quantity
			value = 0
		}
		save_temporary_scope_value_as = {
			name = cbp_consistency_validation_required
			value = 1
		}
	}

	save_temporary_scope_value_as = {
		name = cbp_unsatisfied_quantity
		value = {
			value = scope:cbp_quantity_requested
			subtract = scope:cbp_actual_removed_quantity
		}
	}
	save_temporary_scope_value_as = {
		name = cbp_country_stock_after
		value = {
			value = scope:cbp_country_stock_before
			subtract = scope:cbp_actual_removed_quantity
			min = 0
		}

	save_temporary_scope_value_as = {

		name = cbp_available_capacity
		value = {
			value = scope:cbp_country_stock_cap
			subtract = scope:cbp_country_stock_before
			min = 0
		}
	}
	save_temporary_scope_value_as = {
		name = cbp_over_capacity_before
		value = {
			value = scope:cbp_country_stock_before
			subtract = scope:cbp_country_stock_cap
			min = 0
		}
	}
	save_temporary_scope_value_as = {
		name = cbp_over_capacity_after
		value = {
			value = scope:cbp_country_stock_after
			subtract = scope:cbp_country_stock_cap
			min = 0
		}
	}
	save_temporary_scope_value_as = {
		name = cbp_over_capacity_created
		value = 0
	}
	save_temporary_scope_value_as = {
		name = cbp_stock_difference_after
		value = {
			value = scope:cbp_market_stock_after
			subtract = scope:cbp_market_stock_before
			subtract = scope:cbp_country_stock_after
			add = scope:cbp_country_stock_before
		}
	}

	if = {
		limit = { scope:cbp_country_stock_before_raw < 0 }
		error_log = "ModeU5 cbp_remove_stock found a negative country stock; the country record is normalized and consistency validation is required."
		save_temporary_scope_value_as = {
			name = cbp_consistency_validation_required
			value = 1
		}
	}
	if = {
		limit = { scope:cbp_country_stock_cap_raw < 0 }
		error_log = "ModeU5 cbp_remove_stock found a negative capacity; capacity is treated as zero."
	}

	save_temporary_scope_value_as = {
		name = cbp_country_record_write_required
		value = 0
	}
	if = {
		limit = {
			OR = {
				scope:cbp_actual_removed_quantity > 0
				scope:cbp_country_stock_before_raw < 0
			}
		}
		save_temporary_scope_value_as = {

			value = 1
		}
	}

cbp_transfer_stock = {
	$seller_country$ = { save_temporary_scope_as = cbp_seller_country }
	$buyer_country$ = { save_temporary_scope_as = cbp_buyer_country }
	$source_market$ = { save_temporary_scope_as = cbp_source_market }
	$target_market$ = { save_temporary_scope_as = cbp_target_market }
	goods:$good$ = { save_temporary_scope_as = cbp_good }

	save_temporary_scope_value_as = {
		name = cbp_quantity_requested_raw
		value = $quantity$
	}
	save_temporary_scope_value_as = {
		name = cbp_quantity_requested
		value = {
			value = scope:cbp_quantity_requested_raw
			min = 0
		}
	}
	if = {
		limit = { scope:cbp_quantity_requested_raw < 0 }
		error_log = "ModeU5 cbp_transfer_stock received a negative quantity; request normalized to zero."
	}

	cbp_select_capacity_policy_$target_capacity_policy$ = yes
	cbp_transfer_stock_good_$good$ = yes
}

cbp_transfer_stock_calculate = {
	save_temporary_scope_value_as = {
		name = cbp_seller_stock_before
		value = {
			value = scope:cbp_seller_stock_before_raw
			min = 0
		}
	}
	save_temporary_scope_value_as = {
		name = cbp_buyer_stock_before
		value = {
			value = scope:cbp_buyer_stock_before_raw
			min = 0
		}
	}
	save_temporary_scope_value_as = {
		name = cbp_buyer_stock_cap
		value = {
			value = scope:cbp_buyer_stock_cap_raw
			min = 0
		}
	}
	save_temporary_scope_value_as = {
		name = cbp_available_capacity
		value = {
			value = scope:cbp_buyer_stock_cap
			subtract = scope:cbp_buyer_stock_before
			min = 0
		}
	}
	save_temporary_scope_value_as = {
		name = cbp_over_capacity_before
		value = {
			value = scope:cbp_buyer_stock_before
			subtract = scope:cbp_buyer_stock_cap
			min = 0
		}
	}
	save_temporary_scope_value_as = {
		name = cbp_same_market_transfer
		value = 0
	}
	if = {
		limit = { scope:cbp_source_market = scope:cbp_target_market }
		save_temporary_scope_value_as = {
			name = cbp_same_market_transfer
			value = 1
		}
	}
	save_temporary_scope_value_as = {
		name = cbp_invalid_same_record_transfer
		value = 0
	}
	if = {
		limit = {
			scope:cbp_seller_country = scope:cbp_buyer_country
			scope:cbp_source_market = scope:cbp_target_market
		}
		save_temporary_scope_value_as = {
			name = cbp_invalid_same_record_transfer
			value = 1
		}
	}

	if = {
		limit = {
			scope:cbp_capacity_policy = 1
			scope:cbp_invalid_same_record_transfer = 0
		}
		save_temporary_scope_value_as = {
			name = cbp_actual_transferred_quantity
			value = {
				value = scope:cbp_quantity_requested
				max = scope:cbp_seller_stock_before
			}
		}
	}
	else_if = {
		limit = { scope:cbp_invalid_same_record_transfer = 0 }
		save_temporary_scope_value_as = {
			name = cbp_actual_transferred_quantity
			value = {
				value = scope:cbp_quantity_requested
				max = scope:cbp_seller_stock_before
				max = scope:cbp_available_capacity
			}
		}
	}
	else = {
		save_temporary_scope_value_as = {
			name = cbp_actual_transferred_quantity
			value = 0
		}
	}

	save_temporary_scope_value_as = {
		name = cbp_consistency_validation_required
		value = 0
	}
	# Precision note: keep this strict until TECH-01 113 measures engine and map
	# rounding. A gameplay epsilon would change transfer conservation behavior.
	if = {
		limit = {
			scope:cbp_same_market_transfer = 0
			OR = {
				scope:cbp_source_market_stock_before < 0
				scope:cbp_source_market_stock_before < scope:cbp_actual_transferred_quantity
			}
		}
		error_log = "ModeU5 cbp_transfer_stock blocked a source-market aggregate underflow; consistency validation is required."
		save_temporary_scope_value_as = {
			name = cbp_actual_transferred_quantity
			value = 0
		}
		save_temporary_scope_value_as = {
			name = cbp_consistency_validation_required
			value = 1
		}
	}

	save_temporary_scope_value_as = {
		name = cbp_unsatisfied_quantity
		value = {
			value = scope:cbp_quantity_requested
			subtract = scope:cbp_actual_transferred_quantity
		}
	}
	save_temporary_scope_value_as = {
		name = cbp_transferred_quantity
		value = scope:cbp_actual_transferred_quantity
	}
	save_temporary_scope_value_as = {
		name = cbp_seller_stock_after
		value = {
			value = scope:cbp_seller_stock_before
			subtract = scope:cbp_actual_transferred_quantity
			min = 0
		}
	}
	save_temporary_scope_value_as = {
		name = cbp_buyer_stock_after
		value = {
			value = scope:cbp_buyer_stock_before
			add = scope:cbp_actual_transferred_quantity
		}
	}
	save_temporary_scope_value_as = {
		name = cbp_source_market_stock_after
		value = scope:cbp_source_market_stock_before
	}
	save_temporary_scope_value_as = {
		name = cbp_target_market_stock_after
		value = scope:cbp_target_market_stock_before
	}
	if = {
		limit = { scope:cbp_same_market_transfer = 0 }
		save_temporary_scope_value_as = {
			name = cbp_source_market_stock_after
			value = {
				value = scope:cbp_source_market_stock_before
				subtract = scope:cbp_actual_transferred_quantity
			}
		}
		save_temporary_scope_value_as = {
			name = cbp_target_market_stock_after
			value = {
				value = scope:cbp_target_market_stock_before
				add = scope:cbp_actual_transferred_quantity
			}
		}
	}
	save_temporary_scope_value_as = {
		name = cbp_over_capacity_after
		value = {
			value = scope:cbp_buyer_stock_after
			subtract = scope:cbp_buyer_stock_cap
			min = 0
		}
	}
	save_temporary_scope_value_as = {
		name = cbp_over_capacity_created
		value = {
			value = scope:cbp_over_capacity_after
			subtract = scope:cbp_over_capacity_before
			min = 0
		}
	}
	save_temporary_scope_value_as = {
		name = cbp_stock_difference_after
		value = 0
	}

	if = {
		limit = { scope:cbp_seller_stock_before_raw < 0 }
		error_log = "ModeU5 cbp_transfer_stock found a negative seller stock; the seller record is normalized and consistency validation is required."
		save_temporary_scope_value_as = {
			name = cbp_consistency_validation_required
			value = 1
		}
	}
	if = {
		limit = { scope:cbp_buyer_stock_before_raw < 0 }
		error_log = "ModeU5 cbp_transfer_stock found a negative buyer stock; the buyer record is normalized and consistency validation is required."
		save_temporary_scope_value_as = {
			name = cbp_consistency_validation_required
			value = 1
		}
	}
	if = {
		limit = { scope:cbp_buyer_stock_cap_raw < 0 }
		error_log = "ModeU5 cbp_transfer_stock found a negative buyer capacity; capacity is treated as zero."
	}

	save_temporary_scope_value_as = {
		name = cbp_seller_record_write_required
		value = 0
	}
	if = {
		limit = {
			scope:cbp_invalid_same_record_transfer = 0
			OR = {
				scope:cbp_actual_transferred_quantity > 0
				scope:cbp_seller_stock_before_raw < 0
			}
		}
		save_temporary_scope_value_as = {
			name = cbp_seller_record_write_required
			value = 1
		}
	}
	save_temporary_scope_value_as = {
		name = cbp_buyer_record_write_required
		value = 0
	}
	if = {
		limit = {
			scope:cbp_invalid_same_record_transfer = 0
			OR = {
				scope:cbp_actual_transferred_quantity > 0
				scope:cbp_buyer_stock_before_raw < 0
			}
		}
		save_temporary_scope_value_as = {
			name = cbp_buyer_record_write_required
			value = 1
		}
	}
	save_temporary_scope_value_as = {
		name = cbp_market_transfer_write_required
		value = 0
	}

cbp_decay_stock = {
	$country$ = { save_temporary_scope_as = cbp_country }
	$market$ = { save_temporary_scope_as = cbp_market }
	goods:$good$ = { save_temporary_scope_as = cbp_good }

	save_temporary_scope_value_as = {

		value = $decay_rate$
	}
	save_temporary_scope_value_as = {
		name = cbp_decay_rate
		value = {
			value = scope:cbp_decay_rate_raw
			min = 0
			max = 1
		}
	}
	if = {
		limit = {
			OR = {
				scope:cbp_decay_rate_raw < 0
				scope:cbp_decay_rate_raw > 1
			}
		}
		error_log = "ModeU5 cbp_decay_stock received an out-of-range decay rate; rate clamped to [0, 1]."
	}

	cbp_decay_stock_good_$good$ = yes
}

cbp_decay_stock_calculate = {
	save_temporary_scope_value_as = {
		name = cbp_country_stock_before
		value = {
			value = scope:cbp_country_stock_before_raw
			min = 0
		}
	}
	save_temporary_scope_value_as = {
		name = cbp_country_stock_cap
		value = {
			value = scope:cbp_country_stock_cap_raw
			min = 0
		}
	}
	save_temporary_scope_value_as = {
		name = cbp_decayed_quantity
		value = {
			value = scope:cbp_country_stock_before
			multiply = scope:cbp_decay_rate
			max = scope:cbp_country_stock_before
		}
	}
	save_temporary_scope_value_as = {
		name = cbp_consistency_validation_required
		value = 0
	}

	# Precision note: a fractional decay can expose engine rounding here.
	# Validate/rebuild and inspect the raw operands before weakening this guard.
	if = {
		limit = {
			OR = {
				scope:cbp_market_stock_before < 0
				scope:cbp_market_stock_before < scope:cbp_decayed_quantity
			}
		}
		error_log = "ModeU5 cbp_decay_stock blocked a market aggregate underflow; consistency validation is required."
		save_temporary_scope_value_as = {
			name = cbp_decayed_quantity
			value = 0
		}
		save_temporary_scope_value_as = {
			name = cbp_consistency_validation_required
			value = 1
		}
	}

	save_temporary_scope_value_as = {
		name = cbp_unsatisfied_quantity
		value = 0
	}
	save_temporary_scope_value_as = {
		name = cbp_country_stock_after
		value = {
			value = scope:cbp_country_stock_before
			subtract = scope:cbp_decayed_quantity
			min = 0
		}
	}
	save_temporary_scope_value_as = {
		name = cbp_market_stock_after
		value = {
			value = scope:cbp_market_stock_before
			subtract = scope:cbp_decayed_quantity
		}
	}
	save_temporary_scope_value_as = {
		name = cbp_available_capacity
		value = {
			value = scope:cbp_country_stock_cap
			subtract = scope:cbp_country_stock_before
			min = 0
		}
	}
	save_temporary_scope_value_as = {
		name = cbp_over_capacity_before
		value = {
			value = scope:cbp_country_stock_before
			subtract = scope:cbp_country_stock_cap
			min = 0
		}
	}
	save_temporary_scope_value_as = {
		name = cbp_over_capacity_after
		value = {
			value = scope:cbp_country_stock_after
			subtract = scope:cbp_country_stock_cap
			min = 0
		}
	}
	save_temporary_scope_value_as = {
		name = cbp_over_capacity_created
		value = 0
	}
	save_temporary_scope_value_as = {
		name = cbp_stock_difference_after
		value = {
			value = scope:cbp_market_stock_after
			subtract = scope:cbp_market_stock_before
			subtract = scope:cbp_country_stock_after
			add = scope:cbp_country_stock_before
		}
	}

	if = {
		limit = { scope:cbp_country_stock_before_raw < 0 }
		error_log = "ModeU5 cbp_decay_stock found a negative country stock; the country record is normalized and consistency validation is required."
		save_temporary_scope_value_as = {
			name = cbp_consistency_validation_required
			value = 1
		}
	}
	if = {
		limit = { scope:cbp_country_stock_cap_raw < 0 }
		error_log = "ModeU5 cbp_decay_stock found a negative capacity; capacity is treated as zero."
	}

	save_temporary_scope_value_as = {
		name = cbp_country_record_write_required
		value = 0
	}
	if = {
		limit = {
			OR = {
				scope:cbp_decayed_quantity > 0
				scope:cbp_country_stock_before_raw < 0
			}
		}
		save_temporary_scope_value_as = {
			name = cbp_country_record_write_required
			value = 1
		}
	}
	save_temporary_scope_value_as = {
		name = cbp_market_record_write_required
		value = 0
	}
	if = {
		limit = { scope:cbp_decayed_quantity > 0 }
		save_temporary_scope_value_as = {
			name = cbp_market_record_write_required
			value = 1
		}
	}

}

cbp_validate_stock_consistency_prepare = {
	save_temporary_scope_value_as = {
		name = cbp_validation_expected_market_stock
		value = scope:cbp_expected_market_stock
	}
	save_temporary_scope_value_as = {
		name = cbp_validation_actual_market_stock_before
		value = scope:cbp_scanned_market_stock
	}
	save_temporary_scope_value_as = {
		name = cbp_validation_difference_before
		value = {
			value = scope:cbp_validation_actual_market_stock_before
			subtract = scope:cbp_validation_expected_market_stock
		}
	}
	save_temporary_scope_value_as = {
		name = cbp_validation_absolute_difference
		value = scope:cbp_validation_difference_before
	}
	if = {
		limit = { scope:cbp_validation_absolute_difference < 0 }
		save_temporary_scope_value_as = {
			name = cbp_validation_absolute_difference
			value = {
				value = scope:cbp_validation_absolute_difference
				multiply = -1
			}
		}
	}
	save_temporary_scope_value_as = {
		name = cbp_validation_inconsistency_detected
		value = 0
	}
	save_temporary_scope_value_as = {
		name = cbp_validation_inconsistency_severity
		value = 0
	}
	# Exact accounting is intentional. If tiny residuals appear, preserve the
	# signed difference and run the TECH-01 113 precision probe first.
	if = {
		limit = { scope:cbp_validation_difference_before != 0 }
		save_temporary_scope_value_as = {
			name = cbp_validation_inconsistency_detected
			value = 1
		}
		save_temporary_scope_value_as = {
			name = cbp_validation_inconsistency_severity
			value = 1
		}
	}
	if = {
		limit = {
			scope:cbp_validation_absolute_difference >= cbp_stock_consistency_prominent_threshold
		}
		save_temporary_scope_value_as = {
			name = cbp_validation_inconsistency_severity
			value = 2
		}
	}
	save_temporary_scope_value_as = {
		name = cbp_validation_rebuild_called
		value = 0
	}
	save_temporary_scope_value_as = {
		name = cbp_validation_country_source_count
		value = scope:cbp_country_source_count
	}
	save_temporary_scope_value_as = {
		name = cbp_validation_negative_country_source_detected
		value = scope:cbp_negative_country_source_detected
	}
	save_temporary_scope_value_as = {
		name = cbp_validation_over_cap_country_source_detected
		value = scope:cbp_over_cap_country_source_detected
	}
}

cbp_validate_stock_consistency_finalize = {
	save_temporary_scope_value_as = {
		name = cbp_validation_actual_market_stock_after
		value = scope:cbp_scanned_market_stock
	}
	save_temporary_scope_value_as = {
		name = cbp_validation_difference_after
		value = {
			value = scope:cbp_validation_actual_market_stock_after
			subtract = scope:cbp_validation_expected_market_stock
		}
	}
	save_temporary_scope_value_as = {
		name = cbp_validation_failed_after_rebuild
		value = 0
	}
	# A tiny post-write residual may indicate map persistence rounding. Do not
	# hide it with a global epsilon before TECH-01 113 is characterized.
	if = {
		limit = { scope:cbp_validation_difference_after != 0 }
		save_temporary_scope_value_as = {
			name = cbp_validation_failed_after_rebuild
			value = 1
		}
	}
}

	save_temporary_scope_value_as = {
		name = cbp_monthly_market_already_seen
		value = 0
	}

	if = {
		limit = { exists = scope:cbp_market }
		if = {
			limit = {
				NOT = { has_global_variable_list = cbp_monthly_markets_seen_this_cycle }
			}
			add_to_global_variable_list = {
				name = cbp_monthly_markets_seen_this_cycle
				target = scope:cbp_market
			}
			set_global_variable = {
				name = cbp_monthly_markets_seen_new_count
				value = {
					value = global_var:cbp_monthly_markets_seen_new_count
					add = 1
				}
			}
		}
		else_if = {
			limit = {
				NOT = {
					is_target_in_global_variable_list = {
						name = cbp_monthly_markets_seen_this_cycle
						target = scope:cbp_market
					}
				}
			}
			add_to_global_variable_list = {
				name = cbp_monthly_markets_seen_this_cycle
				target = scope:cbp_market
			}
			set_global_variable = {
				name = cbp_monthly_markets_seen_new_count
				value = {
					value = global_var:cbp_monthly_markets_seen_new_count
					add = 1
				}
			}
		}
		else = {
			save_temporary_scope_value_as = {
				name = cbp_monthly_market_already_seen
				value = 1
			}
			set_global_variable = {
				name = cbp_monthly_markets_seen_duplicate_count
				value = {
					value = global_var:cbp_monthly_markets_seen_duplicate_count
					add = 1
				}
			}
		}
	}
	else = {
		error_log = "ModeU5 monthly market registry skipped because no market scope was provided."
	}
}

cbp_fail_initialization = {
	set_global_variable = {
		name = cbp_initialization_state
		value = -1
	}
	set_global_variable = {
		name = cbp_initialization_last_mode
		value = -1
	}
	error_log = "ModeU5 CORE-02 opening stock initialization failed; inspect cbp_initialization_* globals."
}

cbp_repair_current_schema_runtime_marker = {
	if = {
		limit = {
			has_global_variable = cbp_stock_schema_version
			global_var:cbp_stock_schema_version = cbp_current_stock_schema_version
			NOT = { cbp_initialization_complete_trigger = yes }
			NOT = { cbp_initialization_in_progress_trigger = yes }
			NOT = { cbp_initialization_failed_trigger = yes }
		}
		set_global_variable = {
			name = cbp_initialization_state
			value = 2
		}
		set_global_variable = {
			name = cbp_initialization_last_mode
			value = 4
		}
		debug_log = "ModeU5 CORE-02 repaired missing initialization_state for current stock schema."
	}
	else_if = {
		limit = {
			has_global_variable = cbp_stock_schema_version
			global_var:cbp_stock_schema_version = cbp_current_stock_schema_version
			cbp_initialization_failed_trigger = yes
			has_global_variable = cbp_initialization_failure_code
			global_var:cbp_initialization_failure_code = 21
		}
		remove_global_variable = cbp_initialization_failure_detected
		remove_global_variable = cbp_initialization_failure_code
		set_global_variable = {
			name = cbp_initialization_state
			value = 2
		}
		set_global_variable = {
			name = cbp_initialization_last_mode
			value = 5
		}
		debug_log = "ModeU5 CORE-02 repaired legacy current-schema initialization failure code 21."
	}
	else_if = {
		limit = {
			has_global_variable = cbp_stock_schema_version
			global_var:cbp_stock_schema_version = cbp_current_stock_schema_version
			cbp_initialization_failed_trigger = yes
			NOT = { has_global_variable = cbp_initialization_failure_detected }
			NOT = { has_global_variable = cbp_initialization_failure_code }
		}
		set_global_variable = {
			name = cbp_initialization_state
			value = 2
		}
		set_global_variable = {
			name = cbp_initialization_last_mode
			value = 6
		}
		debug_log = "ModeU5 CORE-02 repaired erased legacy current-schema initialization failure marker."
	}
}

cbp_start_game_stock_initialization_dispatcher = {
	cbp_repair_current_schema_runtime_marker = yes

	if = {
		limit = { cbp_stock_runtime_ready_trigger = yes }
		set_global_variable = {
			name = cbp_initialization_last_mode
			value = 2
		}
	}
	else_if = {
		limit = { cbp_initialization_not_started_trigger = yes }
		set_global_variable = {
			name = cbp_initialization_state
			value = 1
		}
		cbp_run_fresh_opening_stock_initialization = yes
		if = {
			limit = { has_global_variable = cbp_initialization_failure_detected }
			cbp_fail_initialization = yes
		}
		else = {
			cbp_commit_successful_initialization = yes
		}
	}
	else_if = {
		limit = { cbp_initialization_in_progress_trigger = yes }
		cbp_initialization_mark_failure = { code = 20 }
		cbp_fail_initialization = yes
	}
	else_if = {
		limit = { cbp_initialization_failed_trigger = yes }
		set_global_variable = {
			name = cbp_initialization_last_mode
			value = -1
		}
		debug_log = "ModeU5 CORE-02 initialization is already failed; startup remains diagnostic-only."
	}
	else_if = {
		limit = {
			has_global_variable = cbp_stock_schema_version
			global_var:cbp_stock_schema_version = cbp_current_stock_schema_version
		}
		cbp_initialization_mark_failure = { code = 21 }
		cbp_fail_initialization = yes
	}
	else = {
		cbp_initialization_mark_failure = { code = 22 }
		cbp_fail_initialization = yes
	}
}

cbp_repair_stock_lifecycle_on_game_load = {
	cbp_initialize_pop_demand_multipliers_once = yes
	cbp_repair_current_schema_runtime_marker = yes

	if = {
		limit = { NOT = { cbp_stock_runtime_ready_trigger = yes } }
		cbp_start_game_stock_initialization_dispatcher = yes
	}
	else = {
		set_global_variable = {
			name = cbp_initialization_last_mode
			value = 3
		}
	}

	cbp_core04_refresh_all_location_market_memory = yes

	set_global_variable = {
		name = cbp_stock_load_repair_version
		value = 1
	}
}

cbp_prepare_reconciliation_controller = {
	if = {
		limit = { NOT = { exists = scope:cbp_reconciliation_controller } }
		remove_global_variable = cbp_reconciliation_controller_selected
		every_country = {
			limit = {
				NOT = { has_global_variable = cbp_reconciliation_controller_selected }
			}
			save_temporary_scope_as = cbp_reconciliation_controller
			set_global_variable = cbp_reconciliation_controller_selected
		}
		remove_global_variable = cbp_reconciliation_controller_selected
	}
}

cbp_reconciliation_prepare = {
	set_local_variable = {
		name = cbp_reconciliation_records_checked
		value = 0
	}
	set_local_variable = {
		name = cbp_reconciliation_inconsistencies_found
		value = 0
	}
	set_local_variable = {
		name = cbp_reconciliation_rebuilds_called
		value = 0
	}
	set_local_variable = {
		name = cbp_reconciliation_failures
		value = 0
	}
}

cbp_reconciliation_accumulate_last_validation = {
	change_local_variable = {
		name = cbp_reconciliation_records_checked
		add = 1
	}
	change_local_variable = {
		name = cbp_reconciliation_inconsistencies_found
		add = scope:cbp_validation_inconsistency_detected
	}
	change_local_variable = {
		name = cbp_reconciliation_rebuilds_called
		add = scope:cbp_validation_rebuild_called
	}
	change_local_variable = {
		name = cbp_reconciliation_failures
		add = scope:cbp_validation_failed_after_rebuild
	}
}

cbp_reconciliation_finalize = {
	save_temporary_scope_value_as = {
		name = cbp_reconciliation_records_checked_result
		value = local_var:cbp_reconciliation_records_checked
	}
	save_temporary_scope_value_as = {
		name = cbp_reconciliation_inconsistencies_found_result
		value = local_var:cbp_reconciliation_inconsistencies_found
	}
	save_temporary_scope_value_as = {
		name = cbp_reconciliation_rebuilds_called_result
		value = local_var:cbp_reconciliation_rebuilds_called
	}
	save_temporary_scope_value_as = {
		name = cbp_reconciliation_failures_result
		value = local_var:cbp_reconciliation_failures
	}

	if = {
		limit = {
			cbp_debug_capture_enabled_trigger = yes
			exists = scope:cbp_reconciliation_controller
		}
		scope:cbp_reconciliation_controller = {
			set_variable = {
				name = cbp_debug_last_reconciliation_type
				value = scope:cbp_reconciliation_type
			}
			set_variable = {
				name = cbp_debug_last_reconciliation_records_checked
				value = scope:cbp_reconciliation_records_checked_result
			}
			set_variable = {
				name = cbp_debug_last_reconciliation_inconsistencies_found
				value = scope:cbp_reconciliation_inconsistencies_found_result
			}
			set_variable = {
				name = cbp_debug_last_reconciliation_rebuilds_called
				value = scope:cbp_reconciliation_rebuilds_called_result
			}
			set_variable = {
				name = cbp_debug_last_reconciliation_failures
				value = scope:cbp_reconciliation_failures_result
			}
			set_variable = {
				name = cbp_debug_last_reconciliation_cycle_stamp
				value = 0
			}
			set_variable = {
				name = cbp_debug_last_reconciliation_initialization_gate_passed
				value = 0
			}
		}
	}

	if = {
		limit = { scope:cbp_reconciliation_failures_result > 0 }
		error_log = "ModeU5 US-11 reconciliation completed with at least one market cache still inconsistent after rebuild."
	}
}

cbp_run_monthly_stock_cycle = {
	if = {
		limit = { cbp_stock_runtime_ready_trigger = yes }
		cbp_prepare_performance_mode_human_relevant_markets = yes
		cbp_run_monthly_capacity_refresh_for_current_country = yes
		cbp_prepare_monthly_market_seen_registry = yes
		cbp_prepare_human_relevant_full_ledger_markets = yes
		save_temporary_scope_as = cbp_country
		cbp_run_monthly_promoted_market_local_cycle = yes
		cbp_run_monthly_country_trade_owner_cycle = yes
		cbp_note_promoted_market_live_trade_owner_pass = yes
		if = {
			limit = { cbp_audit_enabled_trigger = yes }
			cbp_run_monthly_stock_reconciliation_once = yes
		}
	}
	else = {
		if = {
			limit = { cbp_debug_capture_enabled_trigger = yes }
			set_variable = {
				name = cbp_debug_last_monthly_capacity_refresh_gate_passed
				value = 0
			}
		}
	}
}

cbp_run_monthly_stock_reconciliation_once = {
	if = {
		limit = {
			cbp_stock_runtime_ready_trigger = yes
			cbp_audit_enabled_trigger = yes
		}
		save_temporary_scope_as = cbp_reconciliation_controller
		save_temporary_scope_value_as = {
			name = cbp_reconciliation_month_stamp
			value = {
				value = current_year
				multiply = 12
				add = current_month
			}
		}
		if = {
			limit = {
				NOT = { has_global_variable = cbp_last_monthly_reconciliation_stamp }
			}
			set_global_variable = {
				name = cbp_last_monthly_reconciliation_stamp
				value = scope:cbp_reconciliation_month_stamp
			}
			cbp_run_allowed_stock_consistency_validation = yes
			if = {
				limit = { cbp_debug_capture_enabled_trigger = yes }
				set_variable = {
					name = cbp_debug_last_reconciliation_cycle_stamp
					value = scope:cbp_reconciliation_month_stamp
				}
				set_variable = {
					name = cbp_debug_last_reconciliation_initialization_gate_passed
					value = 1
				}
			}
		}
		else_if = {
			limit = {
				global_var:cbp_last_monthly_reconciliation_stamp != scope:cbp_reconciliation_month_stamp
			}
			set_global_variable = {
				name = cbp_last_monthly_reconciliation_stamp
				value = scope:cbp_reconciliation_month_stamp
			}
			cbp_run_allowed_stock_consistency_validation = yes
			if = {
				limit = { cbp_debug_capture_enabled_trigger = yes }
				set_variable = {
					name = cbp_debug_last_reconciliation_cycle_stamp
					value = scope:cbp_reconciliation_month_stamp
				}
				set_variable = {
					name = cbp_debug_last_reconciliation_initialization_gate_passed
					value = 1
				}
			}
		}
	}
}
