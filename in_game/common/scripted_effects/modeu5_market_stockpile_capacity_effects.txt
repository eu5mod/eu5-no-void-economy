# Synchronize vanilla market stockpile capacity with ModeU5's authoritative
# country x market capacity records.
#
# Scope contract:
# - $market$ is the market whose center location receives the modifier;
# - country capacity records must already have been refreshed for the market;
# - the market-center location is scope:modeu5_market.location.

modeu5_refresh_market_center_stockpile_capacity = {
	$market$ = { save_temporary_scope_as = modeu5_market }

	modeu5_prepare_promoted_market_country_cache = {
		market = scope:modeu5_market
	}

	scope:modeu5_market.location = {
		set_local_variable = {
			name = cbp_market_stockpile_capacity_accumulator
			value = 0
		}
	}

	if = {
		limit = { has_global_variable_list = cbp_countries_present_in_market }
		every_in_global_list = {
			variable = cbp_countries_present_in_market
			save_temporary_scope_as = modeu5_market_capacity_country

			if = {
				limit = {
					has_variable_map = cbp_stock_cap_by_market
					is_key_in_variable_map = {
						name = cbp_stock_cap_by_market
						target = scope:modeu5_market
					}
				}
				save_temporary_scope_value_as = {
					name = cbp_market_current_country_capacity
					value = "variable_map(cbp_stock_cap_by_market|scope:modeu5_market)"
				}
				scope:modeu5_market.location = {
					change_local_variable = {
						name = cbp_market_stockpile_capacity_accumulator
						add = scope:cbp_market_current_country_capacity
					}
				}
			}
		}
	}

	scope:modeu5_market.location = {
		remove_location_modifier = modeu5_market_stockpile_capacity
		add_location_modifier = {
			modifier = modeu5_market_stockpile_capacity
			months = 2
			mode = replace
			size = local_var:cbp_market_stockpile_capacity_accumulator
			recalculate_immediately = yes
		}
		remove_local_variable = cbp_market_stockpile_capacity_accumulator
	}
}
