# Callback handler for cmf_on_callback.
# var:cmf_callback is the flag of the setting, alert, or action bar element that was interacted with.
# Scope: country
nve__handle_cmf_callback = {
	switch = {
		trigger = var:cmf_callback
		# General Gameplay (trade_rework_tab) Tab
		## Country Level Stocks (no_void_economy_group) Group
		flag:no_void_economy__only_sold_goods_create_revenue_nve_setting = {
			cmm_sync_dropdown_option_alias = {
				setting = no_void_economy__only_sold_goods_create_revenue_nve_setting
				index = 3
				alias = variable_mapcmmflagno_void_economy__activate_no_void_economy0
			}
			nve_cmf_activate_no_void_economy = yes
		}
		# Economic Balance (balance_tab) Tab
		## Design (economy_group) Group
		flag:no_void_economy__activate_goods_decay_setting = {
			nve_cmf_activate_good_decay = yes
		}
		# Debug & Audit () Tab
		## Debug & Audit (debug_audit_group) Group
		flag:no_void_economy__monthly_stock_check_setting = {
			no_void_economy__monthly_stock_check_setting_on_changed = yes
		}
		flag:no_void_economy__save_mode_setting = {
			no_void_economy__save_mode_setting_on_changed = yes
		}
	}
}
