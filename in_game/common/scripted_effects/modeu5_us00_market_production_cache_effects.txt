modeu5_refresh_us00_vanilla_market_production_cache_good = {
	if = {
		limit = {
			has_global_variable_map = modeu5_$key$_vanilla_produced_markets
			is_key_in_global_variable_map = {
				name = modeu5_$key$_vanilla_produced_markets
				target = scope:modeu5_market
			}
		}
		remove_from_global_variable_map = {
			name = modeu5_$key$_vanilla_produced_markets
			key = scope:modeu5_market
		}
	}
	if = {
		limit = { scope:modeu5_market = { is_produced_in_market = goods:$good$ } }
		add_to_global_variable_map = {
			name = modeu5_$key$_vanilla_produced_markets
			key = scope:modeu5_market
			value = 1
		}
	}
}

modeu5_us10_ui_prepare_current_market_table_from_us00_cache = {
	modeu5_us10_ui_prepare_current_market_table = yes
	modeu5_us10_ui_capture_cached_selected_market_produced_by_market = yes

		debug_log = "ModeU5 US-10-UI FILTER-CACHE market_index=[THIS.GetVariable('modeu5_us10_ui_selected_market_index').GetValue|0] market_count=[THIS.GetVariable('modeu5_us10_ui_market_count').GetValue|0] selected_market_available=[THIS.GetVariable('modeu5_us10_ui_selected_market_available').GetValue|0] countries_present=[THIS.GetVariable('modeu5_us10_ui_countries_present_in_market_count').GetValue|0] cached_rows=[THIS.GetVariable('modeu5_us10_ui_produced_positive_rows').GetValue|0] stock_fallback_rows=[THIS.GetVariable('modeu5_us10_ui_stock_fallback_rows').GetValue|0] wheat_cache=[THIS.GetVariable('modeu5_us10_ui_wheat_is_produced_in_market').GetValue|0] iron_cache=[THIS.GetVariable('modeu5_us10_ui_iron_is_produced_in_market').GetValue|0] cloth_cache=[THIS.GetVariable('modeu5_us10_ui_cloth_is_produced_in_market').GetValue|0]"
	}
}

modeu5_us10_ui_capture_cached_selected_market_produced_by_market = {
	save_temporary_scope_as = modeu5_us10_ui_country
	set_variable = { name = modeu5_us10_ui_produced_positive_rows value = 0 }
	set_variable = { name = modeu5_us10_ui_stock_fallback_rows value = 0 }
	set_variable = { name = modeu5_us10_ui_countries_present_in_market_count value = 0 }

	if = {
		limit = { has_global_variable_list = modeu5_us10_ui_selected_market }
		every_in_global_list = {
			variable = modeu5_us10_ui_selected_market
			save_temporary_scope_as = modeu5_us10_ui_selected_market_scope
			scope:modeu5_us10_ui_selected_market_scope = { save_temporary_scope_as = modeu5_market_country_cache_market }
			modeu5_rebuild_countries_present_in_market = yes

			if = {
				limit = { has_global_variable_list = modeu5_countries_present_in_market }
				every_in_global_list = {
					variable = modeu5_countries_present_in_market
					scope:modeu5_us10_ui_country = {
						set_variable = { name = modeu5_us10_ui_countries_present_in_market_count value = { value = var:modeu5_us10_ui_countries_present_in_market_count add = 1 } }
					}
				}
			}

			modeu5_us10_ui_capture_cached_all_market_produced_rows = yes
		}
	}
}

modeu5_us10_ui_capture_cached_market_produced_row = {
	scope:modeu5_us10_ui_country = {
		set_variable = { name = modeu5_us10_ui_$key$_produced_by_market value = 0 }
		set_variable = { name = modeu5_us10_ui_$key$_ledger_produced_by_market value = 0 }
		set_variable = { name = modeu5_us10_ui_$key$_is_produced_in_market value = 0 }
		set_variable = { name = modeu5_us10_ui_$key$_visible value = 0 }
		set_variable = { name = modeu5_us10_ui_$key$_visibility_source value = 0 }
	}

	if = {
		limit = {
			has_global_variable_map = modeu5_$key$_vanilla_produced_markets
			is_key_in_global_variable_map = {
				name = modeu5_$key$_vanilla_produced_markets
				target = scope:modeu5_us10_ui_selected_market_scope
			}
			"global_variable_map(modeu5_$key$_vanilla_produced_markets|scope:modeu5_us10_ui_selected_market_scope)" > 0
		}
		scope:modeu5_us10_ui_country = {
			set_variable = { name = modeu5_us10_ui_$key$_produced_by_market value = 1 }
			set_variable = { name = modeu5_us10_ui_$key$_is_produced_in_market value = 1 }
			set_variable = { name = modeu5_us10_ui_$key$_visible value = 1 }
			set_variable = { name = modeu5_us10_ui_$key$_visibility_source value = 1 }
			set_variable = { name = modeu5_us10_ui_produced_positive_rows value = { value = var:modeu5_us10_ui_produced_positive_rows add = 1 } }
		}
	}
	else_if = {
		limit = {
			scope:modeu5_us10_ui_country = {
				OR = {
					var:modeu5_us10_ui_$key$_market_stock > 0
					var:modeu5_us10_ui_$key$_country_stock > 0
				}
			}
		}
		scope:modeu5_us10_ui_country = {
			set_variable = { name = modeu5_us10_ui_$key$_visible value = 1 }
			set_variable = { name = modeu5_us10_ui_$key$_visibility_source value = 2 }
			set_variable = { name = modeu5_us10_ui_stock_fallback_rows value = { value = var:modeu5_us10_ui_stock_fallback_rows add = 1 } }
		}
	}
}
