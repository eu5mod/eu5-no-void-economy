# US-10 UI selector overrides.
#
# Keep the player-facing selector aligned with the intended UX:
# - market candidates are markets where the country owns at least one location;
# - good rows are visible only when the selected market produces the good.

modeu5_us10_ui_rebuild_country_market_selector = {
	save_temporary_scope_as = modeu5_us10_ui_country
	modeu5_us10_ui_clear_market_selector_work_list = yes

	set_variable = { name = modeu5_us10_ui_market_count value = 0 }
	set_variable = { name = modeu5_us10_ui_market_iterator_index value = 0 }

	every_owned_location = {
		market = {
			save_temporary_scope_as = modeu5_us10_ui_candidate_market
			if = {
				limit = {
					OR = {
						NOT = { has_global_variable_list = modeu5_us10_ui_market_selector_markets }
						NOT = { is_target_in_global_variable_list = { name = modeu5_us10_ui_market_selector_markets target = scope:modeu5_us10_ui_candidate_market } }
					}
				}
				add_to_global_variable_list = { name = modeu5_us10_ui_market_selector_markets target = scope:modeu5_us10_ui_candidate_market }
				scope:modeu5_us10_ui_country = {
					set_variable = {
						name = modeu5_us10_ui_market_count
						value = { value = var:modeu5_us10_ui_market_count add = 1 }
					}
				}
			}
		}
	}
}

modeu5_us10_ui_capture_good_row = {
	goods:$good$ = { save_temporary_scope_as = modeu5_us10_ui_row_good }

	modeu5_read_country_stock_record = {
		country = scope:modeu5_us10_ui_country
		market = scope:modeu5_us10_ui_selected_market_scope
		good = $good$
	}
	modeu5_read_void_economy_record = {
		country = scope:modeu5_us10_ui_country
		market = scope:modeu5_us10_ui_selected_market_scope
		good = $good$
	}
	scope:modeu5_us10_ui_country = {
		save_temporary_scope_value_as = { name = modeu5_us10_ui_row_market_capacity value = var:modeu5_us10_ui_selected_market_capacity }
		set_variable = { name = modeu5_us10_ui_$key$_country_stock value = scope:modeu5_read_country_stock }
		set_variable = { name = modeu5_us10_ui_$key$_country_capacity value = scope:modeu5_read_country_stock_cap }
		set_variable = { name = modeu5_us10_ui_$key$_market_stock value = scope:modeu5_read_market_stock }
		set_variable = { name = modeu5_us10_ui_$key$_market_capacity value = scope:modeu5_us10_ui_row_market_capacity }
		set_variable = { name = modeu5_us10_ui_$key$_produced value = 0 }
		if = {
			limit = {
				scope:modeu5_us10_ui_selected_market_scope ?= {
					is_produced_in_market = scope:modeu5_us10_ui_row_good
				}
			}
			set_variable = { name = modeu5_us10_ui_$key$_produced value = 1 }
		}
		set_variable = { name = modeu5_us10_ui_$key$_added value = scope:modeu5_us00_added }
		set_variable = { name = modeu5_us10_ui_$key$_rejected value = scope:modeu5_us00_rejected }
		set_variable = { name = modeu5_us10_ui_$key$_country_over_cap value = scope:modeu5_read_over_capacity }
		save_temporary_scope_value_as = { name = modeu5_us10_ui_$key$_market_over_cap_value value = { value = scope:modeu5_read_market_stock subtract = scope:modeu5_us10_ui_row_market_capacity min = 0 } }
		set_variable = { name = modeu5_us10_ui_$key$_market_over_cap value = scope:modeu5_us10_ui_$key$_market_over_cap_value }
		save_temporary_scope_value_as = { name = modeu5_us10_ui_$key$_overproduction_percent_value value = { value = scope:modeu5_us00_read_overproduction_ratio multiply = 100 min = 0 } }
		set_variable = { name = modeu5_us10_ui_$key$_overproduction_percent value = scope:modeu5_us10_ui_$key$_overproduction_percent_value }
		save_temporary_scope_value_as = { name = modeu5_us10_ui_$key$_production_penalty_percent_value value = { value = scope:modeu5_us00_read_production_penalty multiply = 100 } }
		set_variable = { name = modeu5_us10_ui_$key$_production_penalty_percent value = scope:modeu5_us10_ui_$key$_production_penalty_percent_value }
		if = {
			limit = { scope:modeu5_us00_read_overproduction_ratio > 0.05 }
			set_variable = { name = modeu5_us10_ui_$key$_overproduction_warning value = 1 }
		}
		else = { set_variable = { name = modeu5_us10_ui_$key$_overproduction_warning value = 0 } }
	}
}
