# Historical deterministic US-17 / US-20 route-reconciliation scaffold.
#
# Production US-17 uses four native country auto-modifiers plus the operation-
# aware add_gold effect in cbp_trade_owner_modifier_reconciliation_effects.txt.
# It never calls this file's historical add_gold path. These effects remain for
# the combined deterministic fixture and US-20 diagnostics.
#
# Runtime placement:
#   cbp_run_monthly_country_trade_owner_cycle
#     -> every_trade
#        -> save owner/from_market/to_market/traded_goods
#        -> cbp_capture_country_trade_owner_trade_quantity
#        -> cbp_run_us17_us20_route_reconciliation
#
# Scope and merge boundary:
# - fixture-only money reconciliation uses country-scope add_gold;
# - trade_profit / country_income display-accounting remains TECH-01 expected-blocked;
# - US-20 loss is always removed at destination;
# - market-level destination loss uses the central stock helper to apply a negative vanilla market-supply delta;
# - promoted destination base receipt and country-stock loss use receiver country-stock dispatchers.
#
# Four-case support:
# - destination non-promoted: destination market loss only;
# - destination promoted: base receipt/add-transfer + destination market loss + receiver country-stock loss.

cbp_prepare_trade_efficiency_reconciliation_runtime_metrics_once = {
	save_temporary_scope_value_as = {
		name = cbp_trade_efficiency_month_stamp_candidate
		value = { value = current_year multiply = 12 add = current_month }
	}
	if = {
		limit = {
			OR = {
				NOT = { has_global_variable = cbp_trade_efficiency_month_stamp }
				global_var:cbp_trade_efficiency_month_stamp != scope:cbp_trade_efficiency_month_stamp_candidate
			}
		}
		set_global_variable = { name = cbp_trade_efficiency_month_stamp value = scope:cbp_trade_efficiency_month_stamp_candidate }
		remove_global_variable = cbp_trade_efficiency_routes_seen
		remove_global_variable = cbp_trade_efficiency_routes_quantity_confirmed
		remove_global_variable = cbp_trade_efficiency_routes_quantity_blocked
		remove_global_variable = cbp_trade_efficiency_money_delta_routes
		remove_global_variable = cbp_trade_efficiency_money_delta_blocked_routes
		remove_global_variable = cbp_trade_efficiency_treasury_application_routes
		remove_global_variable = cbp_trade_efficiency_treasury_application_blocked_routes
		remove_global_variable = cbp_trade_efficiency_income_application_blocked_routes
		remove_global_variable = cbp_trade_efficiency_income_application_probe_blocked
		remove_global_variable = cbp_trade_efficiency_probe_read_country_income_blocked
		remove_global_variable = cbp_trade_efficiency_probe_read_trade_profit_blocked
		remove_global_variable = cbp_trade_efficiency_probe_add_trade_profit_blocked
		remove_global_variable = cbp_trade_efficiency_probe_country_income_relation_blocked
		remove_global_variable = cbp_trade_efficiency_goods_delta_blocked_routes
		remove_global_variable = cbp_us20_case_origin_unpromoted_destination_unpromoted_routes
		remove_global_variable = cbp_us20_case_origin_promoted_destination_unpromoted_routes
		remove_global_variable = cbp_us20_case_origin_unpromoted_destination_promoted_routes
		remove_global_variable = cbp_us20_case_origin_promoted_destination_promoted_routes
		remove_global_variable = cbp_us20_market_level_remove_good_blocked_routes
		remove_global_variable = cbp_us20_market_goods_supply_loss_routes
		remove_global_variable = cbp_us20_goods_loss_routes
		remove_global_variable = cbp_us20_goods_loss_target_selection_blocked_routes
		remove_global_variable = cbp_us20_receiver_capacity_sort_blocked_routes
		remove_global_variable = cbp_us20_receiver_selected_explicit_routes
		remove_global_variable = cbp_us20_receiver_selected_trade_owner_routes
		remove_global_variable = cbp_us20_receiver_allocator_blocked_routes
		remove_global_variable = cbp_trade_efficiency_owner_missing_routes
	}
}

cbp_ensure_trade_efficiency_reconciliation_accumulator = {
	if = { limit = { NOT = { has_variable = gui_cbp_trade_efficiency_money_delta_total } } set_variable = { name = gui_cbp_trade_efficiency_money_delta_total value = 0 } }
	if = { limit = { NOT = { has_variable = gui_cbp_trade_efficiency_treasury_delta_total } } set_variable = { name = gui_cbp_trade_efficiency_treasury_delta_total value = 0 } }
	if = { limit = { NOT = { has_variable = cbp_trade_efficiency_routes_seen } } set_variable = { name = cbp_trade_efficiency_routes_seen value = 0 } }
	if = { limit = { NOT = { has_variable = cbp_trade_efficiency_money_routes_blocked } } set_variable = { name = cbp_trade_efficiency_money_routes_blocked value = 0 } }
	if = { limit = { NOT = { has_variable = cbp_trade_efficiency_treasury_application_blocked } } set_variable = { name = cbp_trade_efficiency_treasury_application_blocked value = 0 } }
	if = { limit = { NOT = { has_variable = cbp_trade_efficiency_income_application_blocked } } set_variable = { name = cbp_trade_efficiency_income_application_blocked value = 0 } }
	if = { limit = { NOT = { has_variable = cbp_trade_efficiency_goods_routes_blocked } } set_variable = { name = cbp_trade_efficiency_goods_routes_blocked value = 0 } }
}

		}
	}
}

cbp_note_us20_four_case_accounting_path = {
	if = {
		limit = { scope:cbp_us20_origin_market_is_promoted = 0 scope:cbp_us20_target_market_is_promoted = 0 }
		if = { limit = { NOT = { has_global_variable = cbp_us20_case_origin_unpromoted_destination_unpromoted_routes } } set_global_variable = { name = cbp_us20_case_origin_unpromoted_destination_unpromoted_routes value = 1 } }
		else = { change_global_variable = { name = cbp_us20_case_origin_unpromoted_destination_unpromoted_routes add = 1 } }
	}
	else_if = {
		limit = { scope:cbp_us20_origin_market_is_promoted > 0 scope:cbp_us20_target_market_is_promoted = 0 }
		if = { limit = { NOT = { has_global_variable = cbp_us20_case_origin_promoted_destination_unpromoted_routes } } set_global_variable = { name = cbp_us20_case_origin_promoted_destination_unpromoted_routes value = 1 } }
		else = { change_global_variable = { name = cbp_us20_case_origin_promoted_destination_unpromoted_routes add = 1 } }
	}
	else_if = {
		limit = { scope:cbp_us20_origin_market_is_promoted = 0 scope:cbp_us20_target_market_is_promoted > 0 }
		if = { limit = { NOT = { has_global_variable = cbp_us20_case_origin_unpromoted_destination_promoted_routes } } set_global_variable = { name = cbp_us20_case_origin_unpromoted_destination_promoted_routes value = 1 } }
		else = { change_global_variable = { name = cbp_us20_case_origin_unpromoted_destination_promoted_routes add = 1 } }
	}
	else = {
		if = { limit = { NOT = { has_global_variable = cbp_us20_case_origin_promoted_destination_promoted_routes } } set_global_variable = { name = cbp_us20_case_origin_promoted_destination_promoted_routes value = 1 } }
		else = { change_global_variable = { name = cbp_us20_case_origin_promoted_destination_promoted_routes add = 1 } }
	}
}

cbp_compute_us17_us20_route_formula_from_current_values = {
	save_temporary_scope_value_as = { name = gui_cbp_trade_efficiency_route_quantity value = { value = scope:gui_cbp_trade_efficiency_route_quantity min = 0 } }
	save_temporary_scope_value_as = {
		name = gui_cbp_buying_selling_efficiency_clamped
		# The variable name is historical; the business rule is a maximum-capped sum.
		value = { value = scope:gui_cbp_trade_efficiency_buying_efficiency add = scope:gui_cbp_trade_efficiency_selling_efficiency max = 1 }
	}
	save_temporary_scope_value_as = {
		name = gui_cbp_trade_efficiency_sell_side_bonus
		value = { value = scope:gui_cbp_trade_efficiency_route_quantity multiply = scope:gui_cbp_trade_efficiency_sell_price multiply = scope:gui_cbp_trade_efficiency_selling_efficiency }
	}
	save_temporary_scope_value_as = { name = gui_cbp_trade_efficiency_export_cost_factor value = { value = scope:gui_cbp_trade_efficiency_export_cost_modifier add = 1 } }
	save_temporary_scope_value_as = {
		name = gui_cbp_trade_efficiency_buy_side_bonus
		value = { value = scope:gui_cbp_trade_efficiency_route_quantity multiply = scope:gui_cbp_trade_efficiency_buy_price multiply = scope:gui_cbp_trade_efficiency_buying_efficiency multiply = scope:gui_cbp_trade_efficiency_export_cost_factor }
	}
	save_temporary_scope_value_as = { name = gui_cbp_trade_efficiency_old_price_side_bonus value = { value = scope:gui_cbp_trade_efficiency_sell_side_bonus add = scope:gui_cbp_trade_efficiency_buy_side_bonus } }
	cbp_compute_trade_maintenance_efficiency_delta = yes
	cbp_compute_buying_selling_efficiency_delta = yes
	save_temporary_scope_value_as = {
		name = gui_cbp_trade_efficiency_route_money_delta
		value = { value = scope:gui_cbp_trade_efficiency_old_price_side_bonus multiply = -1 add = scope:gui_cbp_trade_maintenance_efficiency_delta add = scope:gui_cbp_buying_selling_efficiency_delta }
	}
	save_temporary_scope_value_as = {
		name = gui_cbp_us20_trade_maintenance_loss_factor
		value = { value = 1 add = { value = scope:gui_cbp_us20_trade_maintenance multiply = -1 } min = 0 max = 1 }
	}
	save_temporary_scope_value_as = {
		name = gui_cbp_us20_goods_received_loss
		value = {
			value = scope:gui_cbp_us20_trade_maintenance_loss_factor
			multiply = cbp_us17_us20_route_loss_coefficient_max
		}
	}
	save_temporary_scope_value_as = {
		name = gui_cbp_us20_target_goods_amount_received
		value = { value = scope:gui_cbp_us20_goods_amount_sent add = { value = scope:gui_cbp_us20_goods_received_loss multiply = -1 } min = 0 }
	}
	save_temporary_scope_value_as = {
		name = gui_cbp_us20_goods_reconciliation_delta
		value = { value = scope:gui_cbp_us20_target_goods_amount_received add = { value = scope:gui_cbp_us20_engine_goods_amount_received multiply = -1 } }
	}
}
