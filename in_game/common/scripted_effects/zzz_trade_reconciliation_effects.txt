# ModeU5 US-17 / US-20 route-local trade reconciliation.
#
# Runtime placement:
#   modeu5_run_monthly_country_trade_owner_cycle
#     -> every_trade
#        -> save owner/from_market/to_market/traded_goods
#        -> modeu5_capture_country_trade_owner_trade_quantity
#        -> modeu5_run_us17_us20_route_reconciliation
#
# This effect computes a route-local reconciliation delta. Applying that delta to
# vanilla remains explicitly separated from the formula:
# - money-side application requires a confirmed country-income or trade-profit API;
# - goods-side delivery has a four-case origin/destination market accounting matrix;
# - the base add/transfer of received goods is distinct from the loss removal;
# - the US-20 loss is always removed at destination, never treated as a transfer.
#
# Four goods-accounting cases:
#   origin non-promoted, destination non-promoted:
#     remove loss at destination market / destination country-market surface.
#   origin promoted, destination non-promoted:
#     remove loss at destination market / destination country-market surface.
#   origin non-promoted, destination promoted:
#     add received goods at destination country-market only, then remove loss at
#     destination market + destination country-market.
#   origin promoted, destination promoted:
#     transfer received goods country-market -> country-market, then remove loss
#     at destination market + destination country-market.
#
# The parent call and this effect both gate on modeu5_trade_rework_enabled_trigger.
# This file deliberately does not reintroduce:
#   every_market_center_in_country -> every_trade

		name = modeu5_trade_efficiency_month_stamp_candidate

		limit = {
			OR = {
				NOT = { has_global_variable = modeu5_trade_efficiency_month_stamp }

		remove_global_variable = modeu5_us20_case_origin_unpromoted_destination_unpromoted_routes
		remove_global_variable = modeu5_us20_case_origin_promoted_destination_unpromoted_routes
		remove_global_variable = modeu5_us20_case_origin_unpromoted_destination_promoted_routes
		remove_global_variable = modeu5_us20_case_origin_promoted_destination_promoted_routes
		remove_global_variable = modeu5_us20_market_level_remove_good_blocked_routes
		remove_global_variable = modeu5_us20_goods_loss_routes

		set_global_variable = { name = modeu5_trade_efficiency_routes_quantity_blocked value = 0 }

		set_global_variable = { name = modeu5_trade_efficiency_goods_delta_blocked_routes value = 0 }
		set_global_variable = { name = modeu5_us20_case_origin_unpromoted_destination_unpromoted_routes value = 0 }

modeu5_probe_trade_income_profit_application_surfaces = {
	# Probe matrix required before applying money deltas to vanilla:
	# 1. read trade-owner country income before route adjustment;
	# 2. read trade route profit before route adjustment;
	# 3. add the route reconciliation delta to trade route profit;
	# 4. read trade-owner country income after route adjustment;
	# 5. verify whether old_country_income = new_country_income - added_route_profit.
	#
	# If that equality holds after mutating trade profit, route profit feeds country
	# income and the delta should be applied to route profit only. If it does not
	# hold, country income and trade-route profit may be separate display/ledger
	# surfaces and the implementation must decide whether to apply both.
	#
	# Do not add speculative effect names here. Replace these blocked counters only
	# after TECH-01 confirms concrete EU5 read/write APIs.
	set_global_variable = { name = modeu5_trade_efficiency_probe_read_country_income_blocked value = 1 }
	set_global_variable = { name = modeu5_trade_efficiency_probe_read_trade_profit_blocked value = 1 }
	set_global_variable = { name = modeu5_trade_efficiency_probe_add_trade_profit_blocked value = 1 }
	set_global_variable = { name = modeu5_trade_efficiency_probe_country_income_relation_blocked value = 1 }
	debug_log = "ModeU5 US17/US20 INCOME_PROFIT_PROBE blocked read_country_income=blocked read_trade_profit=blocked add_trade_profit=blocked equation=old_country_income_equals_new_country_income_minus_added_trade_profit pending=TECH01"
}

	save_temporary_scope_value_as = { name = modeu5_buying_selling_efficiency_delta value = 0 }

modeu5_compute_us17_us20_route_formula_from_current_values = {
	save_temporary_scope_value_as = { name = modeu5_trade_efficiency_route_quantity value = { value = scope:modeu5_trade_efficiency_route_quantity min = 0 } }
	save_temporary_scope_value_as = {
		name = modeu5_buying_selling_efficiency_clamped
		value = { value = scope:modeu5_trade_efficiency_buying_efficiency add = scope:modeu5_trade_efficiency_selling_efficiency divide = 2 min = 0 max = 1 }
	}
	save_temporary_scope_value_as = {
		name = modeu5_trade_efficiency_sell_side_bonus
		value = { value = scope:modeu5_trade_efficiency_route_quantity multiply = scope:modeu5_trade_efficiency_sell_price multiply = scope:modeu5_trade_efficiency_selling_efficiency }
	}
	save_temporary_scope_value_as = { name = modeu5_trade_efficiency_export_cost_factor value = { value = scope:modeu5_trade_efficiency_export_cost_modifier add = 1 } }
	save_temporary_scope_value_as = {
		name = modeu5_trade_efficiency_buy_side_bonus
		value = { value = scope:modeu5_trade_efficiency_route_quantity multiply = scope:modeu5_trade_efficiency_buy_price multiply = scope:modeu5_trade_efficiency_buying_efficiency multiply = scope:modeu5_trade_efficiency_export_cost_factor }
	}
	save_temporary_scope_value_as = { name = modeu5_trade_efficiency_old_price_side_bonus value = { value = scope:modeu5_trade_efficiency_sell_side_bonus add = scope:modeu5_trade_efficiency_buy_side_bonus } }
	modeu5_compute_trade_maintenance_efficiency_delta = yes
	modeu5_compute_buying_selling_efficiency_delta = yes
	save_temporary_scope_value_as = {
		name = modeu5_trade_efficiency_route_money_delta
		value = { value = scope:modeu5_trade_efficiency_old_price_side_bonus multiply = -1 add = scope:modeu5_trade_maintenance_efficiency_delta add = scope:modeu5_buying_selling_efficiency_delta }
	}
	save_temporary_scope_value_as = {
		name = modeu5_us20_trade_maintenance_loss_factor
		value = { value = 1 add = { value = scope:modeu5_us20_trade_maintenance multiply = -1 } min = 0 max = 1 }
	}
	save_temporary_scope_value_as = { name = modeu5_us20_goods_received_loss value = { value = scope:modeu5_us20_trade_maintenance_loss_factor multiply = 0.1 } }
	save_temporary_scope_value_as = {
		name = modeu5_us20_target_goods_amount_received
		value = { value = scope:modeu5_us20_goods_amount_sent add = { value = scope:modeu5_us20_goods_received_loss multiply = -1 } min = 0 }
	}
	save_temporary_scope_value_as = {
		name = modeu5_us20_goods_reconciliation_delta
		value = { value = scope:modeu5_us20_target_goods_amount_received add = { value = scope:modeu5_us20_engine_goods_amount_received multiply = -1 } }
	}
}

modeu5_classify_us20_route_market_accounting_surfaces = {
	# Origin promotion decides whether the base receipt path is add-at-destination or
	# country-market transfer. Destination promotion decides whether a country
	# receiver can exist for the detailed receipt/loss surface.
	save_temporary_scope_value_as = { name = modeu5_us20_origin_market_is_promoted value = 0 }
	save_temporary_scope_value_as = { name = modeu5_us20_target_market_is_promoted value = 0 }
	if = {
		limit = { scope:modeu5_us20_test_origin_market_promoted > 0 }
		save_temporary_scope_value_as = { name = modeu5_us20_origin_market_is_promoted value = 1 }
	}
	else_if = {
		limit = {
			exists = scope:modeu5_trade_owner_source_market
			has_global_variable_list = modeu5_promoted_markets_this_cycle
			is_target_in_global_variable_list = {
				name = modeu5_promoted_markets_this_cycle
				target = scope:modeu5_trade_owner_source_market
			}
		}
		save_temporary_scope_value_as = { name = modeu5_us20_origin_market_is_promoted value = 1 }
	}
	if = {
		limit = { scope:modeu5_us20_test_target_market_promoted > 0 }
		save_temporary_scope_value_as = { name = modeu5_us20_target_market_is_promoted value = 1 }
	}
	else_if = {
		limit = {
			exists = scope:modeu5_trade_owner_target_market
			has_global_variable_list = modeu5_promoted_markets_this_cycle
			is_target_in_global_variable_list = {
				name = modeu5_promoted_markets_this_cycle
				target = scope:modeu5_trade_owner_target_market
			}
		}
		save_temporary_scope_value_as = { name = modeu5_us20_target_market_is_promoted value = 1 }
	}
	modeu5_note_us20_four_case_accounting_path = yes
}

	# Case 1/2 destination non-promoted:
	#   origin promotion does not matter for the loss. Remove at destination market

	modeu5_note_us20_market_level_remove_good_blocked_route = yes

}

	# Receiver-selection contract for promoted destination markets:
	# 1. If a receiving country was stored by the earlier transfer_stock / demand
	#    resolution path, use it.
	# 2. Else, if the trade owner is present in the destination market, use it.
	# 3. Else, reuse/extend US-10 candidate selection with a receiver-allocation
	#    profile rather than a supplier profile.
	#
	# Receiver profile:
	# - ignore minimum-stock and reserve/protection thresholds;
	# - prefer lower fill-ratio / under-capacity countries;
	# - eligible receiver must be under capacity before receipt;
	# - do not clip the receipt to available capacity for MVP.
	if = {
		limit = { exists = scope:modeu5_us20_receiving_country }
		scope:modeu5_us20_receiving_country = { save_temporary_scope_as = modeu5_us20_goods_loss_target_country }
	}
	else_if = {
		limit = { exists = scope:modeu5_trade_owner_country scope:modeu5_us20_trade_owner_present_in_target_market > 0 }
		scope:modeu5_trade_owner_country = { save_temporary_scope_as = modeu5_us20_goods_loss_target_country }
	}
	else = {
		modeu5_note_us20_receiver_capacity_sort_blocked_route = yes
		modeu5_note_us20_goods_loss_target_selection_blocked_route = yes
	}
}

	#   destination market + destination country-market.
	# Case 4: origin promoted, destination promoted:
	#   transfer received goods country-market -> country-market, then remove loss
	#   at destination market + destination country-market.
