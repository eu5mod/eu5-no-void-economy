# ModeU5 US-17 trade-owner country modifier inputs and formula path.
#
# Scope contract:
# - modeu5_capture_trade_owner_country_modifier_inputs runs in country scope;
# - that country must be scope:modeu5_trade_owner_country;
# - import_efficiency, selling_efficiency, and merchant_maintenance_efficiency
#   are country modifiers owned by the trade owner, never by the scheduler
#   country, market-center owner, source market, or target market;
# - US-17's semantic buying-efficiency input maps to EU5 import_efficiency.
#
# Base merchant maintenance is not a country modifier. It is calculated from the
# route's trade_volume and the effective loaded define:
#
#   trade_volume * define:NCountry|MERCHANT_MAINTENANCE_COST
#
# NVE may override that define. Reading it here automatically uses the loaded
# mod value and avoids duplicating it in scripted effects.
#
# The buying/selling average has only an upper bound of 1. Negative values are
# preserved and therefore create a negative maintenance saving (an added cost).

modeu5_capture_trade_owner_country_modifier_inputs = {
	save_temporary_scope_value_as = {
		name = gui_cbp_trade_efficiency_buying_efficiency
		value = modeu5_trade_owner_buying_efficiency
	}
	save_temporary_scope_value_as = {
		name = gui_cbp_trade_efficiency_selling_efficiency
		value = modeu5_trade_owner_selling_efficiency
	}
	save_temporary_scope_value_as = {
		name = gui_cbp_trade_efficiency_merchant_maintenance_efficiency
		value = modeu5_trade_owner_merchant_maintenance_efficiency
	}
	save_temporary_scope_value_as = {
		name = gui_cbp_trade_efficiency_base_maintenance_unit_cost
		value = modeu5_trade_base_merchant_maintenance_cost
	}
	save_temporary_scope_value_as = {
		name = gui_cbp_trade_efficiency_base_maintenance_amount
		value = {
			value = scope:cbp_trade_owner_trade_volume
			multiply = scope:gui_cbp_trade_efficiency_base_maintenance_unit_cost
			min = 0
		}
	}
	save_temporary_scope_value_as = {
		name = cbp_trade_efficiency_country_modifier_inputs_available
		value = 1
	}
}

# #105 maintenance-side route effect:
#
# base_maintenance_amount =
#   trade_volume * define:NCountry|MERCHANT_MAINTENANCE_COST
#
# current_maintenance_factor =
#   max(0, 1 - merchant_maintenance_efficiency)
#
# adjusted_base_maintenance =
#   base_maintenance_amount * current_maintenance_factor
#
# maintenance_saving =
#   adjusted_base_maintenance * capped_average_buying_selling_efficiency
#
# A negative buying/selling average is intentionally retained, producing a
# negative saving and therefore an added route cost.
modeu5_compute_trade_maintenance_efficiency_delta_from_owner_modifiers = {
	save_temporary_scope_value_as = {
		name = gui_cbp_trade_efficiency_merchant_maintenance_factor
		value = {
			value = 1
			add = {
				value = scope:gui_cbp_trade_efficiency_merchant_maintenance_efficiency
				multiply = -1
			}
			min = 0
		}
	}
	save_temporary_scope_value_as = {
		name = gui_cbp_trade_efficiency_adjusted_base_maintenance_amount
		value = {
			value = scope:gui_cbp_trade_efficiency_base_maintenance_amount
			multiply = scope:gui_cbp_trade_efficiency_merchant_maintenance_factor
		}
	}
	save_temporary_scope_value_as = {
		name = gui_cbp_trade_maintenance_efficiency_delta
		value = {
			value = scope:gui_cbp_trade_efficiency_adjusted_base_maintenance_amount
			multiply = scope:gui_cbp_buying_selling_efficiency_clamped
		}
	}
}

modeu5_compute_us17_us20_route_formula_from_owner_modifiers = {
	save_temporary_scope_value_as = {
		name = gui_cbp_trade_efficiency_route_quantity
		value = {
			value = scope:gui_cbp_trade_efficiency_route_quantity
			min = 0
		}
	}

	# Maximum-only cap: values below zero remain below zero.
	save_temporary_scope_value_as = {
		name = gui_cbp_buying_selling_efficiency_clamped
		value = {
			value = scope:gui_cbp_trade_efficiency_buying_efficiency
			add = scope:gui_cbp_trade_efficiency_selling_efficiency
			divide = 2
			max = 1
		}
	}

	save_temporary_scope_value_as = {
		name = gui_cbp_trade_efficiency_sell_side_bonus
		value = {
			value = scope:gui_cbp_trade_efficiency_route_quantity
			multiply = scope:gui_cbp_trade_efficiency_sell_price
			multiply = scope:gui_cbp_trade_efficiency_selling_efficiency
		}
	}
	save_temporary_scope_value_as = {
		name = gui_cbp_trade_efficiency_export_cost_factor
		value = {
			value = scope:gui_cbp_trade_efficiency_export_cost_modifier
			add = 1
		}
	}
	save_temporary_scope_value_as = {
		name = gui_cbp_trade_efficiency_buy_side_bonus
		value = {
			value = scope:gui_cbp_trade_efficiency_route_quantity
			multiply = scope:gui_cbp_trade_efficiency_buy_price
			multiply = scope:gui_cbp_trade_efficiency_buying_efficiency
			multiply = scope:gui_cbp_trade_efficiency_export_cost_factor
		}
	}
	save_temporary_scope_value_as = {
		name = gui_cbp_trade_efficiency_old_price_side_bonus
		value = {
			value = scope:gui_cbp_trade_efficiency_sell_side_bonus
			add = scope:gui_cbp_trade_efficiency_buy_side_bonus
		}
	}

	modeu5_compute_trade_maintenance_efficiency_delta_from_owner_modifiers = yes
	save_temporary_scope_value_as = {
		name = gui_cbp_buying_selling_efficiency_delta
		value = 0
	}
	save_temporary_scope_value_as = {
		name = gui_cbp_trade_efficiency_route_money_delta
		value = {
			value = scope:gui_cbp_trade_efficiency_old_price_side_bonus
			multiply = -1
			add = scope:gui_cbp_trade_maintenance_efficiency_delta
			add = scope:gui_cbp_buying_selling_efficiency_delta
		}
	}

	# US-20 remains based on its separately preserved route maintenance input.
	save_temporary_scope_value_as = {
		name = gui_cbp_us20_trade_maintenance_loss_factor
		value = {
			value = 1
			add = {
				value = scope:gui_cbp_us20_trade_maintenance
				multiply = -1
			}
			min = 0
			max = 1
		}
	}
	save_temporary_scope_value_as = {
		name = gui_cbp_us20_goods_received_loss
		value = {
			value = scope:gui_cbp_us20_trade_maintenance_loss_factor
			multiply = 0.1
		}
	}
	save_temporary_scope_value_as = {
		name = gui_cbp_us20_target_goods_amount_received
		value = {
			value = scope:gui_cbp_us20_goods_amount_sent
			add = {
				value = scope:gui_cbp_us20_goods_received_loss
				multiply = -1
			}
			min = 0
		}
	}
	save_temporary_scope_value_as = {
		name = gui_cbp_us20_goods_reconciliation_delta
		value = {
			value = scope:gui_cbp_us20_target_goods_amount_received
			add = {
				value = scope:gui_cbp_us20_engine_goods_amount_received
				multiply = -1
			}
		}
	}
}

modeu5_compute_trade_efficiency_route_money_delta_from_owner_modifiers = {
	if = {
		limit = {
			scope:cbp_trade_efficiency_formula_inputs_available > 0
			scope:cbp_trade_efficiency_country_modifier_inputs_available > 0
		}
		modeu5_compute_us17_us20_route_formula_from_owner_modifiers = yes
	}
	else = {
		modeu5_note_trade_efficiency_money_delta_blocked_route = yes
		save_temporary_scope_value_as = { name = gui_cbp_trade_maintenance_efficiency_delta value = 0 }
		save_temporary_scope_value_as = { name = gui_cbp_trade_efficiency_old_price_side_bonus value = 0 }
		save_temporary_scope_value_as = { name = gui_cbp_buying_selling_efficiency_delta value = 0 }
		save_temporary_scope_value_as = { name = gui_cbp_trade_efficiency_route_money_delta value = 0 }
	}
}

# Live owner-modifier route wrapper. It reuses the established US-17/US-20
# accounting, stock, receiver, and diagnostics effects; only the owner-modifier
# input and formula path differ from the historical deterministic scaffold.
modeu5_run_us17_us20_route_reconciliation_from_owner_modifiers = {
	if = {
		limit = { modeu5_trade_rework_enabled_trigger = yes }
		modeu5_prepare_trade_efficiency_reconciliation_runtime_metrics_once = yes
		modeu5_note_trade_efficiency_route_seen = yes
		modeu5_clear_trade_efficiency_route_values = yes
		modeu5_prepare_trade_efficiency_route_quantity_input = yes
		if = {
			limit = {
				scope:cbp_trade_owner_quantity_matched > 0
				scope:cbp_trade_owner_goods_quantity > 0
			}
			modeu5_note_trade_efficiency_quantity_confirmed = yes
			modeu5_compute_trade_efficiency_route_money_delta_from_owner_modifiers = yes
			modeu5_accumulate_trade_efficiency_route_delta_to_trade_owner = yes

			# Preserve the gross route reconciliation, then apply only the country
			# treasury share after Crown-power and trade-income allocation.
			save_temporary_scope_value_as = {
				name = cbp_trade_efficiency_gross_route_money_delta
				value = scope:gui_cbp_trade_efficiency_route_money_delta
			}
			save_temporary_scope_value_as = {
				name = cbp_trade_efficiency_crown_power_share
				value = "estate_power(estate_type:crown_estate)"
			}
			save_temporary_scope_value_as = {
				name = cbp_trade_efficiency_trade_income_modifier
				value = modifier:trade_income
			}
			save_temporary_scope_value_as = {
				name = gui_cbp_trade_efficiency_route_money_delta
				value = {
					value = scope:cbp_trade_efficiency_gross_route_money_delta
					multiply = scope:cbp_trade_efficiency_crown_power_share
					multiply = scope:cbp_trade_efficiency_trade_income_modifier
				}
			}
			modeu5_apply_trade_efficiency_income_reconciliation_to_trade_owner = yes
			save_temporary_scope_value_as = {
				name = gui_cbp_trade_efficiency_route_money_delta
				value = scope:cbp_trade_efficiency_gross_route_money_delta
			}

			modeu5_compute_us20_goods_received_delta = yes
			modeu5_apply_us20_goods_delta_to_target_market_good = yes
		}
		else = {
			modeu5_note_trade_efficiency_quantity_blocked = yes
		}
	}
}
