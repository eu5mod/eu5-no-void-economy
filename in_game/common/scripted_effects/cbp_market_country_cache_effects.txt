# ModeU5 PERF-03 market -> countries-present work cache.
#
# EU5 exposes country -> every_market_present_in_country and market ->
# every_location_in_market, but no confirmed persistent per-market variable-list
# owner. Build a deduplicated work cache for the current market from its
# locations, then consume that list inside market-driven effects instead of
# scanning every_country.
#
# Q8.5 promotes the existing dirty writer/consumer surface from probe-only
# evidence to a guarded runtime repair consumer. Dirty lists remain scheduling
# state only; they do not create durable per-market country-list storage.
# PERF-03 counters and rebuild markers are observability-only and write only in
# debug/audit runtime; the countries-present and dirty-market work lists remain
# active in every runtime mode.

cbp_clear_countries_present_in_market_work_cache = {
	if = {
		limit = { cbp_observability_enabled_trigger = yes }
		remove_global_variable = cbp_market_country_cache_locations_scanned
		remove_global_variable = cbp_market_country_cache_locations_with_owner
		remove_global_variable = cbp_market_country_cache_countries_added
		remove_global_variable = cbp_market_country_cache_last_rebuild_started
		remove_global_variable = cbp_market_country_cache_last_rebuild_failed
		remove_global_variable = cbp_market_country_cache_last_rebuild_empty

		set_global_variable = { name = cbp_market_country_cache_locations_scanned value = 0 }
		set_global_variable = { name = cbp_market_country_cache_locations_with_owner value = 0 }
		set_global_variable = { name = cbp_market_country_cache_countries_added value = 0 }
	}

	if = {
		limit = { has_global_variable_list = cbp_countries_present_in_market }
		clear_global_variable_list = cbp_countries_present_in_market
	}
}

cbp_increment_market_country_cache_location_scan = {
	if = {
		limit = { cbp_observability_enabled_trigger = yes }
		set_global_variable = {
			name = cbp_market_country_cache_locations_scanned
			value = {
				value = global_var:cbp_market_country_cache_locations_scanned
				add = 1
			}
		}
	}
}

cbp_increment_market_country_cache_location_with_owner = {
	if = {
		limit = { cbp_observability_enabled_trigger = yes }
		set_global_variable = {
			name = cbp_market_country_cache_locations_with_owner
			value = {
				value = global_var:cbp_market_country_cache_locations_with_owner
				add = 1
			}
		}
	}
}

cbp_increment_market_country_cache_country_added = {
	if = {
		limit = { cbp_observability_enabled_trigger = yes }
		set_global_variable = {
			name = cbp_market_country_cache_countries_added
			value = {
				value = global_var:cbp_market_country_cache_countries_added
				add = 1
			}
		}
	}
}

cbp_add_country_to_current_market_country_cache = {
	if = {
		limit = {
			NOT = { has_global_variable_list = cbp_countries_present_in_market }
		}
		add_to_global_variable_list = {
			name = cbp_countries_present_in_market
			target = scope:cbp_market_country_cache_country
		}
		cbp_increment_market_country_cache_country_added = yes
	}
	else_if = {
		limit = {
			NOT = {
				is_target_in_global_variable_list = {
					name = cbp_countries_present_in_market
					target = scope:cbp_market_country_cache_country
				}
			}
		}
		add_to_global_variable_list = {
			name = cbp_countries_present_in_market
			target = scope:cbp_market_country_cache_country
		}
		cbp_increment_market_country_cache_country_added = yes
	}
}

cbp_rebuild_countries_present_in_market = {
	cbp_clear_countries_present_in_market_work_cache = yes
	if = {
		limit = { cbp_observability_enabled_trigger = yes }
		set_global_variable = cbp_market_country_cache_last_rebuild_started
	}

	if = {
		limit = { exists = scope:cbp_market_country_cache_market }
		scope:cbp_market_country_cache_market = {
			every_location_in_market = {
				cbp_increment_market_country_cache_location_scan = yes
				owner ?= {
					cbp_increment_market_country_cache_location_with_owner = yes
					save_temporary_scope_as = cbp_market_country_cache_country
					cbp_add_country_to_current_market_country_cache = yes
				}
			}
		}
		if = {
			limit = {
				cbp_observability_enabled_trigger = yes
				NOT = { has_global_variable_list = cbp_countries_present_in_market }
			}
			set_global_variable = cbp_market_country_cache_last_rebuild_empty
		}
	}
	else = {
		if = {
			limit = { cbp_observability_enabled_trigger = yes }
			set_global_variable = cbp_market_country_cache_last_rebuild_failed
		}
		error_log = "ModeU5 PERF-03 market-country cache rebuild skipped because no market scope was provided."
	}
}

cbp_mark_market_country_cache_dirty = {
	if = {
		limit = {
			exists = scope:cbp_market_country_cache_dirty_market
			NOT = { has_global_variable_list = cbp_market_country_cache_dirty_markets }
		}
		add_to_global_variable_list = {
			name = cbp_market_country_cache_dirty_markets
			target = scope:cbp_market_country_cache_dirty_market
		}
	}
	else_if = {
		limit = {
			exists = scope:cbp_market_country_cache_dirty_market
			NOT = {
				is_target_in_global_variable_list = {
					name = cbp_market_country_cache_dirty_markets
					target = scope:cbp_market_country_cache_dirty_market
				}
			}
		}
		add_to_global_variable_list = {
			name = cbp_market_country_cache_dirty_markets
			target = scope:cbp_market_country_cache_dirty_market
		}
	}
}

cbp_mark_current_market_country_cache_dirty = {
	save_temporary_scope_as = cbp_market_country_cache_dirty_market
	cbp_mark_market_country_cache_dirty = yes
}

cbp_repair_dirty_market_country_caches = {
	if = {
		limit = { cbp_observability_enabled_trigger = yes }
		remove_global_variable = cbp_market_country_cache_dirty_repair_count
		set_global_variable = { name = cbp_market_country_cache_dirty_repair_count value = 0 }
	}

	if = {
		limit = { has_global_variable_list = cbp_market_country_cache_dirty_markets }
		every_in_global_list = {
			variable = cbp_market_country_cache_dirty_markets
			save_temporary_scope_as = cbp_market_country_cache_market
			cbp_rebuild_countries_present_in_market = yes
			if = {
				limit = { cbp_observability_enabled_trigger = yes }
				set_global_variable = {
					name = cbp_market_country_cache_dirty_repair_count
					value = {
						value = global_var:cbp_market_country_cache_dirty_repair_count
						add = 1
					}
				}
			}
		}
		clear_global_variable_list = cbp_market_country_cache_dirty_markets
	}
}

cbp_repair_dirty_market_country_caches_if_needed = {
	if = {
		limit = { has_global_variable_list = cbp_market_country_cache_dirty_markets }
		cbp_repair_dirty_market_country_caches = yes
	}
}
