# Shared CBP aggregation point for country policy and government-reform changes.
# New Core consumers of either hook must join this dispatcher instead of
# declaring another feature-specific on_policy_changed or on_reform_change.

on_policy_changed = {
	on_actions = {
		cbp_country_governance_changed
	}
}

on_reform_change = {
	on_actions = {
		cbp_country_governance_changed
	}
}

cbp_country_governance_changed = {
	effect = {
		if = {
			limit = { scope_type = country }
			# Monthly refresh remains authoritative for research, temporary
			# modifiers, and sources without a dedicated confirmed hook.
			cbp_refresh_us17_native_profit_modifiers_for_current_country = yes
		}
	}
}
