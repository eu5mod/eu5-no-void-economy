on_location_changed_owner = {
	on_actions = {
		cbp_core03_probe_location_changed_owner
	}
}

cbp_core03_probe_location_changed_owner = {
	effect = {
		cbp_core03_handle_location_changed_owner = yes
		# Re-evaluate the permanent owner-core Control floor immediately after the
		# owner transition. The effect itself verifies the current owner and core.
		cbp_enforce_owner_core_control_floor = yes
		cbp_us04_handle_location_changed_owner = yes
		# The exposure probe observes all live location-owner changes during a long
		# revalidation run. Seeing the same location twice is not enough evidence of
		# a duplicate engine callback; it can also be a later legitimate ownership
		# change. Keep the hook/scope validation, but do not let the old lifetime
		# seen-location list turn repeated locations into hard failures.
		if = {
			limit = { has_global_variable_list = cbp_core03_probe_seen_locations }
			clear_global_variable_list = cbp_core03_probe_seen_locations
		}
		cbp_core03_probe_record_location_changed_owner = yes
	}
}
