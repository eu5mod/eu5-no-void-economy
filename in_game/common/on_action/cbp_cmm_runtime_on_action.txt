# Non-generated CMM runtime overlay.
# Keep warning/reset behavior out of nve__cmm_* files because the CMM Visual Editor rewrites them.

cmf_on_mod_registration = {
	on_actions = {
		cbp_cmm_runtime_registration_pulse
	}
}

cbp_cmm_runtime_registration_pulse = {
	effect = {
		cbp_cmm_register_review_pop = yes
		cbp_cmm_mark_planned_services_restricted = yes
		cbp_cmm_refresh_nve_main_enabled = yes
	}
}

cmf_on_callback = {
	on_actions = {
		cbp_cmm_runtime_callback_pulse
	}
}

cbp_cmm_runtime_callback_pulse = {
	effect = {
		cbp_handle_cmm_runtime_callback = yes
	}
}
