on_game_start = {
	on_actions = {
		delay = { days = 1 }
		cbp_start_game_stock_initialization_pulse
	}
}

on_game_load = {
	on_actions = {
		delay = { days = 1 }
		cbp_load_game_stock_initialization_pulse
	}
}

cbp_start_game_stock_initialization_pulse = {
	effect = {
		# This delayed on-action may run from an empty/root scope. Keep country-scoped
		# CMM reads and location traversal out of this pulse; it only writes the
		# global US-04 initialization version marker. Country-owned locations are
		# initialized lazily from monthly/yearly country pulses.
		cbp_initialize_pop_demand_multipliers_once = yes
		cbp_advance_us04_sparse_index_load_generation = yes
		cbp_start_game_stock_initialization_dispatcher = yes
		cbp_core04_refresh_all_location_market_memory = yes
	}
}

cbp_load_game_stock_initialization_pulse = {
	effect = {
		# Load repair is idempotent. It fills missing startup markers, but it does
		# not overwrite a ready schema or revive a failed/incompatible schema.
		# Advancing the sparse-index generation makes each country rebuild its
		# country-owned US-04 lists exactly once on its next monthly/yearly pulse.
		cbp_advance_us04_sparse_index_load_generation = yes
		cbp_repair_stock_lifecycle_on_game_load = yes
	}
}

monthly_country_pulse = {
	on_actions = {
		cbp_monthly_stock_cycle_pulse
	}
}

yearly_country_pulse = {
	on_actions = {
		cbp_yearly_pop_demand_adaptation_pulse
	}
}

on_location_changed_rank = {
	on_actions = {
		cbp_capacity_location_rank_changed_pulse
	}
}

cbp_capacity_location_rank_changed_pulse = {
	effect = {
		cbp_handle_location_capacity_changed = yes
		# Rank is one of the floor inputs, so enforce the new threshold immediately
		# in the existing hardcoded rank-change hook.
		cbp_enforce_owner_core_control_floor = yes
	}
}

on_capital_moved = {
	on_actions = {
		cbp_capacity_capital_moved_pulse
	}
}

cbp_capacity_capital_moved_pulse = {
	effect = {
		cbp_handle_capital_moved_capacity = yes
	}
}

cbp_monthly_stock_cycle_pulse = {
	effect = {
		cbp_initialize_pop_demand_multipliers_for_current_country_once = yes
		cbp_refresh_pop_demand_live_integration_from_cmm_country_scope = yes
		cbp_run_monthly_stock_cycle_q8_7_owner_switch = yes
		# Monthly CORE-04 memory remains required, but owner-core Control-floor
		# evaluation is deliberately excluded from this high-frequency traversal.
		cbp_core04_refresh_current_country_location_market_memory_monthly = yes
	}
}

cbp_monthly_stock_reconciliation_pulse = {
	effect = {
		cbp_run_monthly_stock_reconciliation_once = yes
	}
}

cbp_yearly_pop_demand_adaptation_pulse = {
	effect = {
		cbp_initialize_pop_demand_multipliers_for_current_country_once = yes
		cbp_refresh_pop_demand_live_integration_from_cmm_country_scope = yes
		cbp_run_yearly_pop_demand_adaptation_for_current_country = yes
		# No confirmed integration/core-status on-action exists. Reuse the combined
		# CORE-04 location traversal once per year as the low-frequency fallback.
		cbp_core04_refresh_current_country_location_market_memory = yes
	}
}

four_yearly_country_pulse = {
	on_actions = {
		cbp_four_yearly_stock_reconciliation_pulse
	}
}

cbp_four_yearly_stock_reconciliation_pulse = {
	effect = {
		cbp_run_four_yearly_stock_reconciliation_once = yes
	}
}
